country_event = {
	#Support from Our Brothers Abroad?
	id = SPS.9
	immediate = {
		log = "[GetDateText]: [Root.GetName]: event SPS.9"
	}
	title = SPS.9.t
	desc = SPS.9.d
	picture = GFX_report_event_iberia_soldier
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = SPS.9.a
		if = {
			limit = {
				ITA = {
					has_country_leader = {
						name = "Carlo Scorza"
					}
				}
			}
			ITA = {
				country_event = {
					id = SPS.17
					days = 1
				}
			}
		}
		if = {
			limit = {
				GER = {
					NOT = {
						OR = {
							has_country_leader = {
								name = "Ulrike Meinhof"
							}
							has_country_leader = {
								name = "Horst Mahler"
							}
						}
					}
				}
			}
			GER = {
				country_event = {
					id = SPS.18
					days = 1
				}
			}
		}
		hidden_effect = {
			if = {
				limit = {
					has_country_flag = SPS_pillaged
				}
				country_event = {
					id = SPS.22
					days = 13
				}
			}
		}
	}
}
