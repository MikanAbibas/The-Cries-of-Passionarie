# NSDAP compromise malus removal - fires 90 days after compromise

	id = GER_reichstag_voting.1
	immediate = { log = "[GetDateText]: [Root.GetName]: event GER_reichstag_voting.1" }
	title = GER_reichstag_voting.1.t
	desc = GER_reichstag_voting.1.d

	is_triggered_only = yes
	hidden = yes

	option = {
		name = GER_reichstag_voting.1.a
		subtract_from_variable = { GER_reichstag_law_modifier_war_support_factor = GER_compromise_nsdap_stab_applied }
		subtract_from_variable = { GER_reichstag_law_modifier_army_morale_factor = GER_compromise_nsdap_pp_applied }
	}
}

# Vote cooldown check - fires 61 days after a vote concludes; tells the player if a new vote can start
country_event = {
	id = GER_reichstag_voting.8
	immediate = { log = "[GetDateText]: [Root.GetName]: event GER_reichstag_voting.8" }
	title = GER_reichstag_voting.8.t
	desc = GER_reichstag_voting.8.d

	is_triggered_only = yes
	hidden = yes

	option = {
		name = GER_reichstag_voting.8.a
		if = {
			limit = {
				has_country_flag = GER_reichstag_law_active
				NOT = { has_country_flag = GER_reichstag_vote_cooldown }
				NOT = { has_country_flag = GER_reichstag_vote_in_progress }
				NOT = { has_country_flag = GER_reichstag_suspended }
			}
			country_event = { id = GER_reichstag_voting.9 }
		}
	}
}
