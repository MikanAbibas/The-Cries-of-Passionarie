country_event = { #skirmishes with Romanian border guards fires for hungary
	id = ZZZ_balkan_war.1
	immediate = {log = "[GetDateText]: [Root.GetName]: event ZZZ_balkan_war.1"}
	title = ZZZ_balkan_war.1.t
	desc = ZZZ_balkan_war.1.d
	picture = GFX_report_event_Generic_Papers
	is_triggered_only = yes
	
	option = {
		name = ZZZ_balkan_war.1.a
		#HUN remove negative modifiers
		set_country_flag = ZZZ_prepping_for_war
		ROM = {
			country_event = {
				id = ZZZ_balkan_war.2
				days = 2
			}
		}
		add_threat = 10
		remove_ideas = HUN_suffocated_economy
		ROM = { save_global_event_target_as = BALKAN_opponent }
		# A belligerent only breaks with its bloc if the enemy is a bloc partner. Fighting the
		# other camp's proxy costs it nothing - that is what makes this a proxy war.
		BALKAN_sever_if_former_ally = yes
		# An overlord only releases both when it owns both sides of the quarrel.
		if = {
			limit = {
				ROM = { is_subject_of = GER }
				HUN = { is_subject_of = GER }
			}
			GER = {
				end_puppet = ROM
				end_puppet = HUN
			}
		}
		if = {
			limit = {
				ROM = { is_subject_of = ITA }
				HUN = { is_subject_of = ITA }
			}
			ITA = {
				end_puppet = ROM
				end_puppet = HUN
			}
		}
	}
}

country_event = { # Opens the Balkan crisis, 40 days after the German Civil War begins.
	# Hidden: the player learns of it through RA_GLOBAL_NEWS.49, not a popup.
	# Fired from GER_GCW_begin (common/scripted_effects/TNO_Germany_scripted_effects.txt).
	id = ZZZ_balkan_war.13
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event ZZZ_balkan_war.13 (Balkan crisis opens)"
		set_global_flag = ZZZ_balkan_troubles
		# Opens the border-war window: the decade of single-state contests before the
		# crisis proper. Closed again by the Crisis dispatcher when it is built.
		set_global_flag = BALKAN_BORDER_WARS
		set_variable = { global.BALKAN_tension = 0 }
		# Kossovo and Montenegro open to Belgrade once Yugoslavia forms; the date is the
		# fallback so a Serbia that never unites still gets its chance.
		if = {
			limit = { has_global_flag = CRO_yugoslavia }
			set_global_flag = BALKAN_italian_claims_open
		}
		else = {
			random_country = {
				limit = { tag = SER }
				country_event = { id = balkanborder.4 days = 900 }
			}
		}
		# Who sponsors whom, from the Great Game rather than hardcoded sides.
		BALKAN_resolve_backing = yes
		every_country = {
			news_event = RA_GLOBAL_NEWS.49
		}
	}
}
