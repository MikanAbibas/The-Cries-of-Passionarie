### chain the rework's decisions fire comes across on its own.

	id = IEA_decol.1

}

		name = IEA_decol.10.a
		hidden_effect = {
			SML = {

			if = {
				limit = {

				limit = {
					check_variable = { SML.decol_favor = 3 }

country_event = { #overall decolonisation sucesful
	id = IEA_decol.100
	immediate = {log = "[GetDateText]: [Root.GetName]: event IEA_decol.100"}
	title = IEA_decol.100.t
	desc = IEA_decol.100.d
	picture = GFX_report_event_italian_plane

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = IEA_decol.100.a
		clr_country_flag = DemItaly_IEA_Decolonization
		mark_focus_tree_layout_dirty = yes

		if = {
			limit = {
				has_country_flag = ITA_salvage_the_project_extra_effect
			}
			custom_effect_tooltip = ITA_salvage_the_project_extra_effect_tt

			set_temp_variable = { Region_i = 2 }
			set_temp_variable = { EconValueChange_v = 8 }
			ITA_EmpireManagement_EconValueChange = yes
			set_temp_variable = { Region_i = 3 }
			set_temp_variable = { EconValueChange_v = 8 }
			ITA_EmpireManagement_EconValueChange = yes
			set_temp_variable = { Region_i = 4 }
			set_temp_variable = { EconValueChange_v = 8 }
			ITA_EmpireManagement_EconValueChange = yes
			set_temp_variable = { Region_i = 5 }
			set_temp_variable = { EconValueChange_v = 8 }
			ITA_EmpireManagement_EconValueChange = yes
		}

		if = {
			limit = {
				has_country_flag = ITA_summon_the_colonials_extra_effect
			}
			custom_effect_tooltip = ITA_summon_the_colonials_extra_effect_tt

			#set up an array of the regions tags to loop through
			clear_temp_array = IEA_tags
			add_to_array = { IEA_tags = ERT }
			add_to_array = { IEA_tags = SML }
			add_to_array = { IEA_tags = ORM }
			add_to_array = { IEA_tags = KEN }

			for_each_scope_loop = {
				array = IEA_tags

				if = {
					limit = {
						check_variable = { decol_negotiation_status = 3 } #negotiations succesful
					}
					
					give_military_access = ITA
					ITA = {
						diplomatic_relation = {
							country = PREV
							relation = docking_rights
							active = yes
						}
					}
				}
			}
		}
	}
}
