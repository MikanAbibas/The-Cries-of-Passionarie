# New Scramble for Africa (z_plans/SCRAMBLE_FOR_AFRICA_PLAN.md). Titles only: the text is the user's.

# SFA.1: the hidden start (plan 4.0). Hosted on the USA, 160 days after the Oberkommando collapse
# (MZB_REICHSTAAT.8071 / .809). The debug decision fires it at once.
country_event = {
	id = SFA.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event SFA.1"
		if = {
			limit = { NOT = { has_global_flag = SFA_active } }
			if = {
				# only possible after Brudermord: wait for the war's end
				limit = { has_global_flag = south_african_war_happening }
				country_event = { id = SFA.1 days = 30 }
			}
			else_if = {
				# an OFN total victory gets its own mechanics (plan 3.3)
				limit = { has_global_flag = SAW_outcome_ofn_total }
				log = "[GetDateText]: New Scramble for Africa: none after an OFN total victory"
			}
			else = {
				SFA_start = yes
			}
		}
	}

	option = {
		name = SFA.1.a
	}
}
