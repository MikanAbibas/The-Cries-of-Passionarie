	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = ITA_muti_colonial.1.a
		add_timed_idea = {
			idea = ITA_colonial_disaster_idea_1
			days = 180
		}
		hidden_effect = {
			for_each_loop = {
				array = ITA_EM_Regions_Status
				if = {
					limit = {
						check_variable = { v = 0 } # Region is a Colony of the Empire
						check_variable = { ITA_EM_Regions_DiploValue^i > 0 } # Has more than 0 Control
					}
					set_temp_variable = { Region_i = i }
					set_temp_variable = { DiploValueChange_v = -10 }
					ITA_EmpireManagement_DiploValueChange = yes
				}
			}
		}
		custom_effect_tooltip = ITA_there_are_no_inferiors_tooltip
	}
}
