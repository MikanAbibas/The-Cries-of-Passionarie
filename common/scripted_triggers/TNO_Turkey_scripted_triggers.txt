TUR_DtS_controls_this_region = {
	#NOTE: Should be called in a state scope
	has_state_flag = TUR_DtS_is_a_region #Return true if the state scope inputted is a DtS region
	any_of_scopes = { #and if ANY state in this region...
		array = TUR_DtS_states_in_this_region
		#is owned by Turkey or by a puppet
		owner = {
			OR = {
				tag = TUR
				is_puppet_of = TUR
			}
		}
		NOT = { state = 1369 } #the one exception is Marakada- it's <2k people, and shouldn't comporimise all of Northern Syria on its own
		#there are actually quite a few outcomes where Turkey owns just marakada and not the rest of north syria
	}
}

TUR_HB_target_is_communist = {
	var:TUR_HB_op_targeted_country^TUR_HB_selected_operation = {
		NOT = { tag = TUR } #doesn't matter for domestic operations
		OR = {
			AND = { #Check if the target itself exists and meets this.
				exists = yes 
				OR = {
					has_government = communist
					has_government = socialist
				}
			}
			AND = { #Syrian National State can also count for Syria.
				tag = SYR
				SNS = {
					exists = yes 
					OR = {
						has_government = communist
						has_government = socialist
					}
				}
			}
			AND = { #IRI can also count for Persia.
				tag = PER
				IRI = {
					exists = yes 
					OR = {
						has_government = communist
						has_government = socialist
					}
				}
			}
		}
	}
}

			check_variable = { money_reserves_m > var:TUR_DtS_Selected_Region:TUR_DtS_dev1_action_money_cost }
			check_variable = { credit_rating > 2 } #cannot draw from debt at TERRIBLE credit
		}
		
	}
	
	
}
