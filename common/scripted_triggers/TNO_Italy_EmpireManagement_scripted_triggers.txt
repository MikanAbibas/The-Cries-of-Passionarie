# Used to check if an ENI action can be taken in one region
EmpireManagement_ENIActionIsAvailable = {
	if = {
		limit = { NOT = { has_variable = i } }
		set_temp_variable = { i = EM_SelectedEntry_i }
	}

	# Base triggers
	check_variable = { ITA_EM_HasOil^i > 0 } # Has to have oil
	check_variable = { ITA_EM_OilExploitation^i < 1 } # ENI presence must be lower than max - the variable is a fraction, clamped to 1, and displayed with |0%
	check_variable = { ITA_EM_Regions_Status^i < 3 } # Has to be in your sphere
	check_variable = { ITA_EM_OilExploitationCooldown^i = 0 } # Must not be on cooldown
}

# Money cost triggers are irrelevant so we only check for PP here
EmpireManagement_ENIActionCostTrigger = {
	set_temp_variable = { PPCost = 50 }
	if = {
		limit = {
			OR = {
				check_variable = { EM_SelectedEntry_i = 1  } # Sudan
				check_variable = { EM_SelectedEntry_i = 23 } # Oman
				check_variable = { EM_SelectedEntry_i = 26 } # Egypt
			}
		}
		multiply_temp_variable = { PPCost = 2 }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { EM_SelectedEntry_i = 0  } # Libya
				check_variable = { EM_SelectedEntry_i = 19 } # Iraq
				check_variable = { EM_SelectedEntry_i = 20 } # Saudi Arabia
				check_variable = { EM_SelectedEntry_i = 21 } # Gulf
				check_variable = { EM_SelectedEntry_i = 24 } # Algeria
			}
		}
		multiply_temp_variable = { PPCost = 3 }
	}
	has_political_power > var:PPCost
}
