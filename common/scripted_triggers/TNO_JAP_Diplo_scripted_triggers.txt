# Country scope. A sphere member Japan can squeeze: its sphere status idea is one of the subordinate ones.
# Co-equal members (Sphere_Coequal) and nations outside the sphere (no status idea) are not, so the
# Sphere Diplomacy GUI's interactions give them the same bonus Japan gets instead of a malus.
JAP_sphere_is_subordinate = {
	OR = {
		has_idea = Sphere_Imp_Protectorate
		has_idea = Sphere_Economic_Dependent
		has_idea = Sphere_Fully_Dependent
		has_idea = Sphere_Special_Econ_Zone
	}
}
