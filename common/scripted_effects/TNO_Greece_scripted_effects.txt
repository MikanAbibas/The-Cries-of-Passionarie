GRE_resistance_startup = {
	
	set_variable = {
		GRE_resistance_state_view = -1
	}
	set_country_flag = GRE_resistance_mechanic

	#Using four arrays to make this easier (scoped for 8 states).
	# Why not use one array for each state? Tracking 8 arrays would be unnecessarily complicated.
	# Why not reuse index access from Control of The State? Avoids overlap and confusion though CoTS and Resistance wouldn't happen at the same time anyways.

	# 1 - Government control.
	add_to_array = { GRE_Resistance_Government_Control = 0 }
	resize_array = { GRE_Resistance_Government_Control = 8 }

	# 2 - EDES control.
	add_to_array = { GRE_Resistance_EDES_Control = 0 }
	resize_array = { GRE_Resistance_EDES_Control = 8 }

	# 3 - EAM control.
	add_to_array = { GRE_Resistance_EAM_Control = 0 }
	resize_array = { GRE_Resistance_EAM_Control = 8 }

	# 4 - Controller of state.
	add_to_array = { GRE_Resistance_State_Controller = 0 }
	resize_array = { GRE_Resistance_State_Controller = 8 }

	# We set up everything - index goes from 0-7 due to having 8 states
	# First - government control.
	set_variable = { GRE_Resistance_Government_Control^0 = 35 } # State 0 is EDES HQ
	set_variable = { GRE_Resistance_Government_Control^1 = 85 }
	set_variable = { GRE_Resistance_Government_Control^2 = 85 }
	set_variable = { GRE_Resistance_Government_Control^3 = 40 } # State 3 is EAM HQ
	set_variable = { GRE_Resistance_Government_Control^4 = 70 }
	set_variable = { GRE_Resistance_Government_Control^5 = 80 }
	set_variable = { GRE_Resistance_Government_Control^6 = 75 }
	set_variable = { GRE_Resistance_Government_Control^7 = 80 }

	# Second - EDES control.
	set_variable = { GRE_Resistance_EDES_Control^0 = 60 } # State 0 is EDES HQ
	set_variable = { GRE_Resistance_EDES_Control^1 = 10 }
	set_variable = { GRE_Resistance_EDES_Control^2 = 5 }
	set_variable = { GRE_Resistance_EDES_Control^3 = 7 } # State 3 is EAM HQ
	set_variable = { GRE_Resistance_EDES_Control^4 = 5 }
	set_variable = { GRE_Resistance_EDES_Control^5 = 15 }
	set_variable = { GRE_Resistance_EDES_Control^6 = 20 }
	set_variable = { GRE_Resistance_EDES_Control^7 = 5 }

	# Third - EAM control.
	set_variable = { GRE_Resistance_EAM_Control^0 = 5 } # State 0 is EDES HQ
	set_variable = { GRE_Resistance_EAM_Control^1 = 5 }
	set_variable = { GRE_Resistance_EAM_Control^2 = 10 }
	set_variable = { GRE_Resistance_EAM_Control^3 = 53 } # State 3 is EAM HQ
	set_variable = { GRE_Resistance_EAM_Control^4 = 25 }
	set_variable = { GRE_Resistance_EAM_Control^5 = 5 }
	set_variable = { GRE_Resistance_EAM_Control^6 = 5 }
	set_variable = { GRE_Resistance_EAM_Control^7 = 15 }

	# Fourth - Controller of state: all set to 0 to clear everything
	# 1 is government, 2 is EDES, 3 is EAM
	set_variable = { GRE_Resistance_State_Controller^0 = 0 } # State 0 is EDES HQ
	set_variable = { GRE_Resistance_State_Controller^1 = 0 }
	set_variable = { GRE_Resistance_State_Controller^2 = 0 }
	set_variable = { GRE_Resistance_State_Controller^3 = 0 } # State 3 is EAM HQ
	set_variable = { GRE_Resistance_State_Controller^4 = 0 }
	set_variable = { GRE_Resistance_State_Controller^5 = 0 }
	set_variable = { GRE_Resistance_State_Controller^6 = 0 }
	set_variable = { GRE_Resistance_State_Controller^7 = 0 }

	# Clear state control status.
	set_variable = {
		EDES_states = 0
	}
	set_variable = {
		EAM_states = 0
	}
	set_variable = {
		govt_states = 0
	}

	for_each_loop = { #Gets index to calculate control of states
		array = GRE_Resistance_Government_Control
		add_to_variable = { # Break up ties.
			GRE_Resistance_Government_Control^i = 1
		}
		add_to_variable = { # Break up ties for a possible EDES-EAM tie - extra to EDES because they have more content lol
			GRE_Resistance_EDES_Control^i = 2
		}
		if = {
			limit = { # If government has more control than EAM and EDES combined, add 1 to Government controlled state.
				AND = {
					check_variable = {
						GRE_Resistance_Government_Control^i > GRE_Resistance_EDES_Control^i
					}
					check_variable = {
						GRE_Resistance_Government_Control^i > GRE_Resistance_EAM_Control^i
					}
				}
			}
			add_to_variable = {
				govt_states = 1
			}
			set_variable = { GRE_Resistance_State_Controller^i = 1 } # Mark it for the GUI as a government state
		}
		else_if = {
			limit = { # If EDES has more control than EAM and government combined, add 1 to EDES controlled state.
				AND = {
					check_variable = {
						GRE_Resistance_EDES_Control^i > GRE_Resistance_Government_Control^i
					}
					check_variable = {
						GRE_Resistance_EDES_Control^i > GRE_Resistance_EAM_Control^i
					}
				}
			}
			add_to_variable = {
				EDES_states = 1
			}
			set_variable = { GRE_Resistance_State_Controller^i = 2 } # Mark it for the GUI as an EDES state
		}
		else_if = {
			limit = { # If EAM has more control than EDES and government combined, add 1 to EAM controlled state.
				AND = {
					check_variable = {
						GRE_Resistance_EAM_Control^i > GRE_Resistance_Government_Control^i
					}
					check_variable = {
						GRE_Resistance_EAM_Control^i > GRE_Resistance_EDES_Control^i
					}
				}
			}
			add_to_variable = {
				EAM_states = 1
			}
			set_variable = { GRE_Resistance_State_Controller^i = 3 } # Mark it for the GUI as an EAM state
		}
		subtract_from_variable = { # Tiebreaker done, revert to pre-tiebreaker control.
			GRE_Resistance_Government_Control^i = 1
		}
		subtract_from_variable = { # Tiebreaker done, revert to pre-tiebreaker control.
			GRE_Resistance_EDES_Control^i = 2
		}
	}
	log = "[GetDateText]: [Root.GetName]: GRE_resistance_startup - Government: [?govt_states] - EDES: [?EDES_states] - EAM: [?EAM_states]."
	# Actions 
	set_variable = {
		GRE_resistance_actions_left = 0
	}
	activate_mission = GRE_resistance_weekly_tick
	add_to_variable = {
		resistance_gui_dirty = 1
	}
}

GRE_Party_Stability_mod = {
	
	# Uses two temp vars - GRE_Party_Calc (determines which calculation you use: Modify Kyrkos/Papandreou/Unity) and GRE_Party_Amount.
	
	# If you modify a faction, party unity goes down by 1/4th of the amount youre modifying.
	# If you modify a faction with less than 51% influence, add another 1/4th to unity reduction.
	# If you do not modify what calculation you are using, default to Kyrkos.

	if = {
		limit = { # We are modifying Kyrkos' influence.
			check_variable = {
				GRE_Party_Calc?1 = 1
			}
		}
		set_temp_variable = { 
			total = 100 
		}
		add_to_variable = {
			GRE_EDA_kyrkos_progress = GRE_Party_Amount
		}
		subtract_from_temp_variable = {
			total = GRE_EDA_kyrkos_progress
		}
		set_variable = { # Papandreou's influence will always be 100 - Kyrkos influence.
			GRE_EDA_papandreou_progress = total
		}
		set_temp_variable = { # Bypass temp var tooltip fuckeries
			GRE_Party_Unity_Mod = GRE_Party_Amount
		}
		if = { # Done in this way to make sure tooltip is correct.
			limit = {
				check_variable = { # If less than majority, subtract another 1/4th.
					GRE_EDA_kyrkos_progress < 51
				}
			}
			divide_temp_variable = {
				GRE_Party_Unity_Mod = 2
			}
			subtract_from_variable = {
				GRE_EDA_unity = GRE_Party_Unity_Mod
			}
		}
		else = { # Else, just subtract 1/4th
			divide_temp_variable = {
				GRE_Party_Unity_Mod = 4
			}
			subtract_from_variable = {
				GRE_EDA_unity = GRE_Party_Unity_Mod
			}
		}
	}

	else_if = {
		limit = { # We are modifying Papandreou's influence.
			check_variable = {
				GRE_Party_Calc?1 = 2
			}
		}
		set_temp_variable = { 
			total = 100 
		}
		subtract_from_variable = { # Since Kyrkos' influence decides how much Papandreou has, subtract from him.
			GRE_EDA_kyrkos_progress = GRE_Party_Amount
		}
		subtract_from_temp_variable = {
			total = GRE_EDA_kyrkos_progress
		}
		set_variable = { # Papandreou's influence will always be 100 - Kyrkos influence.
			GRE_EDA_papandreou_progress = total
		}
		set_temp_variable = { # Bypass temp var tooltip fuckeries
			GRE_Party_Unity_Mod = GRE_Party_Amount
		}
		if = { # Done in this way to make sure tooltip is correct.
			limit = {
				check_variable = { # If less than majority, subtract another 1/4th.
					GRE_EDA_papandreou_progress < 51
				}
			}
			divide_temp_variable = {
				GRE_Party_Unity_Mod = 2
			}
			subtract_from_variable = {
				GRE_EDA_unity = GRE_Party_Unity_Mod
			}
		}
		else = { # Else, just subtract 1/4th
			divide_temp_variable = {
				GRE_Party_Unity_Mod = 4
			}
			subtract_from_variable = {
				GRE_EDA_unity = GRE_Party_Unity_Mod
			}
		}
	}

	else_if = {
		limit = { # We are modifying party unity.
			check_variable = {
				GRE_Party_Calc?1 = 3
			}
		}
		add_to_variable = { # Simply add amount to party unity, nothing big.
			GRE_EDA_unity = GRE_Party_Amount
		}
	}

	add_to_variable = {
		party_gui_dirty = 1
	}

	custom_effect_tooltip = GRE_Party_Stability_mod_tt
}

GRE_CoTS_Modify_Control = {
	
	# Uses two temp variables - GRE_CoTS_Faction_Mod (index of faction you are modifying) and GRE_CoTS_Amount (amount of control you want to change).

	if = {
		limit = {
			check_variable = {
				GRE_CoTS_Faction_Mod > -1
			}
		}
		add_to_variable = {	#If you somehow forget to specify who you are modifying, defaults to party
			GRE_CoTS_Faction_Control^GRE_CoTS_Faction_Mod?0 = GRE_CoTS_Amount?1 #If there is no set value, add 1
		}
		custom_effect_tooltip = GRE_CoTS_Mod_tt
	}
	else = { # If you give a negative index, effect does nothing
		log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Modify_Control - handed negative index."
	}

	#Gets average
	GRE_CoTS_Get_Average = yes

	log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Modify_Control - values after modification: Party - [?GRE_CoTS_Faction_Control^Party]. Military - [?GRE_CoTS_Faction_Control^Military]. Government - [?GRE_CoTS_Faction_Control^Government]."
	log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Modify_Control - values after modification: People - [?GRE_CoTS_Faction_Control^People]. Business - [?GRE_CoTS_Faction_Control^Business]."
}

GRE_CoTS_Get_Average = {
	set_temp_variable = { # Clears everything
		GRE_CoTS_Average_temp = 0
	}
	for_each_loop = { #Clamps everything, gets average
		array = GRE_CoTS_Faction_Control
		clamp_variable = { 
			var = GRE_CoTS_Faction_Control^i
			min = 0 
			max = 100 
		}
		add_to_temp_variable = { # Add everything into average
			GRE_CoTS_Average_temp = GRE_CoTS_Faction_Control^i
		}
		log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Average - Average check - [?GRE_CoTS_Average_temp]"
	}
	set_variable = {
		GRE_CoTS_Average = GRE_CoTS_Average_temp
	}
	divide_variable = {
		GRE_CoTS_Average = 5
	}
	log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Average - Average calculated - [?GRE_CoTS_Average]"
}

GRE_CoTS_Get_Winner = {
	hidden_effect = {
		for_each_loop = { #Gets index to calculate control
			array = GRE_CoTS_Faction_Control
			set_temp_variable = {
				GRE_CoTS_Faction_Mod = i
			}
			if = {
				limit = { # If faction is under Karamanlis, add 1 point
					GRE_CoTS_Faction_Under_Karamanlis = yes
				}
				add_to_temp_variable = {
					GRE_CoTS_Karamanlis = 1
				}
				log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Winner - Scripted trigger is under Karamanlis."
				if = {
					limit = { # If control is lost, add another point
						GRE_CoTS_Faction_Under_Karamanlis_Lost = yes
					}
					add_to_temp_variable = {
						GRE_CoTS_Karamanlis = 1
					}
					log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Winner - Scripted trigger is under Karamanlis - lost."
				}
			}
			else_if = {
				limit = { # If faction is under government, add 1 point to government
					GRE_CoTS_Faction_Under_Government = yes
				}
				add_to_temp_variable = {
					GRE_CoTS_Government = 1
				}
				if = {
					limit = { # If control is total, add another point
						GRE_CoTS_Faction_Under_Government_Total = yes
					}
					add_to_temp_variable = {
						GRE_CoTS_Government = 1
					}
					log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Winner - Scripted trigger is under government - total."
				}
				log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Winner - Scripted trigger is under government."
			}
			else = { # If neutral nothing happens
				log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Winner - Scripted trigger is neutral."
			}
		}
		log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Winner - [?GRE_CoTS_Karamanlis]/[?GRE_CoTS_Government]."
		if = { # If a tie happens, ensure Karamanlis wins.
			limit = {
				check_variable = {
					GRE_CoTS_Government = GRE_CoTS_Karamanlis
				}
			}
			add_to_temp_variable = {
				GRE_CoTS_Karamanlis = 1
			}
		}
		if = {
			limit = { # Government wins
				check_variable = {
					GRE_CoTS_Government > GRE_CoTS_Karamanlis
				}
			}
			log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Winner - Government victory."
			GRE = {
				country_event = {
					id = gre_unveiling.1 days = 3
				}
			}
		}
		else_if = { # Karamanlis wins
			limit = {
				check_variable = {
					GRE_CoTS_Government < GRE_CoTS_Karamanlis
				}
			}
			log = "[GetDateText]: [Root.GetName]: GRE_CoTS_Get_Winner - Karamanlis victory."
			GRE = {
				country_event = {
					id = gre_dominos.1 days = 3
				}
			}
		}
	}

	custom_effect_tooltip = GRE_CoTS_Victor_Calc_tt
}
