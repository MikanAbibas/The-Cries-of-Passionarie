# Phase E of the Greece and Turkey import (GREECE_INTEGRATION_PLAN.md): who governs Greece after its
# elections. On the Mediterranean branch only debug decisions loaded these trees. GRE scope throughout.

# After GRE_election_select_national_victors. The higher of EDA and ERE governs (EK, KKE and MK have no
# trees); a tie goes to the EDA. The second election, at the end of the EDA's first term, also
# decides Pyromaglou's successor.
GRE_election_form_government = {
	if = {
		limit = { has_country_flag = GRE_colonels_junta } # the colonels hold on (Phase G)
	}
	else_if = {
		limit = { check_variable = { ERE_popularity_national > EDA_popularity_national } }
		GRE_form_ERE_government = yes
	}
	else_if = {
		limit = { has_country_flag = GRE_second_election }
		GRE_EDA_succession = yes
	}
	else = {
		GRE_form_EDA_government = yes
	}
	clr_country_flag = GRE_second_election
}

GRE_form_EDA_government = {
	set_politics = { ruling_party = social_democrat }
	if = {
		limit = { NOT = { has_country_leader = { name = "Komninos Pyromaglou" ruling_only = yes } } }
		GRE_med_promote_Komninos_Pyromaglou = yes
	}
	set_country_flag = GRE_EDA_government
	load_focus_tree = GRE_EDA_First_Term_Tree
	mark_focus_tree_layout_dirty = yes
}

# Kanellopoulos in 1964, Karamanlis (med's ERE candidate in the second campaign) after it. The ERE tree
# runs on Control of the State, which only a debug decision started on the branch.
GRE_form_ERE_government = {
	set_politics = { ruling_party = social_conservative }
	if = {
		limit = { has_country_flag = GRE_second_election }
		if = {
			limit = { NOT = { has_country_leader = { name = "Konstantinos Karamanlis" ruling_only = yes } } }
			GRE_med_promote_Konstantinos_Karamanlis = yes
		}
	}
	else = {
		if = {
			limit = { NOT = { has_country_leader = { name = "Panagiotis Kanellopoulos" ruling_only = yes } } }
			GRE_med_promote_Panagiotis_Kanellopoulos = yes
		}
	}
	if = {
		limit = { NOT = { has_country_flag = GRE_ERE_control_gui_enabled } }
		GRE_CoTS_init = yes
	}
	clr_country_flag = GRE_EDA_government
	set_country_flag = GRE_ERE_government
	load_focus_tree = GRE_ERE_First_Term_Tree
	mark_focus_tree_layout_dirty = yes
}

# The EDA keeps power: whoever leads the influence bar (GRE_Party_Stability_init, the Papandreou/Kyrkos
# GUI) succeeds Pyromaglou. A tie, or a bar never started, goes to Papandreou.
GRE_EDA_succession = {
	if = {
		limit = { check_variable = { GRE_EDA_kyrkos_progress > GRE_EDA_papandreou_progress } }
		set_politics = { ruling_party = socialist }
		if = {
			limit = { NOT = { has_country_leader = { name = "Leonidas Kyrkos" ruling_only = yes } } }
			create_country_leader = {
				name = "Leonidas Kyrkos"
				desc = "POLITICS_LEONIDAS_KYRKOS_DESC"
				picture = "gfx/leaders/GRE/Leonidas_Kyrkos.dds"
				expire = "2011.8.28.1"
				ideology = socialist_revolutionary_frontism_subtype
				traits = { }
			}
		}
		set_country_flag = GRE_Kyrkos_government
		load_focus_tree = GRE_EDA_Kyrkos_Tree
		GRE_kyrkos_referendum_setup_effect = yes
	}
	else = {
		set_politics = { ruling_party = social_democrat }
		if = {
			limit = { NOT = { has_country_leader = { name = "Andreas Papandreou" ruling_only = yes } } }
			GRE_med_promote_Andreas_Papandreou = yes
		}
		set_country_flag = GRE_Papandreou_government
		load_focus_tree = GRE_EDA_Papandreou_Tree
	}
	mark_focus_tree_layout_dirty = yes
	GRE_EDA_oil_crisis_response = yes # the crisis may already be on (Phase F)
}

# The Oil Crisis reaches the EDA's second government (Phase F). Med's responses were reachable only by
# debug decision: Papandreou's opens his System Shock branch (and through it the negotiations fork),
# Kyrkos's opens the branch that holds his referendum. Once, while either tree is loaded. Med's debug
# decisions also added GRE_oil_crisis_specific, whose values nothing ever sets; it is left out.
GRE_EDA_oil_crisis_response = {
	if = {
		limit = {
			has_global_flag = oil_crisis_full_swing
			NOT = { has_country_flag = GRE_EDA_oil_crisis }
			NOT = { has_country_flag = GRE_EDA_oil_crisis_response_sent }
		}
		if = {
			limit = { has_focus_tree = GRE_EDA_Papandreou_Tree }
			set_country_flag = GRE_EDA_oil_crisis_response_sent
			country_event = { id = greEDA.247 days = 1 }
		}
		else_if = {
			limit = { has_focus_tree = GRE_EDA_Kyrkos_Tree }
			set_country_flag = GRE_EDA_oil_crisis_response_sent
			country_event = { id = greEDA.382 days = 1 }
		}
	}
}
