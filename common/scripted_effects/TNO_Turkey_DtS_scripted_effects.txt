TUR_DtS_Change_Consciousness = {
	#Fired in a COUNTRY scope (rare for dts :trellwater:)

	#EXAMPLE ON HOW TO USE:
	#set_temp_variable = { tur_dts_con_change_amount = 10 }
	#set_temp_variable = { tur_dts_targeted_state = 200 } #all kurdish states
	#TUR_DtS_Change_Consciousness = yes

	#temp var: tur_dts_con_change_amount --- the amount to change consciousness by
	#temp var: tur_dts_targeted_state --- the region you are changing values in

	#Regions: 1397 Batumi | 230 Armenia | 677 Afrin | 352 Turkish Kurdistan | 1242 Idlip 
	# 781 Mosul | 1369 Rojava | 680 North Syria | 554 Damascus | 1266 Thrace
	#Not at start: 164 Rhodes/Naxos | 1240 Lebanon | 1328 Cyprus
	#100- target EVERY region
	#200- target every kurdish region
	#300- target every arabic region
	#400- target every greco-turkish region

	#It would be easily possible to combine TUR_DtS_Change_Consciousness and TUR_DtS_Change_Unrest
	#but it would make the code a lot less intuitive IMO -aero

	#Step 0: clear the array so you could do both consciousness and unrest at once
	clear_temp_array = tur_dts_targeted_states

	#Step 1: Add all targeted states into one array.
	if = { #All states
		limit = { check_variable = { tur_dts_targeted_state = 100 } }
		for_each_loop = {
			array = TUR_DtS_Regions
			value = v
			add_to_temp_array = { tur_dts_targeted_states = v }
		}
	}
	else_if = { #Kurdish states
		limit = { check_variable = { tur_dts_targeted_state = 200 } } 
		add_to_temp_array = { tur_dts_targeted_states = 352 }
		add_to_temp_array = { tur_dts_targeted_states = 781 }
		add_to_temp_array = { tur_dts_targeted_states = 1369 }
	}
	else_if = { #Arabic States
	limit = { check_variable = { tur_dts_targeted_state = 300 } } 
		add_to_temp_array = { tur_dts_targeted_states = 680 }
		add_to_temp_array = { tur_dts_targeted_states = 554 }
		add_to_temp_array = { tur_dts_targeted_states = 1240 }
	}
	else_if = { #Greco-Turkish States
	limit = { check_variable = { tur_dts_targeted_state = 400 } } 
		add_to_temp_array = { tur_dts_targeted_states = 164 }
		add_to_temp_array = { tur_dts_targeted_states = 1266 }
		add_to_temp_array = { tur_dts_targeted_states = 1328 }
	}
	else = { #Single state
		add_to_temp_array = { tur_dts_targeted_states = tur_dts_targeted_state }
	}

	#Step 2: Change the con in all those states.
	for_each_scope_loop = {
		array = tur_dts_targeted_states
		if = {
			limit = { TUR_DtS_controls_this_region = yes }
			add_to_variable = { TUR_DtS_Consciousness = tur_dts_con_change_amount }
			add_to_variable = { TUR_DtS_dCon_From_Recent_Events = tur_dts_con_change_amount }
		}
	}

	#Step 3: Make a custom loc tooltip.
	if = {
		limit = { check_variable = { tur_dts_con_change_amount > 0 } }
		custom_effect_tooltip = TUR_DTS_Consciousness_Increase_Tooltip
	}
	if = { 
		limit = { check_variable = { tur_dts_con_change_amount < 1 } }
		multiply_temp_variable = { tur_dts_con_change_amount = -1 }
		custom_effect_tooltip = TUR_DTS_Consciousness_Decrease_Tooltip
	}	

	TUR_DtS_Clamp_Variables = yes
}

TUR_DtS_Monthly_Calculation = {
	for_each_scope_loop = { 
		array = TUR_DtS_Regions
		
		if = { #Check if revolt was avoided BEFORE monthly change
			limit = { 
				#TUR_DtS_controls_this_region = yes
				check_variable = { TUR_DtS_current_incident = 8 }
				NOT = { check_variable = { TUR_DtS_Unrest > 90 } }
			}
			set_variable = { TUR_DtS_current_incident = 0 }
		}
		
		if = { #Second step: Have con/unrest increment each month
			limit = { TUR_DtS_controls_this_region = yes }

			#dCon: delta (change) in consciousness
			#dUnr: delta of unrest

			#CONSCIOUSNESS CHANGE
			#Base & Events
			add_to_variable = { TUR_DtS_Consciousness = TUR.TUR_DTS_DCON_BASE }
			set_variable = { TUR_DtS_dCon_From_Recent_Events = 0} #Reset the tooltip for impacts from player actions this month
			set_variable = { TUR_DtS_dCon_From_Recent_CU_Events = 0} #Reset the tooltip for impacts from protests or strikes

			if = {
				limit = {
					NOT = { has_state_flag = TUR_DtS_Development_Cancelled }
					NOT = { has_state_flag = TUR_DtS_Development_Done }
				}
				#lack of investment
				add_to_variable = { TUR_DtS_months_without_investment = 1 }
				set_variable = { TUR_DtS_dCon_Underinvestment = TUR_DtS_months_without_investment }
				if = { #Underinvestment is not a concern for the AI
					limit = { TUR = { is_ai = yes } }
					set_variable = { TUR_DtS_dCon_Underinvestment = 0 }
				}
				clamp_variable = { #The "Buffer" period won't decrease your unrest
					var = TUR_DtS_dCon_Underinvestment
					min = 0	max = 999999
				}
				multiply_variable = { TUR_DtS_dCon_Underinvestment = TUR.TUR_DTS_CON_PER_MONTH_WITHOUT_INVESTMENT}
				round_variable = TUR_DtS_dCon_Underinvestment
				add_to_variable = { TUR_DtS_Consciousness = TUR_DtS_dCon_Underinvestment }

				#Overinvestment
				set_variable = { TUR_DtS_dCon_Overinvestment = 0 }
				set_variable = { TUR_DtS_dCon_Overinvestment = TUR_DtS_dCon_Overinvestment_Hidden}
				add_to_variable = { TUR_DtS_Consciousness = TUR_DtS_dCon_Overinvestment }
				set_variable = { TUR_DtS_dCon_Overinvestment_Hidden = 0 }
				
			}
			else_if = { #Change from cancelled project
				limit = {
					has_state_flag = TUR_DtS_Development_Cancelled
				}
				add_to_variable = { TUR_DtS_Consciousness = TUR.TUR_DTS_DCON_FROM_CANCEL }
			}
			
			#UNREST CHANGE
			#Base & Events
			set_variable = { TUR_DtS_dUnr_Base = TUR_DtS_Consciousness }
			#log = "DtS: Unrest Tick: [GetID]: Starting Consciousness Is [?TUR_DtS_Consciousness]"
			divide_variable = { TUR_DtS_dUnr_Base = 100 }
			multiply_variable = { TUR_DtS_dUnr_Base = TUR.TUR_DTS_DUNR_MAXIMUM_BASE }
			#log = "DtS: Unrest Tick: [GetID]: Unrest Before Round Is [?TUR_DtS_dUnr_Base]"
			round_variable = TUR_DtS_dUnr_Base
			#log = "DtS: Unrest Tick: [GetID]: Unrest After Round Is [?TUR_DtS_dUnr_Base]"
			add_to_variable = { TUR_DtS_Unrest = TUR_DtS_dUnr_Base }

			set_variable = { TUR_DtS_dUnr_From_Recent_Events = 0} #Reset the tooltip for impacts from player actions this month
			set_variable = { TUR_DtS_dUnr_From_Recent_CU_Events = 0} #Reset the tooltip for impacts from protests or strikes
			
			if = {
				limit = {
					NOT = { has_state_flag = TUR_DtS_Development_Cancelled }
					NOT = { has_state_flag = TUR_DtS_Development_Done }
				}
				#Lack of investment only affects consciousness

				#Overinvestment
				set_variable = { TUR_DtS_dUnr_Overinvestment = 0 }
				set_variable = { TUR_DtS_dUnr_Overinvestment = TUR_DtS_dUnr_Overinvestment_Hidden}
				add_to_variable = { TUR_DtS_Unrest = TUR_DtS_dUnr_Overinvestment }
				set_variable = { TUR_DtS_dUnr_Overinvestment_Hidden = 0 }
				
			}
			else_if = { #Change from cancelled project
				limit = {
					has_state_flag = TUR_DtS_Development_Cancelled
				}
				add_to_variable = { TUR_DtS_Unrest = TUR.TUR_DTS_DUNR_FROM_CANCEL }
			}

	
		}
		else = { #Reset variables for a non controlled territory
			set_variable = { TUR_DtS_dCon_From_Recent_Events = 0 } 
			set_variable = { TUR_DtS_dCon_From_Recent_CU_Events = 0 } 
			set_variable = { TUR_DtS_dCon_Underinvestment = 0 }
			set_variable = { TUR_DtS_dCon_Overinvestment = 0 }
			set_variable = { TUR_DtS_dCon_Overinvestment_Hidden = 0 }
			set_variable = { TUR_DtS_dUnr_From_Recent_Events = 0 } 
			set_variable = { TUR_DtS_dUnr_From_Recent_CU_Events = 0 } 
			set_variable = { TUR_DtS_dUnr_Overinvestment = 0 }
			set_variable = { TUR_DtS_dUnr_Overinvestment_Hidden = 0 }
		}
		TUR_DtS_Clamp_Variables = yes

		if = { #Third Step: Check if revolt was avoided AFTER monthly change
			#Revolt can be avoided either before OR after monthly change- to be kind to the player <3
			limit = { 
				TUR_DtS_controls_this_region = yes
				check_variable = { TUR_DtS_current_incident = 8 }
				check_variable = { TUR_DtS_Unrest > 90 }
			}
			if = { #Note: Damascus sometimes causes NASIL to rise up instead.
				limit = { 
					state = 554
					check_variable = { TUR_DtS_Vanguard > -1 }
				}
				680 = { TUR_DtS_Begin_An_Uprising = yes }
				subtract_from_variable = { TUR_DtS_Vanguard = 20 }
			}
			else_if = { #2nd edge case: Thrace becomes an OOB (not a tag) if the ITW is still ongoing.
				limit = {
					state = 1266
					TUR = { has_war_with = GRE }
				}
				GRE = {
					load_oob = "GRE_itw_thrace"
					set_division_template_lock = {
						division_template = "Epitropí Apelefthérosis Thrákis"
						is_locked = yes
					}
				}
			}
			else = { #Fires the fuction to begin an uprising FROM THE SCOPE OF THE REGION THAT WILL REVOLT
				TUR_DtS_Begin_An_Uprising = yes
			}
		}
		#Step 3.5: after revolt is checked, undo the current incident
		set_variable = { TUR_DtS_current_incident = 0 }
		
		
		if = { #Fourth step: Fire new con/unrest events
			limit = { 
				TUR_DtS_controls_this_region = yes 
				#NOT = { has_state_flag = TUR_DtS_incident_last_month }
				OR = {
					check_variable = { TUR_DtS_Unrest > 40 }
					check_variable = { TUR_DtS_Consciousness > 40 }
				}
			}
			#To get the incident chance, unrest and consciousness are averaged into a % chance.
			set_temp_variable = { incident_chance = TUR_DtS_Unrest }
			add_to_temp_variable = { incident_chance = TUR_DtS_Consciousness }
			divide_temp_variable = { incident_chance = 200 }
			#Then a random number [from 0 to 1 ] is picked
			set_temp_variable = { random_incident = random }
			if = {
				limit = { 
					OR = {
						check_variable = { random_incident < incident_chance } #more likely to be below the higher that incident_chance is
						AND = {
							check_variable = { TUR_DtS_Unrest > 99 } #no randomness if it's a revolt
							NOT = { has_state_flag = DtS_No_Revolt_Martial_Law } #but only if a revolt can actually happen
							NOT = { has_state_flag = DtS_No_Revolt_Local_Collaborators }
						}
					}
				} 
				#set_state_flag = TUR_DtS_incident_last_month #Removed- If put in, events can happen every month
				TUR_DtS_Cause_Incident = yes 
			}
		}
		#else = { clr_state_flag = TUR_DtS_incident_last_month }
		
		#Fifth step: Do some monthly econ changes! 
		TUR_DtS_Resolve_Regional_GDP_Growth = yes
		TUR_DtS_Resolve_DidTooMuchAtOnce_Cooldown = yes
		
		
	}
	#Sixth step: Do monthly econ recalculations
	TUR = { TUR_DtS_Calculate_Regional_GDPs = yes }
	
	TUR_DtS_Clamp_Variables = yes

	#Seventh step: have the AI recalculate its weights
	TUR_DtS_AI_calculate_region_priorities = yes
	#Idea: could be moved to 3rd and 26th of month to lessen the lag on the 1st of the month?
}

TUR_DtS_Calculate_Regional_Development_Progress = {
	#Fired in a state scope.
	
	clr_state_flag = TUR_DtS_development_LEVEL_below_0
	clr_state_flag = TUR_DtS_development_LEVEL_above_100
	set_variable = { TUR_DtS_development_LEVEL = 0 }
	
	if = { #First, check if the region is failing. If it is then you can skip annoying calculations
		limit = {
			TUR_DtS_controls_this_region = yes
			check_variable = { TUR_DtS_regional_GDP < TUR_DtS_regional_GDP_goals^TUR_DtS_development_STAGE }
		}
		set_state_flag = TUR_DtS_development_LEVEL_below_0
		#log = "TUR DtS Regional_Development_Progress: Current GDP is below regional GDP [?TUR_DtS_regional_GDP] vs [?TUR_DtS_regional_GDP_goals^TUR_DtS_development_STAGE]"
	}
	else_if = { #Otherwise, calculate the progress done so far on this stage.
		limit = { 
			TUR_DtS_controls_this_region = yes
		}
		
		
		set_temp_variable = { TUR_DtS_NEXT_development_STAGE = TUR_DtS_development_STAGE }
		add_to_temp_variable = { TUR_DtS_NEXT_development_STAGE = 1 }
		
		#TUR_DtS_total_gdp_gap_this_stage: The total amount that GDP has to increase this stage. Next Stage's limit minus Current Stage's limit
		set_temp_variable = { TUR_DtS_total_gdp_gap_this_stage = TUR_DtS_regional_GDP_goals^TUR_DtS_NEXT_development_STAGE }
		subtract_from_temp_variable = { TUR_DtS_total_gdp_gap_this_stage = TUR_DtS_regional_GDP_goals^TUR_DtS_development_STAGE }
		#log = "TUR DtS Regional_Development_Progress: GDP gap this stage is [?TUR_DtS_total_gdp_gap_this_stage]"
		
		#TUR_DtS_progress_on_the_current_stage: How far you've progressed on this stage alone.
		set_temp_variable = { TUR_DtS_progress_on_the_current_stage = TUR_DtS_regional_GDP }
		subtract_from_temp_variable = { TUR_DtS_progress_on_the_current_stage = TUR_DtS_regional_GDP_goals^TUR_DtS_development_STAGE }
		#log = "TUR DtS Regional_Development_Progress: GDP gap this stage is [?TUR_DtS_progress_on_the_current_stage]"
		
		#Current Level on stage is determined by ( progress so far / total progress needed this stage). x100, since paradox is dumb with percents.
		set_variable = { TUR_DtS_development_LEVEL = TUR_DtS_progress_on_the_current_stage }
		divide_variable = { TUR_DtS_development_LEVEL = TUR_DtS_total_gdp_gap_this_stage }
		multiply_variable = { TUR_DtS_development_LEVEL = 100 }
		
		if = { 
			limit = { check_variable = { TUR_DtS_development_LEVEL > 99 } }
			set_state_flag = TUR_DtS_development_LEVEL_above_100
		}
	}
}

TUR_DtS_Resolve_Regional_GDP_Growth = {
	#Fired in a state scope
	
	#First, boost GDP.
	for_each_scope_loop = {
		array = TUR_DtS_states_in_this_region

		set_temp_variable = { state_value_multiplier_temp = PREV.TUR_DtS_regional_GDP_growth_monthly }
		if = { #Cut the effective change in half, if you're between stages
			limit = { has_state_flag = TUR_DtS_development_LEVEL_above_100 }
			add_to_temp_variable = { state_value_multiplier_temp = 1 }
			multiply_temp_variable = { state_value_multiplier_temp = 0.5 }
		}
		econ_state_value_change_multiply_specified_state = yes #changes state gdp by state_value_multiplier_temp
		
	}
	
	#Second, note down which GDP boosts have run out- and move time forward for the rest.
	for_each_loop = {
		array = TUR_DtS_regional_modifiers_MONTHS #or TUR_DtS_regional_modifiers_VALUE, which has the SAME LENGTH
		value = v
		index = i
		
		if = { #If this is the last month, slate the modifier for removal
			limit = { check_variable = { v = 1 } }
			
			#First, slate it for removal.
			add_to_temp_array = { 
				array = DtS_gdp_changes_to_remove
				value = i
				index = 0 #Last in, First out- that way each element in TUR_DtS_regional_modifiers_MONTHS is removed in reverse order, preventing off-by-one errors.
			}
			
			subtract_from_variable = { TUR_DtS_regional_GDP_growth_monthly = TUR_DtS_regional_modifiers_VALUE^i }
		}
		else = { add_to_variable = { TUR_DtS_regional_modifiers_MONTHS^i = -1 } } #If it isn't the last month, decrement months by 1.
	} 
	
	#Third, remove the GDP boosts slated for removal.
	for_each_loop = {
		array = DtS_gdp_changes_to_remove
		value = index_of_expired_bonus
		
		remove_from_array = {
			array = TUR_DtS_regional_modifiers_MONTHS
			index = index_of_expired_bonus
		}
		remove_from_array = {
			array = TUR_DtS_regional_modifiers_VALUE
			index = index_of_expired_bonus
		}
	}
	
	clear_array = DtS_gdp_changes_to_remove
}

TUR_DtS_Recalculate_Stage_Impact_On_GDP_Growth = {
	#Fired in a state scope

	#TUR_DtS_Region_GDP_Growth_Impact | How much national GDP growth should change if you own this region.
	#TUR_DtS_Region_GDP_Growth_Impact_Actual | the ingame impact of this state on national growth

	#TUR_DtS_Prev_Stages_Extract_Or_Manu | adds an item for each new stage, and the extract/manu value

	set_variable = { TUR_DtS_Region_GDP_Growth_Impact = 0 }

	for_each_loop = {
		array = TUR_DtS_Prev_Stages_and_EI_ratios
		value = v
		index = i

		set_temp_variable = { dts_impact_from_this_stage = TUR.DTS_STAGE_INCREASE_IMPACT_BASE^i }
		#log = "TUR_DtS_SII: [This.GetID]: Index [?i]: BASE VALUE: [?dts_impact_from_this_stage]"

		set_temp_variable = { dts_sii_pop_impact = TUR_DtS_regional_population_M }
		add_to_temp_variable = { dts_sii_pop_impact = -1 }
		multiply_temp_variable = { dts_sii_pop_impact = TUR.DTS_SII_IMPACT_FROM_STATE_SIZE }
		add_to_temp_variable = { dts_sii_pop_impact = 1 }
		multiply_temp_variable = { dts_impact_from_this_stage = dts_sii_pop_impact }
		#log = "TUR_DtS_SII: [This.GetID]: Index [?i]: POP IMPACT: [?dts_sii_pop_impact]"

		if = { #If the stage was extraction based, then add in a modifier dependent on how many trade blocs you access
			limit = { check_variable = { v < 0.4 } }
			multiply_temp_variable = { dts_impact_from_this_stage = TUR.DTS_EXTRACTION_TRADE_IMPACTS^TUR_DtS_Num_Trade_Partners }
		}
		else_if = { #If stage was undecided, then debuff it by a constant.
			limit = { check_variable = { v < 0.61 } }
			multiply_temp_variable = { dts_impact_from_this_stage = TUR.DTS_SII_MIXED_POLICY_IMPACT }
		}
		#If stage is manufacturing based, just use the base increase. 

		add_to_variable = { TUR_DtS_Region_GDP_Growth_Impact = dts_impact_from_this_stage }
		#log = "TUR_DtS_SII: [This.GetID]: Index [?i]: AFTER EVM: [?dts_impact_from_this_stage]"
	}
	#log = "TUR_DtS_SII: [This.GetID]: FINAL IMPACT: [?TUR_DtS_Region_GDP_Growth_Impact]"

	if = {
		limit = { NOT = { check_variable = { TUR_DtS_Region_GDP_Growth_Impact = TUR_DtS_Region_GDP_Growth_Impact_Actual} } }

TUR_DtS_Apply_Stage_Impact_On_GDP_Growth = { 
	#Fired in a state scope

	#This function will apply a GDP growth impact if you control the region & didn't before,
	#And remove any GDP growth impact if you don't control it.

	#First check: Take away growth if you don't own the state
	if = {
		limit = {
			NOT = { TUR_DtS_controls_this_region = yes }
			NOT = { check_variable = { TUR_DtS_Region_GDP_Growth_Impact_Actual = 0 } }
		}
		#Take away the GDP growth from this state
		# TUR = {
		# 	set_temp_variable = { gdp_growth_temp = PREV.TUR_DtS_Region_GDP_Growth_Impact_Actual }
		# 	multiply_temp_variable = { gdp_growth_temp = -0.01 }
		# 	econ_gdp_growth_change = yes
		# }
		subtract_from_variable = { TUR.gdp_growth = TUR_DtS_Region_GDP_Growth_Impact_Actual } #No need for ttips
		TUR = { calculate_display_variables = yes }
		#Then set it so the state isn't impacting GDP growth
		set_variable = { TUR_DtS_Region_GDP_Growth_Impact_Actual = 0 }
	}
	#Second check: Apply GDP growth if it isn't applied
	if = {
		limit = {
			TUR_DtS_controls_this_region = yes

		# 	econ_gdp_growth_change = yes
		# }
		add_to_variable = { TUR.gdp_growth = TUR_DtS_Region_GDP_Growth_Impact } #No need for ttips
		TUR = { calculate_display_variables = yes }
		#Then set it so the state has impacted GDP growth
		set_variable = { TUR_DtS_Region_GDP_Growth_Impact_Actual = TUR_DtS_Region_GDP_Growth_Impact }
	}

TUR_DtS_AI_calculate_region_priorities = {
	#Fired from the TUR scope.
	#This function assigns a "danger level" to all regions.

	set_variable = { TUR_DtS_AI_greatest_concern = 0 }
	set_variable = { TUR_DtS_AI_priority_region = 781 }
	for_each_loop = {
		array = TUR_DtS_Regions
		value = region

		
		#First, calculate the "Total Weight" of this region.
		set_variable = { region:TUR_DtS_AI_concern_over_this_region = 1 } 

		add_to_variable = { region:TUR_DtS_AI_concern_over_this_region = region:TUR_DtS_Unrest }
		multiply_variable = { region:TUR_DtS_AI_concern_over_this_region = 1.7 } #Unr > Con
		add_to_variable = { region:TUR_DtS_AI_concern_over_this_region = region:TUR_DtS_Consciousness }

		divide_variable = { region:TUR_DtS_AI_concern_over_this_region = 8 } #Go from 0-271 to ~ 0-24

		add_to_variable = { region:TUR_DtS_AI_concern_over_this_region = region:TUR_DtS_regional_population_M }

		if = { #The AI stops caring so much if it can't revolt
			limit = { 
				var:region = { has_state_flag = DtS_No_Revolt_Martial_Law } 
			}
			multiply_variable = { region:TUR_DtS_AI_concern_over_this_region = 0.6 }
		}
		else_if = {
			limit = {
				var:region = { has_state_flag = DtS_No_Revolt_Local_Collaborators }
			}
			multiply_variable = { region:TUR_DtS_AI_concern_over_this_region = 0.4 }
		}
		else_if = { #The AI cares way more about regions about to revolt.
			limit = { check_variable = { region:TUR_DtS_current_incident = 8 } }
			if = { #If you're "too far gone", the AI abandons it and just hopes you'll revolt so it can crush it.
				limit = {
					check_variable = { region:TUR_DtS_Consciousness > 95 }
					#var:region = { has_state_flag = DtS_political_integration_cooldown }
					#var:region = { has_state_flag = DtS_military_withdrawl_cooldown }
					NOT = { check_variable = { region = 352 } }
				}
				multiply_variable = { region:TUR_DtS_AI_concern_over_this_region = 0.7 }
			}
			else = { #If you're not too far gone, then you're priotity number one.
				multiply_variable = { region:TUR_DtS_AI_concern_over_this_region = 1.4 }
			}
		}

		if = { #If turkey does not start with control of this region, it will never be prioritized.
			limit = { var:region = { TUR_DtS_controls_this_region = no } }
			multiply_variable = { region:TUR_DtS_AI_concern_over_this_region = 0.001}
		}
		else_if = { #the turkish AI is addicted to keeping Kurdistan under control
			limit = { check_variable = { region = 352 } }
			multiply_variable = { region:TUR_DtS_AI_concern_over_this_region = 1.15 }
		}
		else_if = { #but cares less about states with heavy collaboration
			limit = { 
				check_variable = { region = 554 } 
				SNS = { is_puppet_of = TUR }
			}
			multiply_variable = { region:TUR_DtS_AI_concern_over_this_region = 0.65 }
		}

		#Finally, after that long function, check if this region is higher priority than any previous region.
		#If it is, make it the new top priority reigon.
		if = {
			limit = { 
				check_variable = { region:TUR_DtS_AI_concern_over_this_region > TUR_DtS_AI_greatest_concern } 
			}
			set_variable = { TUR_DtS_AI_greatest_concern = region:TUR_DtS_AI_concern_over_this_region }
			set_variable = { TUR_DtS_AI_priority_region = region }
		}
	}

	#After all that, the AI changes its selected region to the highest priority one.
	if = { 
		limit = { is_ai = yes }
		set_temp_variable = { DtS_temp_selected_region = TUR_DtS_AI_priority_region }
		TUR_DtS_Change_Selected_Region = yes
	}
}
