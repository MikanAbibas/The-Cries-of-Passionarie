ITA_nuclear_calc = {
	clamp_variable = {
		var = ITA_nuclear_development_monthly
		min = 0
		max = 100
	}
	clamp_variable = { 
		var = ITA_nuclear_development_completion
		min = 0
		max = 100
	}
	clamp_variable = { 
		var = ITA_nuclear_maintenance_cost
		min = 0
	}
	clamp_variable = { 
		var = ITA_nuclear_science_cost
		min = 0
	}
	clamp_variable = { 
		var = ITA_nuclear_material_cost
		min = 0
	}
	clamp_variable = { 
		var = ITA_nuclear_material_quality
		min = 0
		max = 100
	}
	clamp_variable = { 
		var = ITA_scientist_happiness
		min = 0
		max = 100
	}
	ITA_nuclear_multipliers = yes
	set_variable = { ITA_nuclear_development_cost = 0}
	add_to_variable = { ITA_nuclear_development_cost = ITA_nuclear_maintenance_cost }
	add_to_variable = { ITA_nuclear_development_cost = ITA_nuclear_science_cost }
	add_to_variable = { ITA_nuclear_development_cost = ITA_nuclear_material_cost }
	
	##pie chart
	
	set_variable = { ITA_nuclear_maintenance_cost_percent = ITA_nuclear_maintenance_cost }
	set_variable = { ITA_nuclear_science_cost_percent = ITA_nuclear_science_cost }
	set_variable = { ITA_nuclear_material_cost_percent = ITA_nuclear_material_cost }
	
	divide_variable = { ITA_nuclear_maintenance_cost_percent = ITA_nuclear_development_cost }
	divide_variable = { ITA_nuclear_science_cost_percent = ITA_nuclear_development_cost }
	divide_variable = { ITA_nuclear_material_cost_percent = ITA_nuclear_development_cost }
	
	multiply_variable = { ITA_nuclear_maintenance_cost_percent = 100 }
	multiply_variable = { ITA_nuclear_science_cost_percent = 100 }
	multiply_variable = { ITA_nuclear_material_cost_percent = 100 }
	
	add_to_variable = { ITA_nuclear_material_cost_percent = ITA_nuclear_science_cost_percent }
	add_to_variable = { ITA_nuclear_maintenance_cost_percent = ITA_nuclear_material_cost_percent }
	
	# tooltip purposes
	set_variable = { ITA_nuclear_maintenance_tt = 100 }
	subtract_from_variable = { ITA_nuclear_maintenance_tt = ITA_nuclear_material_cost_percent }
	
	set_variable = { ITA_nuclear_material_tt = ITA_nuclear_material_cost_percent }
	subtract_from_variable = { ITA_nuclear_material_tt = ITA_nuclear_science_cost_percent }
	
	##misc cost stuff 
	
	set_variable = { ITA_nuke_misc_costs_var = ITA_nuclear_development_cost }
	divide_variable = { ITA_nuke_misc_costs_var = 1000 }
	
	if = {
		# Was "= 100", and ITA_nuclear_development_completion accumulates a float
		# every month, so it never landed on exactly a hundred and the event that
		# makes Italy a nuclear power never fired.
		limit = {
			check_variable = {
				var = ITA_nuclear_development_completion
				value = 100
				compare = greater_than_or_equals
			}
		}
		country_event = {
			id = NuclearItaly_med.2
		}
		set_variable = { ITA_nuclear_development_completion = 0 }
	}
	add_to_variable = { ITA_nuke_gui_dirty_var = 0.01 }
}

ITA_nuclear_monthly = {
	add_to_variable = { ITA_nuclear_development_completion = ITA_nuclear_development_monthly }
	set_temp_variable = { ITA_old_nuclear_development_cost = ITA_nuclear_development_cost}
	custom_effect_tooltip = ITA_nuclear_monthly_progress.tt
	subtract_from_variable = { ITA_scientist_happiness = ITA_scientist_decay }
	subtract_from_variable = { ITA_nuclear_material_quality = ITA_nuclear_material_decay }
	set_temp_variable = { ITA_nuclear_development_cost_monthly = ITA_nuclear_development_cost }
	divide_temp_variable = { ITA_nuclear_development_cost_monthly = 12 }
	divide_temp_variable = { ITA_nuclear_development_cost_monthly = 1000 }
	log = "[?ITA_nuclear_development_cost_monthly]"
	if = {
		limit = {
			ITA = {
				check_variable = {
					var = money_reserves
					value = ITA_nuclear_development_cost_monthly
					compare = greater_than_or_equals
				}
			}
		}
		ITA = {
			subtract_from_variable = {money_reserves = ITA_nuclear_development_cost_monthly }
		}
	}
	else = {
		ITA = {
			add_to_variable = {national_debt = ITA_nuclear_development_cost_monthly }
		}
	}
	ITA_nuclear_calc = yes
}

ITA_Verona_check_vars = {
	clamp_variable = { var = ITA_Verona_Ciano_Percent min = 0 max = 100 }
	clamp_variable = { var = ITA_Verona_Fascist_Opposition_Percent min = 0 max = 100 }
	clamp_variable = { var = ITA_Verona_outside_pressure min = 0 max = 100 }

	set_temp_variable = { 100_check = ITA_Verona_Ciano_Percent }
	add_to_temp_variable = { 100_check = ITA_Verona_Fascist_Opposition_Percent }
	if = {
		limit = {
			NOT = {
				check_variable = { 100_check = 100 }
			}
		}
		if = {
			limit = {
				check_variable = { 100_check < 100 }
			}
			set_temp_variable = { loops = 100 }
			subtract_from_temp_variable = { loops = 100_check }
		}
		else = {
			set_temp_variable = { loops = 100_check }
			subtract_from_temp_variable = { loops = 100 }
		}
		for_loop_effect = {
			start = 0
			end = loops

			random_list = {
				50 = {
					add_to_variable = { ITA_Verona_Ciano_Percent = 1 }
				}
				50 = {
					add_to_variable = { ITA_Verona_Fascist_Opposition_Percent = 1 }
				}
			}
		}
	}

	set_variable = { ITA_Verona_win_margin = 50 }
	set_temp_variable = { add_to_margin = ITA_Verona_outside_pressure }
	divide_temp_variable = { add_to_margin = 10 }
	add_to_variable = { ITA_Verona_win_margin = add_to_margin }
}

ITA_scrap_all_battleships_and_carriers = {
	custom_effect_tooltip = ITA_scrap_boats_tooltip

	set_temp_variable = { destroyed_ships = num_ships_with_type@carrier }
	add_to_temp_variable = { destroyed_ships = num_ships_with_type@battleship }
	
	#per ship:
	#20 steel
	#8 chromium
	#50 mil
	set_temp_variable = { steel_to_add = 20 }
	set_temp_variable = { chrom_to_add = 8 }
	set_temp_variable = { money_to_add = 0.05 }
	multiply_temp_variable = { steel_to_add = destroyed_ships }
	multiply_temp_variable = { chrom_to_add = destroyed_ships }
	multiply_temp_variable = { money_to_add = destroyed_ships }

	# stockpile_steel_change and stockpile_chromium_change are defined nowhere, so
	# the hulls were being cut up for nothing. The metal goes to the capital.
	capital_scope = {
		add_resource = {
			type = steel
			amount = steel_to_add
		}
		add_resource = {
			type = chromium
			amount = chrom_to_add
		}
	}
	custom_effect_tooltip = ITA_scrap_boats_resources_tooltip
	set_temp_variable = { money_reserves_temp = money_to_add }
	econ_money_reserves_change_raw_money = yes
	
	
	custom_effect_tooltip = ITA_scrap_boats_end_tooltip
	hidden_effect = {
		destroy_ships = {
			type = ship_hull_carrier
			count = all
		}
		destroy_ships = {
			type = ship_hull_heavy
			count = all
		}
	}
}
