GRE_prepare_election = {
	custom_effect_tooltip = GRE_prepare_election_tooltip
	# med's campaign events: 1964 is Kanellopoulos against Pyromaglou; the second election is
	# Karamanlis against whoever leads the EDA's influence bar (Phase E)
	if = {
		limit = { has_country_flag = GRE_second_election }
		if = {
			limit = { check_variable = { GRE_EDA_kyrkos_progress > GRE_EDA_papandreou_progress } }
			country_event = { id = GRE_election_new.3 }
		}
		else = {
			country_event = { id = GRE_election_new.2 }
		}
	}
	else = {
		country_event = { id = GRE_election_new.1 }
	}
	
	for_each_loop = {
		array = GRE_election_state_IDs
		
		#BoG modifiers
		### 0 is ERE, 1 is EDA, 2 is EK
		
		### pop bonus from elite
	
		set_temp_variable = { GRE_BOG_elite_add = global.GRE_BOG_Approval_Elites }
		subtract_from_variable = { GRE_BOG_elite_add = global.GRE_BOG_Disapprove_Elites }
		divide_temp_variable = { GRE_BOG_elite_add = 2 }
		if = {
			limit = {
				has_government = social_conservative # ERE
			}
			add_to_variable = { ERE_pop_elit^i = GRE_BOG_elite_add }
		}
		
		if = {
			limit = {
				OR = { has_government = social_democrat has_government = socialist } # EDA
			}
			add_to_variable = { EDA_pop_elit^i = GRE_BOG_elite_add }
		}

		if = {
			limit = {
				has_government = social_liberal # EK
			}
			add_to_variable = { EK_pop_elit^i = GRE_BOG_elite_add }
		}
		
		### pop bonus from intellectuals
		
		set_temp_variable = { GRE_BOG_intellectuals_add = global.GRE_BOG_Approval_Intellectuals }
		subtract_from_variable = { GRE_BOG_intellectuals_add = global.GRE_BOG_Disapprove_Intellectuals }
		divide_temp_variable = { GRE_BOG_intellectuals_add = 2 }
		if = {
			limit = {
				has_government = social_conservative # ERE
			}
			add_to_variable = { ERE_pop_int^i = GRE_BOG_intellectuals_add }
		}
	
		if = {
			limit = {
				OR = { has_government = social_democrat has_government = socialist } # EDA
			}
			add_to_variable = { EDA_pop_int^i = GRE_BOG_intellectuals_add }
		}

		if = {
			limit = {
				has_government = social_liberal # EK
			}
			add_to_variable = { EK_pop_int^i = GRE_BOG_intellectuals_add }
		}
		
		### pop bonus from workers
		
		set_temp_variable = { GRE_BOG_workers_add = global.GRE_BOG_Approval_Working_Class }
		subtract_from_variable = { GRE_BOG_workers_add = global.GRE_BOG_Disapprove_Working_Class }
		divide_temp_variable = { GRE_BOG_workers_add = 2 }
		if = {
			limit = {
				has_government = social_conservative # ERE
			}
			add_to_variable = { ERE_pop_erg^i = GRE_BOG_workers_add }
		}

		if = {
			limit = {
				OR = { has_government = social_democrat has_government = socialist } # EDA
			}
			add_to_variable = { EDA_pop_erg^i = GRE_BOG_workers_add }
		}
	
		if = {
			limit = {
				has_government = social_liberal # EK
			}
			add_to_variable = { EK_pop_erg^i = GRE_BOG_workers_add }
		}
		
		GRE_election_adjust_popularities = yes
		
		GRE_set_popularity_bars = yes
	}
}

GRE_election_state_victor_loop_effect = {
	log = "HELLO"
	set_temp_variable = { ERE_popularity_total = ERE_pop_elit^i }
	add_to_temp_variable = { ERE_popularity_total = ERE_pop_int^i }
	add_to_temp_variable = { ERE_popularity_total = ERE_pop_erg^i }
	
	add_to_temp_array = { GRE_election_popularities = ERE_popularity_total }
	
	set_temp_variable = { EK_popularity_total = EK_pop_elit^i }
	add_to_temp_variable = { EK_popularity_total = EK_pop_int^i }
	add_to_temp_variable = { EK_popularity_total = EK_pop_erg^i }
	
	add_to_temp_array = { GRE_election_popularities = EK_popularity_total }
	
	set_temp_variable = { EDA_popularity_total = EDA_pop_elit^i }
	add_to_temp_variable = { EDA_popularity_total = EDA_pop_int^i }
	add_to_temp_variable = { EDA_popularity_total = EDA_pop_erg^i }
	
	add_to_temp_array = { GRE_election_popularities = EDA_popularity_total }
	
	set_temp_variable = { KKE_popularity_total = KKE_pop_elit^i }
	add_to_temp_variable = { KKE_popularity_total = KKE_pop_int^i }
	add_to_temp_variable = { KKE_popularity_total = KKE_pop_erg^i }
	
	add_to_temp_array = { GRE_election_popularities = KKE_popularity_total }
	
	set_temp_variable = { MK_popularity_total = MK_pop_elit^i }
	add_to_temp_variable = { MK_popularity_total = MK_pop_int^i }
	add_to_temp_variable = { MK_popularity_total = MK_pop_erg^i }
	
	add_to_temp_array = { GRE_election_popularities = MK_popularity_total }
	
	find_highest_in_array = {
		array = GRE_election_popularities
		index = gre_victor_index
	}
	log = "state winner: [?gre_victor_index]"
	set_variable = { GRE_state_winner = gre_victor_index } # the winner's index, not its share (Phase E)
	
	if = {
		limit = {
			check_variable = { GRE_state_winner = 0 }
		}
		GRE = {
			add_to_array = { GRE_election_frame = 2 }
		}
	}
	else_if = {
		limit = {
			check_variable = { GRE_state_winner = 1 }
		}
		GRE = {
			add_to_array = { GRE_election_frame = 3 }
		}
	}
	else_if = {
		limit = {
			check_variable = { GRE_state_winner = 2 }
		}
		GRE = {
			add_to_array = { GRE_election_frame = 4 }
		}
	}
	else = {
		add_to_array = { GRE_election_frame = 1 }
	}
	
	clear_temp_array = GRE_election_popularities
}

	set_variable = { GRE_socdem_global_bonus_pop = 0 }
	set_variable = { GRE_auth_global_bonus_pop = 0 }
	set_variable = { GRE_liberal_global_bonus_pop = 0 }
	set_variable = { GRE_conservative_global_bonus_pop = 0 }

	every_owned_state = {
		set_variable = { GRE_general_state_turnout = 1 }
		set_variable = { GRE_socdem_state_turnout = 1 }
		set_variable = { GRE_auth_state_turnout = 1 }
		set_variable = { GRE_liberal_state_turnout = 1 }
		set_variable = { GRE_conservative_state_turnout = 1 }
	}
}

GRE_generate_state_party_pops = { #Fires at game start, and when a democratic warlord absorbs another warlord.
	set_variable = { GRE_state_conservative_pop = PREV.party_popularity@social_conservative }
	set_variable = { GRE_state_liberal_pop = PREV.party_popularity@social_liberal }
	set_variable = { GRE_state_socdem_pop = PREV.party_popularity@social_democrat }
	set_variable = { GRE_state_auth_pop = 0 }
	set_temp_variable = { GRE_state_socialist_temp_pop = PREV.party_popularity@socialist }
	set_temp_variable = { GRE_state_communist_temp_pop = PREV.party_popularity@communist }
	set_temp_variable = { GRE_state_despotism_temp_pop = PREV.party_popularity@despotism }
	set_temp_variable = { GRE_state_auth_temp_pop = PREV.party_popularity@authoritarian_democrat }
	set_temp_variable = { GRE_state_fascism_temp_pop = PREV.party_popularity@fascism }
	set_temp_variable = { GRE_state_natsoc_temp_pop = PREV.party_popularity@national_socialism }
	set_temp_variable = { GRE_state_ultranat_temp_pop = PREV.party_popularity@ultranational_socialism }
	set_temp_variable = { GRE_state_burgsys_temp_pop = PREV.party_popularity@burgundian_system }

	#Calculations for Party Fracturing
	set_temp_variable = { GRE_state_socialist_socdem_fraction = GRE_state_socialist_temp_pop }
	set_temp_variable = { GRE_state_socialist_liberal_fraction = GRE_state_socialist_temp_pop }
	set_temp_variable = { GRE_state_communist_socdem_fraction = GRE_state_communist_temp_pop }
	set_temp_variable = { GRE_state_communist_auth_fraction = GRE_state_communist_temp_pop }
	set_temp_variable = { GRE_state_communist_liberal_fraction = GRE_state_communist_temp_pop }
	set_temp_variable = { GRE_state_fascism_conservative_fraction = GRE_state_fascism_temp_pop }
	set_temp_variable = { GRE_state_fascism_auth_fraction = GRE_state_fascism_temp_pop }
	set_temp_variable = { GRE_state_natsoc_conservative_fraction = GRE_state_natsoc_temp_pop }
	set_temp_variable = { GRE_state_natsoc_auth_fraction = GRE_state_natsoc_temp_pop }
	set_temp_variable = { GRE_state_ultranat_conservative_fraction = GRE_state_ultranat_temp_pop }
	set_temp_variable = { GRE_state_ultranat_auth_fraction = GRE_state_ultranat_temp_pop }
	set_temp_variable = { GRE_state_despotism_socdem_fraction = GRE_state_despotism_temp_pop }
	set_temp_variable = { GRE_state_despotism_conservative_fraction = GRE_state_despotism_temp_pop }
	set_temp_variable = { GRE_state_despotism_auth_fraction = GRE_state_despotism_temp_pop }
	set_temp_variable = { GRE_state_auth_socdem_fraction = GRE_state_auth_temp_pop }
	set_temp_variable = { GRE_state_auth_conservative_fraction = GRE_state_auth_temp_pop }
	set_temp_variable = { GRE_state_auth_auth_fraction = GRE_state_auth_temp_pop }
	set_temp_variable = { GRE_state_auth_liberal_fraction = GRE_state_auth_temp_pop }
	set_temp_variable = { GRE_state_burgsys_conservative_fraction = GRE_state_burgsys_temp_pop }
	set_temp_variable = { GRE_state_burgsys_auth_fraction = GRE_state_burgsys_temp_pop }

	#Breakdowns for Party Fracturing
	multiply_temp_variable = { GRE_state_socialist_socdem_fraction = 0.75 }
	multiply_temp_variable = { GRE_state_socialist_liberal_fraction = 0.25 }
	multiply_temp_variable = { GRE_state_communist_socdem_fraction = 0.75 }
	multiply_temp_variable = { GRE_state_communist_auth_fraction = 0.15 }
	multiply_temp_variable = { GRE_state_communist_liberal_fraction = 0.10 }
	multiply_temp_variable = { GRE_state_fascism_conservative_fraction = 0.7 }
	multiply_temp_variable = { GRE_state_fascism_auth_fraction = 0.3 }
	multiply_temp_variable = { GRE_state_natsoc_conservative_fraction = 0.7 }
	multiply_temp_variable = { GRE_state_natsoc_auth_fraction = 0.3 }
	multiply_temp_variable = { GRE_state_ultranat_conservative_fraction = 0.65 }
	multiply_temp_variable = { GRE_state_ultranat_auth_fraction = 0.35 }
	multiply_temp_variable = { GRE_state_despotism_socdem_fraction = 0.2 }
	multiply_temp_variable = { GRE_state_despotism_conservative_fraction = 0.2 }
	multiply_temp_variable = { GRE_state_despotism_liberal_fraction = 0.2 }
	multiply_temp_variable = { GRE_state_despotism_auth_fraction = 0.4 }
	multiply_temp_variable = { GRE_state_auth_socdem_fraction = 0.1 }
	multiply_temp_variable = { GRE_state_auth_conservative_fraction = 0.2 }
	multiply_temp_variable = { GRE_state_auth_liberal_fraction = 0.2 }
	multiply_temp_variable = { GRE_state_auth_auth_fraction = 0.5 }
	multiply_temp_variable = { GRE_state_burgsys_conservative_fraction = 0.8 }
	multiply_temp_variable = { GRE_state_burgsys_auth_fraction = 0.2 }

	#Adding fractured parties to democratic parties
	add_to_variable = { GRE_state_conservative_pop = GRE_state_fascism_conservative_fraction }
	add_to_variable = { GRE_state_conservative_pop = GRE_state_natsoc_conservative_fraction }
	add_to_variable = { GRE_state_conservative_pop = GRE_state_ultranat_conservative_fraction }
	add_to_variable = { GRE_state_conservative_pop = GRE_state_despotism_conservative_fraction }
	add_to_variable = { GRE_state_liberal_pop = GRE_state_socialist_liberal_fraction }
	add_to_variable = { GRE_state_liberal_pop = GRE_state_communist_liberal_fraction }
	add_to_variable = { GRE_state_liberal_pop = GRE_state_despotism_liberal_fraction }
	add_to_variable = { GRE_state_auth_pop = GRE_state_communist_auth_fraction }
	add_to_variable = { GRE_state_auth_pop = GRE_state_fascism_auth_fraction }
	add_to_variable = { GRE_state_auth_pop = GRE_state_ultranat_auth_fraction }
	add_to_variable = { GRE_state_auth_pop = GRE_state_despotism_auth_fraction }
	add_to_variable = { GRE_state_auth_pop = GRE_state_auth_socdem_fraction }
	add_to_variable = { GRE_state_auth_pop = GRE_state_auth_conservative_fraction }
	add_to_variable = { GRE_state_auth_pop = GRE_state_auth_liberal_fraction }
	add_to_variable = { GRE_state_auth_pop = GRE_state_auth_liberal_fraction }
	add_to_variable = { GRE_state_socdem_pop = GRE_state_socialist_socdem_fraction }
	add_to_variable = { GRE_state_socdem_pop = GRE_state_communist_socdem_fraction }
	add_to_variable = { GRE_state_socdem_pop = GRE_state_despotism_socdem_fraction }

	#Defining temporary state pop variables (to be reset by election)
	set_variable = { GRE_state_socdem_temp_pop = 0 }
	set_variable = { GRE_state_auth_temp_pop = 0 }
	set_variable = { GRE_state_liberal_temp_pop = 0 }
	set_variable = { GRE_state_conservative_temp_pop = 0 }

	if = {
		limit = {
			OWNER = {
				has_government = communist
				has_government = socialist
			}
		}
		multiply_variable = { GRE_state_socdem_pop = 1.1 }
	}
	if = {
		limit = {
			OWNER = {
				has_government = social_democrat
			}
		}
		multiply_variable = { GRE_state_socdem_pop = 1.15 }
	}
	if = {
		limit = {
			OWNER = {
				has_government = social_liberal
			}
		}
		multiply_variable = { GRE_state_liberal_pop = 1.15 }
	}
	if = {
		limit = {
			OWNER = {
				has_government = social_conservative
			}
		}
		multiply_variable = { GRE_state_conservative_pop = 1.15 }
	}
	if = {
		limit = {
			OWNER = {
				has_government = authoritarian_democrat
			}
		}
		multiply_variable = { GRE_state_conservative_pop = 1.05 }
		multiply_variable = { GRE_state_auth_pop = 1.05 }
		multiply_variable = { GRE_state_socdem_pop = 1.05 }
		multiply_variable = { GRE_state_liberal_pop = 1.05 }
	}
	if = {
		limit = {
			OWNER = {
				has_government = despotism
			}
		}
		multiply_variable = { GRE_state_conservative_pop = 1.05 }
		multiply_variable = { GRE_state_auth_pop = 1.05 }
		multiply_variable = { GRE_state_socdem_pop = 1.05 }
		multiply_variable = { GRE_state_liberal_pop = 1.05 }
	}
	if = {
		limit = {
			OWNER = {
				OR = {
					has_government = fascism
					has_government = ultranational_socialism
					has_government = burgundian_system
					has_government = national_socialism
				}
			}
		}
		multiply_variable = { GRE_state_conservative_pop = 1.1 }
	}

	#It's the 1960s.
	if = {
		limit = {
			OR = {
				has_state_category = town
				has_state_category = rural
				has_state_category = pastoral
				has_state_category = wasteland
			}
		}
		multiply_variable = { GRE_state_conservative_pop = 1.3 }
	}
	else = {
		multiply_variable = { GRE_state_conservative_pop = 1.15 }
	}

	#Compensating for half of greece being AuthDem
	if = {
		limit = {
			check_variable = { GRE_state_auth_pop > 0 }
		}
		multiply_variable = { GRE_state_auth_pop = 0.8 }
	}

	GRE_get_state_special_party_pop = yes
}

GRE_get_state_special_party_pop = { #Fired every month, to update pops with material conditions
	set_variable = { GRE_state_socdem_condition_pop = 0 }
	set_variable = { GRE_state_auth_condition_pop = 0 }
	set_variable = { GRE_state_liberal_condition_pop = 0 }
	set_variable = { GRE_state_conservative_condition_pop = 0 }

	#Capitals should usually vote for the ruling party
	if = {
		limit = {
			is_capital = yes
		}
		if = {
			limit = {
				OWNER = {
					OR = {
						has_government = communist
						has_government = socialist
						has_government = social_democrat
					}
				}
			}
			add_to_variable = { GRE_state_socdem_condition_pop = 0.15 }
			multiply_variable = { GRE_state_socdem_condition_pop = 1.5 }
		}
		if = {
			limit = {
				OWNER = {
					OR = {
						has_government = social_liberal
					}
				}
			}
			add_to_variable = { GRE_state_liberal_condition_pop = 0.15 }
			multiply_variable = { GRE_state_liberal_condition_pop = 1.5 }
		}
		if = {
			limit = {
				OWNER = {
					OR = {
						has_government = social_conservative
						has_government = fascism
						has_government = ultranational_socialism
						has_government = national_socialism
						has_government = burgundian_system
					}
				}
			}
			add_to_variable = { GRE_state_conservative_condition_pop = 0.15 }
			multiply_variable = { GRE_state_conservative_condition_pop = 1.5 }
		}
		if = {
			limit = {
				OWNER = {
					OR = {
						has_government = authoritarian_democrat
						has_government = despotism
					}
				}
			}
			add_to_variable = { GRE_state_auth_condition_pop = 0.15 }
			multiply_variable = { GRE_state_auth_condition_pop = 1.5 }
		}
	}

	#State categories - rural states are more rightist
	if = {
		limit = {
			has_state_category = wasteland
		}
		add_to_variable = { GRE_state_conservative_condition_pop = 0.2 }
		multiply_variable = { GRE_state_conservative_condition_pop = 1.4 }
		multiply_variable = { GRE_state_liberal_condition_pop = 0.6 }
		multiply_variable = { GRE_state_socdem_condition_pop = 0.8 }
	}
	if = {
		limit = {
			has_state_category = pastoral
		}
		add_to_variable = { GRE_state_conservative_condition_pop = 0.15 }
		multiply_variable = { GRE_state_conservative_condition_pop = 1.3 }
		multiply_variable = { GRE_state_liberal_condition_pop = 0.7 }
		multiply_variable = { GRE_state_socdem_condition_pop = 0.85 }
	}
	if = {
		limit = {
			has_state_category = rural
		}
		add_to_variable = { GRE_state_conservative_condition_pop = 0.1 }
		multiply_variable = { GRE_state_conservative_condition_pop = 1.2 }
		multiply_variable = { GRE_state_liberal_condition_pop = 0.8 }
		multiply_variable = { GRE_state_socdem_condition_pop = 0.9 }
	}
	if = {
		limit = {
			has_state_category = town
		}
		add_to_variable = { GRE_state_conservative_condition_pop = 0.05 }
		multiply_variable = { GRE_state_conservative_condition_pop = 1.1 }
		multiply_variable = { GRE_state_liberal_condition_pop = 0.9 }
		multiply_variable = { GRE_state_socdem_condition_pop = 0.95 }
	}
	if = {
		limit = {
			has_state_category = large_town
		}
		add_to_variable = { GRE_state_liberal_condition_pop = 0.05 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.9 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.1 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.05 }
	}
	if = {
		limit = {
			has_state_category = city
		}
		add_to_variable = { GRE_state_liberal_condition_pop = 0.1 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.8 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.2 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.1 }
	}
	if = {
		limit = {
			has_state_category = large_city
		}
		add_to_variable = { GRE_state_liberal_condition_pop = 0.15 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.7 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.3 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.15 }
	}
	if = {
		limit = {
			has_state_category = metropolis
		}
		add_to_variable = { GRE_state_liberal_condition_pop = 0.20 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.6 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.4 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.2 }
	}
	if = {
		limit = {
			has_state_category = megalopolis
		}
		add_to_variable = { GRE_state_liberal_condition_pop = 0.25 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.5 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.5 }
		multiply_variable = { GRE_state_socdem_condition_pop = 0.10 }
	}

	#Industry makes states more leftist
	if = {
		limit = {
			check_variable = { TNO_combined_factory_total > 10 }
		}
		add_to_variable = { GRE_state_socdem_condition_pop = 0.25 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.7 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.12 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.3 }
	}
	else_if = {
		limit = {
			check_variable = { TNO_combined_factory_total > 9 }
		}
		add_to_variable = { GRE_state_socdem_condition_pop = 0.20 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.75 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.1 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.25 }
	}
	else_if = {
		limit = {
			check_variable = { TNO_combined_factory_total > 8 }
		}
		add_to_variable = { GRE_state_socdem_condition_pop = 0.15 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.8 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.08 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.20 }
	}
	else_if = {
		limit = {
			check_variable = { TNO_combined_factory_total > 7 }
		}
		add_to_variable = { GRE_state_socdem_condition_pop = 0.1 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.85 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.06 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.15 }
	}
	else_if = {
		limit = {
			check_variable = { TNO_combined_factory_total > 6 }
		}
		add_to_variable = { GRE_state_socdem_condition_pop = 0.05 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.9 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.04 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.1 }
	}
	else_if = {
		limit = {
			check_variable = { TNO_combined_factory_total > 5 }
		}
		multiply_variable = { GRE_state_conservative_condition_pop = 0.95 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.02 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.05 }
	}
	else_if = {
		limit = {
			check_variable = { TNO_combined_factory_total > 4 }
		}
		multiply_variable = { GRE_state_conservative_condition_pop = 1.0 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.0 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.0 }
	}
	else_if = {
		limit = {
			check_variable = { TNO_combined_factory_total > 3 }
		}
		multiply_variable = { GRE_state_conservative_condition_pop = 1.05 }
		multiply_variable = { GRE_state_liberal_condition_pop = 0.98 }
		multiply_variable = { GRE_state_socdem_condition_pop = 0.95 }
	}
	else_if = {
		limit = {
			check_variable = { TNO_combined_factory_total > 2 }
		}
		multiply_variable = { GRE_state_conservative_condition_pop = 1.1 }
		multiply_variable = { GRE_state_liberal_condition_pop = 0.96 }
		multiply_variable = { GRE_state_socdem_condition_pop = 0.9 }
	}

	#Education makes states more leftist or rightist, depending
	if = {
		limit = {
			OWNER = {
				has_idea = tno_academic_base_mass_illiteracy
			}
		}
		multiply_variable = { GRE_state_conservative_condition_pop = 1.5 }
		multiply_variable = { GRE_state_auth_condition_pop = 1.25 }
		multiply_variable = { GRE_state_liberal_condition_pop = 0.5 }
		multiply_variable = { GRE_state_socdem_condition_pop = 0.75 }
	}
	else_if = {
		limit = {
			OWNER = {
				has_idea = tno_academic_base_basic_literacy
			}
		}
		multiply_variable = { GRE_state_conservative_condition_pop = 1.25 }
		multiply_variable = { GRE_state_auth_condition_pop = 1.1 }
		multiply_variable = { GRE_state_liberal_condition_pop = 0.75 }
		multiply_variable = { GRE_state_socdem_condition_pop = 0.9 }
	}
	else_if = {
		limit = {
			OWNER = {
				has_idea = tno_academic_base_primary_schooling
			}
		}
		multiply_variable = { GRE_state_conservative_condition_pop = 1 }
		multiply_variable = { GRE_state_auth_condition_pop = 1 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1 }
	}
	else_if = {
		limit = {
			OWNER = {
				has_idea = tno_academic_base_secondary_schooling
			}
		}
		add_to_variable = { GRE_state_liberal_condition_pop = 0.05 }
		multiply_variable = { GRE_state_conservative_condition_pop = 0.75 }
		multiply_variable = { GRE_state_auth_condition_pop = 0.9 }
		multiply_variable = { GRE_state_liberal_condition_pop = 1.25 }
		multiply_variable = { GRE_state_socdem_condition_pop = 1.1 }
	}
	else_if = {
		limit = {
			OWNER = {
				has_idea = tno_academic_base_tertiary_schooling
			}
