# The party variables it reads exist only under _WIP (GRE_TUR_med_wip_startup_global). Elsewhere they read 0,
# and the correction below then hands one random party 100%; a leak through the Italo-Turkish War's peace did
# that to legacy Turkey until 2026-10-24.
TUR_Pie_Chart_Update = {
	if = {
		limit = { has_global_flag = ITA_MED_WIP_PATHS }
		TUR_Pie_Chart_Update_apply = yes
	}
}

TUR_Pie_Chart_Update_apply = {
	set_temp_variable = { TUR_socialist_temp = TUR_DYP_Popularity }
	multiply_temp_variable = { TUR_socialist_temp = TUR_DYP_Socialist }
	set_temp_variable = { TUR_socdem_temp = TUR_DYP_Popularity }
	multiply_temp_variable = { TUR_socdem_temp = TUR_DYP_Progressive }
	set_temp_variable = { TUR_libdem_temp = TUR_DYP_Popularity }
	multiply_temp_variable = { TUR_libdem_temp = TUR_DYP_Liberal }
	set_temp_variable = { TUR_authdem_temp = TUR_CHP_Popularity }
	set_temp_variable = { TUR_fascist_temp = TUR_UDP_Popularity }
	multiply_temp_variable = { TUR_fascist_temp = TUR_UDP_Nationalist }
	set_temp_variable = { TUR_condem_temp = TUR_UDP_Popularity }
	subtract_from_temp_variable = { TUR_condem_temp = TUR_fascist_temp }
	
	set_temp_variable = { TUR_Popularity_Change_Correction = TUR_socialist_temp }
	add_to_temp_variable = { TUR_Popularity_Change_Correction = TUR_socdem_temp }
	add_to_temp_variable = { TUR_Popularity_Change_Correction = TUR_libdem_temp }
	add_to_temp_variable = { TUR_Popularity_Change_Correction = TUR_condem_temp }
	add_to_temp_variable = { TUR_Popularity_Change_Correction = TUR_authdem_temp }
	add_to_temp_variable = { TUR_Popularity_Change_Correction = TUR_fascist_temp }
	if = {
		limit = {
			NOT = {
				check_variable = { TUR_Popularity_Change_Correction = 100 }
			}
		}
		subtract_from_temp_variable = { TUR_Popularity_Change_Correction = 100 }
		random_list = {
			1 = {
				subtract_from_temp_variable = { TUR_socialist_temp = TUR_Popularity_Change_Correction }
			}
			1 = {
				subtract_from_temp_variable = { TUR_socdem_temp = TUR_Popularity_Change_Correction }
			}
			1 = {
				subtract_from_temp_variable = { TUR_libdem_temp = TUR_Popularity_Change_Correction }
			}
			1 = {
				subtract_from_temp_variable = { TUR_condem_temp = TUR_Popularity_Change_Correction }
			}
			1 = {
				subtract_from_temp_variable = { TUR_authdem_temp = TUR_Popularity_Change_Correction }
			}
			1 = {
				subtract_from_temp_variable = { TUR_fascist_temp = TUR_Popularity_Change_Correction }
			}
		}
	}
	
	set_popularities = {
		communist = 0
		socialist = TUR_socialist_temp
		social_democrat = TUR_socdem_temp
		social_liberal = TUR_libdem_temp
		social_conservative = TUR_condem_temp
		authoritarian_democrat = TUR_authdem_temp
		despotism = 0
		fascism = TUR_fascist_temp
		national_socialism = 0
		ultranational_socialism = 0
		burgundian_system = 0
	}
}

TUR_Change_DYP_Popularity = {
	hidden_effect = {
		set_temp_variable = { TUR_Party_Popularity_Distribute = 100 }
		subtract_from_temp_variable = { TUR_Party_Popularity_Distribute = TUR_DYP_Popularity }
		set_temp_variable = { TUR_CHP_Popularity_Change = TUR_CHP_Popularity }
		divide_temp_variable = { TUR_CHP_Popularity_Change = TUR_Party_Popularity_Distribute }
		multiply_temp_variable = { TUR_CHP_Popularity_Change = TUR_DYP_Pop_Change }
		set_temp_variable = { TUR_UDP_Popularity_Change = TUR_UDP_Popularity }
		divide_temp_variable = { TUR_UDP_Popularity_Change = TUR_Party_Popularity_Distribute }
		multiply_temp_variable = { TUR_UDP_Popularity_Change = TUR_DYP_Pop_Change }
		add_to_variable = { TUR_DYP_Popularity = TUR_DYP_Pop_Change }
		subtract_from_variable = { TUR_CHP_Popularity = TUR_CHP_Popularity_Change }
		subtract_from_variable = { TUR_UDP_Popularity = TUR_UDP_Popularity_Change }
		if = {
			limit = {
				check_variable = { TUR_CHP_Popularity_Change = 0 }
				check_variable = { TUR_UDP_Popularity_Change = 0 }
			}
			divide_temp_variable = { TUR_DYP_Pop_Change = 2 }
			add_to_variable = { TUR_CHP_Popularity = TUR_DYP_Pop_Change }
			add_to_variable = { TUR_UDP_Popularity = TUR_DYP_Pop_Change }
		}
		set_temp_variable = { TUR_Popularity_Change_Correction = TUR_DYP_Popularity }
		add_to_temp_variable = { TUR_Popularity_Change_Correction = TUR_CHP_Popularity }
		add_to_temp_variable = { TUR_Popularity_Change_Correction = TUR_UDP_Popularity }
		if = {
			limit = {
				check_variable = { TUR_Popularity_Change_Correction < 100 }
			}
			add_to_variable = { TUR_DYP_Popularity = 0.01 }
		}
		else_if = {
			limit = {
				check_variable = { TUR_Popularity_Change_Correction > 100 }
			}
			add_to_variable = { TUR_DYP_Popularity = -0.01 }
		}
		clamp_variable = {
			var = TUR_DYP_Popularity
			min = 0
			max = 100
		}
		clamp_variable = {
			var = TUR_CHP_Popularity
			min = 0
			max = 100
		}
		clamp_variable = {
			var = TUR_UDP_Popularity
			min = 0
			max = 100
		}
		TUR_Pie_Chart_Update = yes
	}
	custom_effect_tooltip = TUR_DYP_popularity_change_tt
}

TUR_DoW_change_stage = {
	hidden_effect = {
		clamp_variable = {
			var = TUR_DOW
			min = 0
			max = 100
		}
		
		if = {
			limit = {
				is_on_stage1 = yes
			}
			if = {
				limit = {
					NOT = {
						has_idea = TUR_drums_of_war_1
					}
				}
				remove_ideas = {
					TUR_drums_of_war_2
					TUR_drums_of_war_3
				}
				add_ideas = TUR_drums_of_war_1
			}
		}
		if = {
			limit = {
				is_on_stage2 = yes
			}
			if = {
				limit = {
					NOT = {
						has_idea = TUR_drums_of_war_2
					}
				}
				remove_ideas = {
					TUR_drums_of_war_1
					TUR_drums_of_war_3
				}
				add_ideas = TUR_drums_of_war_2
				#Give Inönü some positive traits
			}
		}
		if = {
			limit = {
				is_on_stage3 = yes
			}
			if = {
				limit = {
					NOT = {
						has_idea = TUR_drums_of_war_3
					}
				}
				remove_ideas = {
					TUR_drums_of_war_1
					TUR_drums_of_war_2
				}
				add_ideas = TUR_drums_of_war_3
				#Give Inönü a negative trait
			}
		}
	}
}

TUR_calculate_peace_outcome_Greece = {
	if = { ##### Total victory against Greece
		limit = {
			TUR = {
				has_war_with = GRE
				controls_state = 731
				controls_state = 47
				controls_state = 185
				controls_state = 1262
				controls_state = 186
			}
		}
		GRE = {
			save_global_event_target_as = losing_country
			remove_ideas = ITW_surrender
			country_event = { id = grePostITW.20 days = 2 }
			# Remove naxos from greece resistance GUI calcs
			remove_from_array = { 
				array = GRE_Resistance_Government_Control 
				index = 7
			}
			remove_from_array = { 
				array = GRE_Resistance_EDES_Control 
				index = 7
			}
			remove_from_array = { 
				array = GRE_Resistance_EAM_Control 
				index = 7
			}
			remove_from_array = { 
				array = GRE_Resistance_State_Controller 
				index = 7
			}
		}
		TUR = {
			save_global_event_target_as = winning_country
			white_peace = GRE
			transfer_state = 187
			1263 = { set_demilitarized_zone = yes }
			187 = { #Add a port to naxos so you can actually leave if you're stuck there
				add_building_construction = {
					type = naval_base
					level = 1
					province = 13241
				}
			}
			add_to_variable = { TUR_DOW = 20 }
			TUR_DoW_change_stage = yes
			set_country_flag = TUR_GTW_won
			set_global_flag = war_won
			set_global_flag = show_peace_popup_alert
			add_to_variable = { TUR_ITW_win_counter = 1 }
			unlock_national_focus = TUR_an_old_rival_a_new_war
			unlock_national_focus = TUR_watching_the_border
			unlock_national_focus = TUR_no_enemy_shall_step_on_turkish_soil
			unlock_national_focus = TUR_onwards_to_athens
			complete_national_focus = TUR_mutilation_progress
		}
		if = { #Early Turkish Victory also ends the Cyprus war
			limit = {
				NOT = { has_global_flag = Cyprus_War_Over }
				has_global_flag = TUR_Cyprus_Conflict
			}
			GRE = { delete_unit_template_and_units = { division_template = "Ethnikí Frourá" } }
			TUR = { delete_unit_template_and_units = { division_template = "Türk Mukavemat Teşkilatı" } }
			TCP = {
				transfer_state = 183
				set_cosmetic_tag = TCP_unified
			}
			183 = {
				add_building_construction = {
					type = naval_base
					level = 2
					province = 11984
				}
			}
			1328 = {
				add_building_construction = {
					type = naval_base
					level = 2
					province = 7193
				}
			}
			TUR = { 
				remove_mission = Liberate_North_Cyprus
				remove_mission = Liberate_South_Cyprus
				
				unlock_national_focus = TUR_a_holiday_in_cyprus
				unlock_national_focus = TUR_to_the_south
				complete_national_focus = TUR_cyprus_cannot_remain_greek
			}
			
		}
		else_if = {
			limit = { #If the Cyprus war hasn't even begun yet (~~paradrops~~ naval invasions), give turkey northern cyprus
				NOT = { has_global_flag = TUR_Cyprus_Conflict }
				NOT = { has_global_flag = Cyprus_War_Over }
			}
			TCP = {
				transfer_state = 1328
				#add_ideas = ITW_surrender
				add_state_core = 1328
				add_state_claim = 183
			}
			TUR = {
				#load_oob = TUR_TCP
				puppet = TCP
			}
			TCP = {
				set_popularities = {
					communist = 0
					socialist = 0
					social_democrat = 14
					social_liberal = 0
					social_conservative = 34
					authoritarian_democrat = 52
					despotism = 0
					fascism = 0
					national_socialism = 0
					ultranational_socialism = 0
					burgundian_system = 0
				}
			}
			TCP = { set_cosmetic_tag = TCP_split }
			add_to_variable = { TUR_ITW_win_counter = 1 }
			#set_global_flag = Cyprus_War_Over #should this be set if the war never happened?
		}
	}
	else_if = { ##### Victory against Greece

			TUR = {
				has_war_with = GRE
				controls_state = 731
			}

			transfer_state = 187

					level = 1

				}

			set_country_flag = TUR_GTW_won
			set_global_flag = war_won
			set_global_flag = show_peace_popup_alert
			add_to_variable = { TUR_ITW_win_counter = 0.5 }
			unlock_national_focus = TUR_an_old_rival_a_new_war
			unlock_national_focus = TUR_watching_the_border
			unlock_national_focus = TUR_no_enemy_shall_step_on_turkish_soil
			unlock_national_focus = TUR_onwards_to_athens
			complete_national_focus = TUR_mutilation_progress
		}
	}
	else_if = { ##### Status Quo in Greece
		limit = {
			TUR = {
				has_war_with = GRE
				has_full_control_of_state = 184
			}
		}
		GRE = {
			save_global_event_target_as = winning_country
			remove_ideas = ITW_surrender
			country_event = { id = grePostITW.20 days = 2 }
		}
		TUR = {
			save_global_event_target_as = losing_country #Counted as a loss for the popup since they don't actually take any land
			white_peace = GRE
			set_country_flag = TUR_GTW_lost
			set_global_flag = war_won
			set_global_flag = show_peace_popup_alert

TUR_TV_update_GUI = {
	clamp_variable = { var = TUR_japan_popularity min = 0 max = 100 }
	clamp_variable = { var = TUR_america_popularity min = 0 max = 100 }
	clamp_variable = { var = TUR_turkey_popularity min = 0 max = 100 }
	
    set_variable = { TUR_japan_pop = TUR_japan_popularity }
    set_variable = { TUR_america_pop = TUR_america_popularity }
	set_variable = { TUR_turkey_pop = TUR_turkey_popularity }

    # Compute start and end values
    # America is first so its start is 0 and end is just its %
    set_temp_variable = { TUR_america_pop_temp = 0 }
    set_temp_variable = { TUR_america_pop_end = TUR_america_pop }

    # Compute frame = SSSEEE by shifting over the start number 3 places and adding the end number
    set_variable = { TUR_america_pop_show = TUR_america_pop_temp } 
    round_variable = TUR_america_pop_show
    multiply_variable = { TUR_america_pop_show = 1000 }
    add_to_variable = { TUR_america_pop_show = TUR_america_pop_end }

    # Compute start and end values
    # Japan starts where America ends
    set_temp_variable = { TUR_japan_pop_temp = TUR_america_pop_end }
    set_temp_variable = { TUR_japan_pop_end = TUR_japan_pop_temp }
    add_to_temp_variable = { TUR_japan_pop_end = TUR_japan_pop }

    set_variable = { TUR_japan_pop_show = TUR_japan_pop_temp } 
    round_variable = TUR_japan_pop_show 
    multiply_variable = { TUR_japan_pop_show = 1000 }
    add_to_variable = { TUR_japan_pop_show = TUR_japan_pop_end }

	# Next is Turkey, starts where Japan ends
	set_temp_variable = { TUR_turkey_pop_temp = TUR_japan_pop_end }
    set_temp_variable = { TUR_turkey_pop_end = TUR_turkey_pop_temp }
    add_to_temp_variable = { TUR_turkey_pop_end = TUR_turkey_pop }

    set_variable = { TUR_turkey_pop_show = TUR_turkey_pop_temp } 
    round_variable = TUR_turkey_pop_show 
    multiply_variable = { TUR_turkey_pop_show = 1000 }
    add_to_variable = { TUR_turkey_pop_show = TUR_turkey_pop_end }
}
