MAD_plague_sim_initial_setup = {
	MAD = {
		# INITIALISATION
		every_owned_state = {
			set_variable = { MAD_plague_pop_state_density = 0.5 }
			set_variable = { MAD_plague_healthcare_access = 0.5 }
		}

		# MAD_plague_pop_state_density - Relates to the state's population density (0 is uninhabited, 1 is urban density)
		# MAD_plague_healthcare_access - Relates to access to healthcare (0 is no access, 1 is easy access)

		# If MAD_plague_healthcare_access > MAD_plague_pop_state_density, the disease will start to be eliminated by itself
		# (while the contamination factor is at least the recovery factor; planes and burning halve the contamination factor)

		543 = { # Rural
			set_variable = { MAD_plague_pop_state_density = 0.5 }
			set_variable = { MAD_plague_healthcare_access = 0.2 }
		}

		706 = { # Small Island
			set_variable = { MAD_plague_pop_state_density = 0.6 }
			set_variable = { MAD_plague_healthcare_access = 0.5 }
		}

		707 = { # Small Island
			set_variable = { MAD_plague_pop_state_density = 0.6 }
			set_variable = { MAD_plague_healthcare_access = 0.4 }
		}

		944 = { # Town
			set_variable = { MAD_plague_pop_state_density = 0.8 }
			set_variable = { MAD_plague_healthcare_access = 0.8 }
		}

		945 = { # Town
			set_variable = { MAD_plague_pop_state_density = 0.7 }
			set_variable = { MAD_plague_healthcare_access = 0.3 }
		}

		946 = { # Town
			set_variable = { MAD_plague_pop_state_density = 0.6 }
			set_variable = { MAD_plague_healthcare_access = 0.5 }
		}

		947 = { # Rural
			set_variable = { MAD_plague_pop_state_density = 0.2 }
			set_variable = { MAD_plague_healthcare_access = 0.1 }
		}

		1041 = { # Rural
			set_variable = { MAD_plague_pop_state_density = 0.3 }
			set_variable = { MAD_plague_healthcare_access = 0.2 }
		}

		1042 = { # Rural
			set_variable = { MAD_plague_pop_state_density = 0.5 }
			set_variable = { MAD_plague_healthcare_access = 0.2 }
		}

		1043 = { # Rural
			set_variable = { MAD_plague_pop_state_density = 0.3 }
			set_variable = { MAD_plague_healthcare_access = 0.1 }
		}

		set_variable = { MAD_plague_number_aid = 5 }
		set_variable = { MAD_plague_base_contamination_factor = 0.3 }
		set_variable = { MAD_plague_recovery_factor = 0.3 }

		# COMPUTE ISLAND GLOBAL STATUS

		set_variable = { MAD_plague_percent_global = 0 }
		set_variable = { MAD_plague_percent_global_norm = 0 }

		every_owned_state = {
			add_to_variable = { MAD.MAD_plague_percent_global = MAD_plague_percent_population }
			add_to_variable = { MAD.MAD_plague_percent_global_norm = 1 }
		}

		divide_variable = { MAD_plague_percent_global = MAD_plague_percent_global_norm }

		MAD_modify_plague_status = yes
	}
}

MAD_plague_sim_tick = {
	MAD = {
		every_owned_state = {
			# MAD_plague_percent_population - Piece of the population currently contaminated (%)
			# MAD_plague_pop_state_density - Local population density of the state (factor)
			# MAD_plague_healthcare_access - Access to hospital or basic healthcare (factor)
			# MAD_plague_contamination_prob - Probability of contaminationin the state (%)
			# MAD_plague_across_border_prob - Probability for the disease to spread to a neighboring state (%)

			# The contamination and recovery factors live on MAD, not on the state: read them as MAD.<var>
			set_temp_variable = { MAD_plague_contamination_prob = MAD_plague_pop_state_density }
			divide_temp_variable = { MAD_plague_contamination_prob = MAD_plague_healthcare_access }
			multiply_temp_variable = { MAD_plague_contamination_prob = MAD.MAD_plague_base_contamination_factor }
			multiply_temp_variable = { MAD_plague_contamination_prob = MAD_plague_percent_population }
			clamp_temp_variable = {
				var = MAD_plague_contamination_prob
			    min = 0
			    max = 1
			}

			set_temp_variable = { MAD_plague_healthy_population = 1 }
			subtract_from_temp_variable = { MAD_plague_healthy_population = MAD_plague_percent_population }

			set_temp_variable = { MAD_plague_newly_contaminated_population = MAD_plague_healthy_population }
			multiply_temp_variable = { MAD_plague_newly_contaminated_population = MAD_plague_contamination_prob }

			# Recoveries: a fixed share of the sick recover each week
			set_temp_variable = { MAD_plague_recovered_population = MAD.MAD_plague_recovery_factor }
			multiply_temp_variable = { MAD_plague_recovered_population = MAD_plague_percent_population }

			set_temp_variable = { MAD_plague_new_percent_population = MAD_plague_percent_population }
			add_to_temp_variable = { MAD_plague_new_percent_population = MAD_plague_newly_contaminated_population }
			subtract_from_temp_variable = { MAD_plague_new_percent_population = MAD_plague_recovered_population }
			clamp_temp_variable = {
				var = MAD_plague_new_percent_population
			    min = 0
			    max = 1
			}

			set_variable = { MAD_plague_percent_population = MAD_plague_new_percent_population }
			# log = "[GetDateText]: [Root.GetName] - [This.GetName]: COMPUTING NEIGHBORS SPREADING"

			every_neighbor_state = {
				# Spread is a sixth of the contamination factor (0.05 at the start), so planes and burning slow it too,
				# and only reaches the neighbor's healthy share
				set_temp_variable = { MAD_plague_across_border_prob = MAD.MAD_plague_base_contamination_factor }
				divide_temp_variable = { MAD_plague_across_border_prob = 6 }

				set_temp_variable = { MAD_plague_add_percent_population = MAD_plague_across_border_prob }
				multiply_temp_variable = { MAD_plague_add_percent_population = PREV.MAD_plague_percent_population }

				set_temp_variable = { MAD_plague_neighbor_healthy_population = 1 }
				subtract_from_temp_variable = { MAD_plague_neighbor_healthy_population = MAD_plague_percent_population }
				multiply_temp_variable = { MAD_plague_add_percent_population = MAD_plague_neighbor_healthy_population }
				# log = "[GetDateText]: [Root.GetName] - [This.GetName]: MAD_plague_add_percent_population = [?This.MAD_plague_add_percent_population]"

				if = {
					limit = {
						has_state_flag = MAD_state_quarantined
					}
					divide_temp_variable = { MAD_plague_add_percent_population = 10 }
				}

				if = {
					limit = {
						PREV = {
							has_state_flag = MAD_state_quarantined
						}
					}
					divide_temp_variable = { MAD_plague_add_percent_population = 10 }
				}

				add_to_variable = { MAD_plague_percent_population = MAD_plague_add_percent_population }
				clamp_variable = {
					var = MAD_plague_percent_population
					min = 0
					max = 1
				}
			}

			# log = "[GetDateText]: [Root.GetName] - [This.GetName]: MAD_plague_percent_population = [?This.MAD_plague_percent_population]"
			# log = "[GetDateText]: [Root.GetName] - [This.GetName]: MAD_plague_pop_state_density = [?This.MAD_plague_pop_state_density]"
			# log = "[GetDateText]: [Root.GetName] - [This.GetName]: MAD_plague_healthcare_access = [?This.MAD_plague_healthcare_access]"
			# log = "[GetDateText]: [Root.GetName] - [This.GetName]: MAD_plague_contamination_prob = [?This.MAD_plague_contamination_prob]"
			# log = "[GetDateText]: [Root.GetName] - [This.GetName]: MAD_plague_across_border_prob = [?This.MAD_plague_across_border_prob]"
		}

		# COMPUTE ISLAND GLOBAL STATUS

		set_variable = { MAD_plague_percent_global = 0 }
		set_variable = { MAD_plague_percent_global_norm = 0 }

		every_owned_state = {
			add_to_variable = { MAD.MAD_plague_percent_global = MAD_plague_percent_population }
			add_to_variable = { MAD.MAD_plague_percent_global_norm = 1 }
		}

		divide_variable = { MAD_plague_percent_global = MAD_plague_percent_global_norm }

		MAD_modify_plague_status = yes
	}
}

# Tiers by island average: none at 0, then <5%, <20%, <40%, <60%, and 60% or more.
# Run in MAD scope.
MAD_modify_plague_status = {
	if = {
		limit = {
			NOT = { check_variable = { MAD_plague_percent_global > 0 } }
		}

		remove_ideas = {
			MAD_idea_die_schwarze_pest_insignificant
			MAD_idea_die_schwarze_pest_small
			MAD_idea_die_schwarze_pest_concerning
			MAD_idea_die_schwarze_pest_catastrophic
			MAD_idea_die_schwarze_pest_disastrous
		}
	}
	else_if = {
		limit = {
			check_variable = { MAD_plague_percent_global < 0.05 }
		}

		if = {
			limit = { NOT = { has_idea = MAD_idea_die_schwarze_pest_insignificant } }

			remove_ideas = {
				MAD_idea_die_schwarze_pest_small
				MAD_idea_die_schwarze_pest_concerning
				MAD_idea_die_schwarze_pest_catastrophic
				MAD_idea_die_schwarze_pest_disastrous
			}

			add_ideas = { MAD_idea_die_schwarze_pest_insignificant }
		}
	}
	else_if = {
		limit = {
			check_variable = { MAD_plague_percent_global < 0.2 }
		}

		if = {
			limit = { NOT = { has_idea = MAD_idea_die_schwarze_pest_small } }

			remove_ideas = {
				MAD_idea_die_schwarze_pest_insignificant
				MAD_idea_die_schwarze_pest_concerning
				MAD_idea_die_schwarze_pest_catastrophic
				MAD_idea_die_schwarze_pest_disastrous
			}

			add_ideas = { MAD_idea_die_schwarze_pest_small }
		}
	}
	else_if = {
		limit = {
			check_variable = { MAD_plague_percent_global < 0.4 }
		}

		if = {
			limit = { NOT = { has_idea = MAD_idea_die_schwarze_pest_concerning } }

			remove_ideas = {
				MAD_idea_die_schwarze_pest_insignificant
				MAD_idea_die_schwarze_pest_small
				MAD_idea_die_schwarze_pest_catastrophic
				MAD_idea_die_schwarze_pest_disastrous
			}

			add_ideas = { MAD_idea_die_schwarze_pest_concerning }
		}
	}
	else_if = {
		limit = {
			check_variable = { MAD_plague_percent_global < 0.6 }
		}

		if = {
			limit = { NOT = { has_idea = MAD_idea_die_schwarze_pest_catastrophic } }

			remove_ideas = {
				MAD_idea_die_schwarze_pest_insignificant
				MAD_idea_die_schwarze_pest_small
				MAD_idea_die_schwarze_pest_concerning
				MAD_idea_die_schwarze_pest_disastrous
			}

			add_ideas = { MAD_idea_die_schwarze_pest_catastrophic }
		}
	}
	else = {
		if = {
			limit = { NOT = { has_idea = MAD_idea_die_schwarze_pest_disastrous } }

			remove_ideas = {
				MAD_idea_die_schwarze_pest_insignificant
				MAD_idea_die_schwarze_pest_small
				MAD_idea_die_schwarze_pest_concerning
				MAD_idea_die_schwarze_pest_catastrophic
			}

			add_ideas = { MAD_idea_die_schwarze_pest_disastrous }
		}
	}
}

# A human MAD who followed the Milch, Hitnagdut or Malagasy route takes over that side when the war starts.
# Run in MAD scope.
MAD_hand_player_to_route_faction = {
	if = {
		limit = { is_ai = no }
		if = {
			limit = {
				has_country_flag = MAD_milch_path
				country_exists = MDG
			}
			MDG = { change_tag_from = ROOT }
		}
		else_if = {
			limit = {
				has_country_flag = MAD_hitnagdut_path
				country_exists = MDJ
			}
			MDJ = { change_tag_from = ROOT }
		}
		else_if = {
			limit = {
				has_country_flag = MAD_malagasy_path
				country_exists = MDM
			}
			MDM = { change_tag_from = ROOT }
		}
	}
}

# Run in the winning side's scope. Its sponsor (MADAGASKAR_CIVIL_WAR.6) gets prestige, port access and
# the flag the scramble for Africa will read. Port access is military access: no docking-rights effect exists.
MAD_proxy_war_reward = {
	if = {
		limit = {
			original_tag = MAD
			USA = { has_country_flag = MAD_proxy_sponsor_MAD }
		}
		give_military_access = USA
		USA = { MAD_proxy_war_sponsor_reward = yes }
	}
	else_if = {
		limit = {
			original_tag = MDG
			GER = { has_country_flag = MAD_proxy_sponsor_MDG }
		}
		give_military_access = GER
		GER = { MAD_proxy_war_sponsor_reward = yes }
	}
	else_if = {
		limit = {
			original_tag = MDJ
			ITA = { has_country_flag = MAD_proxy_sponsor_MDJ }
		}
		give_military_access = ITA
		ITA = { MAD_proxy_war_sponsor_reward = yes }
	}
	else_if = {
		limit = {
			original_tag = MDM
			JAP = { has_country_flag = MAD_proxy_sponsor_MDM }
		}
		give_military_access = JAP
		JAP = { MAD_proxy_war_sponsor_reward = yes }
	}
}

# Removes Madagascar Breakdown and every plague tier from whichever civil-war tags still exist.
# Called once the war is over (a victor is decided, or an outside power took the last side).
MAD_end_civil_war_ideas = {
	every_country = {
		limit = {
			OR = {
				original_tag = MAD
				original_tag = MDG
				original_tag = MDJ
				original_tag = MDM
			}
		}
		if = {
			limit = { has_idea = MAD_idea_madagascar_breakdown }
			remove_ideas = MAD_idea_madagascar_breakdown
		}
		if = {
			limit = {
				OR = {
					has_idea = MAD_idea_die_schwarze_pest_insignificant
					has_idea = MAD_idea_die_schwarze_pest_small
					has_idea = MAD_idea_die_schwarze_pest_concerning
					has_idea = MAD_idea_die_schwarze_pest_catastrophic
					has_idea = MAD_idea_die_schwarze_pest_disastrous
				}
			}
			remove_ideas = {
				MAD_idea_die_schwarze_pest_insignificant
				MAD_idea_die_schwarze_pest_small
				MAD_idea_die_schwarze_pest_concerning
				MAD_idea_die_schwarze_pest_catastrophic
				MAD_idea_die_schwarze_pest_disastrous
			}
		}
	}
}

# Run in the scope of a civil-war side that has just annexed another (on_annex): it takes cores on every
# Madagascar state it now owns.

				state = 543

				state = 945

	}
}

# Called from on_annex while the civil war runs (ROOT annexer, FROM annexed). A tag counts as gone if it no
# longer exists or is the one being annexed, so it works whoever does the annexing, including outside powers.
MAD_civil_war_check_end = {
	# MAD and MDG are gone but both rebel movements remain: they turn on each other
	if = {
		limit = {
			NOT = { has_global_flag = madagaskar_war_of_the_oppressed }
			OR = {
				NOT = { country_exists = MAD }
				FROM = { original_tag = MAD }
			}
			OR = {
				NOT = { country_exists = MDG }
				FROM = { original_tag = MDG }
			}
			country_exists = MDJ
			NOT = { FROM = { original_tag = MDJ } }
			country_exists = MDM
			NOT = { FROM = { original_tag = MDM } }
		}
		set_global_flag = madagaskar_war_of_the_oppressed
		MDM = {
			country_event = {
				id = MADAGASKAR_CIVIL_WAR.3
				days = 7
			}
		}
	}

	# MAD VICTORY
	else_if = {
		limit = {
			country_exists = MAD
			NOT = { FROM = { original_tag = MAD } }
			OR = {
				NOT = { country_exists = MDG }
				FROM = { original_tag = MDG }
			}
			OR = {
				NOT = { country_exists = MDJ }
				FROM = { original_tag = MDJ }
			}
			OR = {
				NOT = { country_exists = MDM }
				FROM = { original_tag = MDM }
			}
		}
		clr_global_flag = madagaskar_civil_war
		clr_global_flag = madagaskar_war_of_the_oppressed
		MAD_end_civil_war_ideas = yes
		MAD = { MAD_proxy_war_reward = yes }
		MAD = { country_event = { id = MADAGASKAR_CIVIL_WAR.5 } }
		USA = { country_event = { id = USA.madagascar.5 } }
	}

	# MDG VICTORY
	else_if = {
		limit = {
			country_exists = MDG
			NOT = { FROM = { original_tag = MDG } }
			OR = {
				NOT = { country_exists = MAD }
				FROM = { original_tag = MAD }
			}
			OR = {
				NOT = { country_exists = MDJ }
				FROM = { original_tag = MDJ }
			}
			OR = {
				NOT = { country_exists = MDM }
				FROM = { original_tag = MDM }
			}
		}
		clr_global_flag = madagaskar_civil_war
		clr_global_flag = madagaskar_war_of_the_oppressed
		MAD_end_civil_war_ideas = yes
		MDG = {
			MAD_proxy_war_reward = yes
			MAD_inherit_madagaskar_admirals = yes
		}
		USA = { country_event = { id = USA.madagascar.4 } }
	}

	# MDJ VICTORY
	else_if = {
		limit = {
			country_exists = MDJ
			NOT = { FROM = { original_tag = MDJ } }
			OR = {
				NOT = { country_exists = MAD }
				FROM = { original_tag = MAD }
			}
			OR = {
				NOT = { country_exists = MDG }
				FROM = { original_tag = MDG }
			}
			OR = {
				NOT = { country_exists = MDM }
				FROM = { original_tag = MDM }
			}
		}
		clr_global_flag = madagaskar_civil_war
		clr_global_flag = madagaskar_war_of_the_oppressed
		MAD_end_civil_war_ideas = yes
		MDJ = {
			country_event = {
				id = MADAGASKAR_CIVIL_WAR.4
				days = 7
			}
		}
		MDJ = {
			MAD_proxy_war_reward = yes
			MAD_inherit_madagaskar_admirals = yes
		}
		USA = { country_event = { id = USA.madagascar.3 } }
	}

	# MDM VICTORY
	else_if = {
		limit = {
			country_exists = MDM
			NOT = { FROM = { original_tag = MDM } }
			OR = {
				NOT = { country_exists = MAD }
				FROM = { original_tag = MAD }
			}
			OR = {
				NOT = { country_exists = MDG }
				FROM = { original_tag = MDG }
			}
			OR = {
				NOT = { country_exists = MDJ }
				FROM = { original_tag = MDJ }
			}
		}
		clr_global_flag = madagaskar_civil_war
		clr_global_flag = madagaskar_war_of_the_oppressed
		MAD_end_civil_war_ideas = yes
		MDM = {
			MAD_proxy_war_reward = yes
			MAD_inherit_madagaskar_admirals = yes
		}
		USA = { country_event = { id = USA.madagascar.6 } }
	}

	# All four gone (an outside power took the last of them)
	else_if = {
		limit = {
			OR = {
				NOT = { country_exists = MAD }
				FROM = { original_tag = MAD }
			}
			OR = {
				NOT = { country_exists = MDG }
				FROM = { original_tag = MDG }
			}
			OR = {
				NOT = { country_exists = MDJ }
				FROM = { original_tag = MDJ }
			}
			OR = {
				NOT = { country_exists = MDM }
				FROM = { original_tag = MDM }
			}
		}
		clr_global_flag = madagaskar_civil_war
		clr_global_flag = madagaskar_war_of_the_oppressed
		MAD_end_civil_war_ideas = yes
	}
}
