GER_calculate_total_seats = {
	#reset seat values
	set_variable = {
		GER_reichstag_seats = 0
	} 
	set_variable = {
		GER_nsdap_seats = 0
	} 
	set_variable = {
		GER_vup_seats = 0
	} 
	set_variable = {
		GER_rsp_seats = 0
	} 
	set_variable = {
		GER_vaaf_seats = 0
	} 
	set_variable = {
		GER_dg_seats = 0
	} 
	set_variable = {
		GER_dap_seats = 0
	} 
	set_variable = {
		GER_ler_seats = 0
	} 
	hidden_effect = {
		GER_reichstag_calculate_state_seats = yes
	}
	every_owned_state = {
		limit = {
			has_state_flag = GER_reichstag_state
		}
		add_to_variable = {
			GER.GER_nsdap_seats = GER_nsdap_seats_state
		}
		add_to_variable = {
			GER.GER_vup_seats = GER_vup_seats_state
		}
		add_to_variable = {
			GER.GER_rsp_seats = GER_rsp_seats_state
		}
		add_to_variable = {
			GER.GER_vaaf_seats = GER_vaaf_seats_state
		}
		add_to_variable = {
			GER.GER_dg_seats = GER_dg_seats_state
		}
		add_to_variable = {
			GER.GER_dap_seats = GER_dap_seats_state
		}
		add_to_variable = {
			GER.GER_ler_seats = GER_ler_seats_state
		}
	}
	
	add_to_variable = {
		GER_reichstag_seats = GER_nsdap_seats
	}
	add_to_variable = {
		GER_reichstag_seats = GER_vup_seats
	}
	add_to_variable = {
		GER_reichstag_seats = GER_rsp_seats
	}
	add_to_variable = {
		GER_reichstag_seats = GER_vaaf_seats
	}
	add_to_variable = {
		GER_reichstag_seats = GER_dg_seats
	}
	add_to_variable = {
		GER_reichstag_seats = GER_dap_seats
	}
	add_to_variable = {
		GER_reichstag_seats = GER_ler_seats
	}

	set_variable = {
		GER_reichstag_seats_half = GER_reichstag_seats
	}

	divide_variable = {
		GER_reichstag_seats_half = 2 
	}

	set_variable = {
		GER_reichstag_seats_majority = GER_reichstag_seats_half
	}

	round_variable = GER_reichstag_seats_majority

	if = {
		limit = {
			check_variable = {
				var = GER_reichstag_seats_majority
				value = GER_reichstag_seats_half
				compare = less_than_or_equals
			}
		}
		add_to_variable = {
			GER_reichstag_seats_majority = 1
		}
	}
}

GER_determine_election_winner_1 = {
	#determines election winner for first election
	hidden_effect = {
		GER_calculate_total_seats = yes
	}
	if = {
		limit = {
			GER_reichstag_nsdap_most_popular_party = yes
		}
		#event
		hidden_effect = {
			country_event = {
				id = GER_gfour_intro.11
				days = 1
			}
		}
	}
	else_if = {
		limit = { 
			GER_reichstag_vup_most_popular_party = yes
		}
		#event
		hidden_effect = {
			country_event = {
				id = GER_gfour_intro.12
				days = 1
			}
		}
	}
	else_if = {
		limit = { 
			GER_reichstag_rsp_most_popular_party = yes
		}
		#event
		hidden_effect = {
			country_event = {
				id = GER_gfour_intro.13
				days = 1
			}
		}
	}
	else_if = {
		limit = { 
			GER_reichstag_vaaf_most_popular_party = yes
		}
		#event
		hidden_effect = {
			country_event = {
				id = GER_gfour_intro.14
				days = 1
			}
		}
	}
}

GER_determine_election_winner_2 = {
	#determines election winner for second election
	hidden_effect = {
		GER_calculate_total_seats = yes
	}
	if = {
		limit = {
			GER_reichstag_nsdap_most_popular_party = yes
		}
		#nsdap wins
		hidden_effect = {
			country_event = {
				id = GER_gfour_election_two.2
				days = 1
			}
		}
	}
	else_if = {
		limit = {
			GER_reichstag_vup_most_popular_party = yes
		}
		#vup wins
		hidden_effect = {
			country_event = {
				id = GER_gfour_election_two.3
				days = 1
			}
		}
	}
	else_if = {
		limit = {
			GER_reichstag_rsp_most_popular_party = yes
		}
		#rsp wins
		hidden_effect = {
			country_event = {
				id = GER_gfour_election_two.4
				days = 1
			}
		}
	}
	else_if = {
		limit = {
			GER_reichstag_vaaf_most_popular_party = yes
			check_variable = {
				var = GER_g4_vaaf_rad_power
				value = 4
				compare = greater_than_or_equals
			}
		}
		#vaaf radical wins
		hidden_effect = {
			country_event = {
				id = GER_gfour_election_two.6
				days = 1
			}
		}
	}
	else_if = {
		limit = {
			GER_reichstag_vaaf_most_popular_party = yes
		}
		#vaaf moderate wins
		hidden_effect = {
			country_event = {
				id = GER_gfour_election_two.5
				days = 1
			}
		}
	}
	else_if = {
		limit = {
			GER_reichstag_dg_most_popular_party = yes
		}
		#die grunen wins
		hidden_effect = {
			country_event = {
				id = GER_gfour_election_two.7
				days = 1
			}
		}
	}
}
