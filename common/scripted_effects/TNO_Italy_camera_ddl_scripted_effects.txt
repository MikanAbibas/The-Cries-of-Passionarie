##############################################
#  #  #     Camera State Code Notes    #  #  #
##############################################
#  - ITA_camera_current_deputies: array holding the num deputies by current (0 is Berlinguer's Clique)
#
#  - ITA_coalition_overton: value 0.01 - 14.99 which defines the coalition's position in the overton window
#  - ITA_coalition_overton_int: integer version of the above, rounded down
#
#  - ITA_policy_overton: value 0.01 - 14.99 which defines the player's policy position in the overton window
#  - ITA_policy_overton_base: base policy overton, set when a new post-election coalition is formed
#  - ITA_policy_overton_modifier: debug modifier which adjusts the above
#  - ITA_policy_overton_preview: value 0.01 - 14.99 which shows the policy position if the current DDL were to be passed
#
#  - ITA_camera_current_coalition: array holding coalition status by current. 1 is not in coalition, 2 is in coalition
#  - ITA_total_coalition_members: number of deputies in the coalition
#  - ITA_camera_pm_current: which current the pm belongs to, ( 0 = left, 1 = center, 2 = right, 3 = Umberto )
#  - ITA_camera_pm_overton: array indexed by pm numbers, stores what overton value each PM has
#
#############################################
#  #  #   Government Formation Notes  #  #  #
#############################################
#
#  - ITA_gf_days_left: number of days left before bad thing happens
#  - ITA_gf_days_total: starting countdown value for government formation
#  - ITA_gf_days_left_pct: value 0-100 indicating progress of "days left" counter for government formation
#
#  - ITA_gf_accept_window: range around the pivot of currents who can be auto-added to your coalition
#  - ITA_gf_negotiate_window: range around the pivot of currents who can be negotiated into your coalition
#  - ITA_gf_accomidate_window: range around the policy overton beyond which currents will leave
#
#  - ITA_camera_pm_terms: an array, index x stores how many times the PM with the number x has been inaguerated
#
#############################################
#  #  #  Disengo Di Legge Code Notes  #  #  #
#############################################
# Current state:
#  - ITA_ddl_active_ddl: id of the currently active legge, or 0 if none
#   - Associated scripted trigger: ITA_ddl_active = yes
#   - Also note ITA_can_introduce_ddl = yes, which checks if there is no active DDL and not in GovForm mode
#  - ITA_ddl_is_repealing: 1 if repealing a DDL, 0 otherwise
#   - Associated scripted trigger: ITA_ddl_repealing = yes
#
#  - ITA_ddl_amendments: up to 4 amendments currently applied to the DDL, 0 indicates none
#  - ITA_ddl_possible_amendments: as many possible amendments as you want
#
#  - ITA_ddl_support_current_deputies: array holding num deputies which support current DDL, by current. DO NOT MODIFY directly, this is generated by effect, instead modify ITA_ddl_support_bonus_deputies
#  - ITA_ddl_support_current_pct: array holding % of deputies which support current DDL, by current. DO NOT MODIFY directly, see above
#  - ITA_ddl_support_bonus_support: array, extra % support for current DDL to add to the above, by current
#  - ITA_ddl_support_bonus_deputies: array, extra raw deputies who support current DDL to add to the above, by current
#  - ITA_ddl_total_support: total number of deputies supporting the current leg
#
#  - ITA_ddl_has_convinced: array, whether the support decision has been taken for a given current
#
#  - ITA_ddl_days_left: number of days left before DDL vote
#  - ITA_ddl_days_total: starting countdown value for DDL vote
#  - ITA_ddl_days_left_pct: value 0-100 indicating progress of "days left" counter for a DDL vote
#
#  - ITA_ddl_focus: focus value, 0-100
#
# DDL values:
#  - ITA_ddl_base_overton: array indexed by DDL id, holds the overton value of every DDL in the game
#  - ITA_ddl_base_support: array, holds the support % of every DDL in the game, at index (ddl_id * 15 + current_id)
#  - ITA_ddl_legge_effects: array, holds the icons for every DDL in the game, 7 per DDL, starting at index (ddl_id * 7)
#  - ITA_ddl_available_to: array indexed by DDL id, holds which current can select this DDL, default is -1 which means no one can select it
#                         ( this changes then a focus is completed that makes this DDL available )
#
#  - ITA_ddl_final_overton: array indexed by DDL id, holds the final overton value (including amendments) of every passed DDL (0 if not yet passed)
#  - ITA_ddl_final_focus: array indexed by DDL id, holds the final focus value (including amendments) of every passed DDL (0 if not yet passed)
#
# Amendment Values:
#  The amendment ids for a ddl with id ddl_id lie in the range [ddl_id * 10, ddl_id * 10 + 9]
#  For example, DDL 5 has amendments 50, 51, 52, 53, ..., 59. Not all of these exist, check ITA_ddl_amend_available to see if they do
#  - ITA_ddl_amend_available: array indexed by amend ID. 0 if no amend with this ID exists, 1 if such an amend does exist, 2 if this amend was passed with the corresponding DDL in the past
#  - ITA_ddl_amend_overton_mod: array indexed by amend ID, modifier added to the base overton of the DDL when tpolicy_lean = he amend is applied
#  - ITA_ddl_amend_focus_mod: array indexed by amend ID, modifier added to the base focus of the DDL when the amend is applied
#  - ITA_ddl_amend_support_mod: array, holds the support % modifier of every amend, at index (amend_id * 15 + current_id)
#
# DDL lists (history, avaiable, able to be repealed)
#  - ITA_ddl_can_repeal: array, holds DDL ids which can be repealed (from promises)
#
#  - ITA_ddl_history_limit: number of entries at the end of the history window to ignore for the purposes of policy lean
#  - ITA_ddl_history: array of past DDLs
#  - ITA_ddl_history_status: associated info for history status, 1 if the bill passed, 2 if vote failed, 3 if bill was repealed, 4 if repeal vote failed
#
#  - ITA_ddl_available: array used by debug menu, autopopulated by all DDLs which haven't been passed yet
#
# PM Ability Stuff
#  - ITA_ddl_pm_has_ability: 1 if the PM has a special ability where a bar must be shown, 0 otherwise
#
# Some utility arrays, unused by the DDL mechanic for now:
#  - ITA_ddl_passed_bills: array of all DLLs which have been passed, including those which were later repealed
#  - ITA_ddl_failed_bills: array of all DLLs which have failed to pass
#  - ITA_ddl_repealed_bills: array of all DLLs which have been repealed
#  - ITA_ddl_repeal_failed_bills: array of all DLLs which have had a failed repeal vote
#
##############################################
#  #  #      Camera GUI Code Notes     #  #  #
##############################################
# Display state:
#  - ITA_camera_coalition_view: 2 if the arch shows the coalition view
#  - ITA_camera_ddl_view: 2 if the arch shows the ddl view
#  - ITA_camera_submenu: 1 if showing the DDL view, 2 GovForm
#   - Associated scripted triggers: ITA_govform_active = yes
#  - ITA_camera_submenu_arr: single element array, holds same value as above
#
# Seat variables:
#  - ITA_camera_seat_x, ITA_camera_seat_y: determines x, y pos of each seat
#  - ITA_camera_seat_x_offset, ITA_camera_seat_y_offset: drawn seat offset, used to indicate current coalition
#  - ITA_camera_seat_current: current id for each seat
#
# Current overview:
#  - ITA_camera_current_color: array holding the frame in GFX_camera_seat_icon to use for each current, by current
#  - ITA_camera_left_currents, ITA_camera_center_currents, ITA_camera_right_currents: used to populate infoboxes
#  - ITA_camera_current_group: array showing which group each current is in. 1 if Left, 2 if Center, 3 if Right
#
# Coalition bar:
#  - ITA_coalition_bar_offset: offset of the red coalition bar on the overton display, based on ITA_coalition_overton
#  - ITA_policy_bar_offset: offset of the blue policy agreement bar, based on ITA_policy_overton

ITA_gf_trigger_election_govform = {
	set_temp_variable = { ITA_total_seats_left = 0 }
	add_to_temp_variable = { ITA_total_seats_left = ITA_camera_current_deputies^0 }
	add_to_temp_variable = { ITA_total_seats_left = ITA_camera_current_deputies^1 }
	add_to_temp_variable = { ITA_total_seats_left = ITA_camera_current_deputies^2 }
	add_to_temp_variable = { ITA_total_seats_left = ITA_camera_current_deputies^3 }
	add_to_temp_variable = { ITA_total_seats_left = ITA_camera_current_deputies^4 }

	set_temp_variable = { ITA_total_seats_center = 0 }
	add_to_temp_variable = { ITA_total_seats_center = ITA_camera_current_deputies^5 }
	add_to_temp_variable = { ITA_total_seats_center = ITA_camera_current_deputies^6 }
	add_to_temp_variable = { ITA_total_seats_center = ITA_camera_current_deputies^7 }
	add_to_temp_variable = { ITA_total_seats_center = ITA_camera_current_deputies^8 }
	add_to_temp_variable = { ITA_total_seats_center = ITA_camera_current_deputies^9 }

	set_temp_variable = { ITA_total_seats_right = 0 }
	add_to_temp_variable = { ITA_total_seats_right = ITA_camera_current_deputies^10 }
	add_to_temp_variable = { ITA_total_seats_right = ITA_camera_current_deputies^11 }
	add_to_temp_variable = { ITA_total_seats_right = ITA_camera_current_deputies^12 }
	add_to_temp_variable = { ITA_total_seats_right = ITA_camera_current_deputies^13 }
	add_to_temp_variable = { ITA_total_seats_right = ITA_camera_current_deputies^14 }

	if = {
		limit = { 
			NOT = { check_variable = { ITA_total_seats_left < ITA_total_seats_right } }
			check_variable = { ITA_total_seats_left > ITA_total_seats_center }
		}
		set_variable = { ITA_winning_wing = 0 }

		if = {
			limit = { 
				check_variable = { ITA_total_seats_right > ITA_total_seats_center }
			}
			set_variable = { ITA_second_wing = 2 }
		}
		else = {
			set_variable = { ITA_second_wing = 1 }
		}
	}
	else_if = {
		limit = { 
			check_variable = { ITA_total_seats_right > ITA_total_seats_left }
			check_variable = { ITA_total_seats_right > ITA_total_seats_center }
		}
		set_variable = { ITA_winning_wing = 2 }

		if = {
			limit = { 
				check_variable = { ITA_total_seats_left > ITA_total_seats_center }
			}
			set_variable = { ITA_second_wing = 0 }
		}
		else = {
			set_variable = { ITA_second_wing = 1 }
		}
	}
	else = {
		set_variable = { ITA_winning_wing = 1 }

		if = {
			limit = { 
				check_variable = { ITA_total_seats_right > ITA_total_seats_left }
			}
			set_variable = { ITA_second_wing = 2 }
		}
		else = {
			set_variable = { ITA_second_wing = 0 }
		}
	}

	set_country_flag = ITA_gf_post_election
	if = {
		limit = {
			ITA_ddl_active = no
		}

		set_variable = { ITA_camera_submenu = 2 }
		clear_array = ITA_camera_submenu_arr 
		add_to_array = { ITA_camera_submenu_arr = 2 }

		ITA_begin_gf = yes
	}
	else = {
		set_country_flag = ITA_gf_queued
	}
}

ITA_gf_collapse = {
	if = {
		limit = { check_variable = { ITA_policy_overton < 5 }}
		set_variable = { ITA_winning_wing = 0 }
	}
	else_if = {
		limit = { check_variable = { ITA_policy_overton < 10 }}
		set_variable = { ITA_winning_wing = 1 }
	}
	else = {
		set_variable = { ITA_winning_wing = 2 }
	} 
	set_variable = { ITA_camera_submenu = 2 }
	clear_array = ITA_camera_submenu_arr 
	add_to_array = { ITA_camera_submenu_arr = 2 }

	set_variable = { ITA_gf_failure_count = 0 }

	ITA_begin_gf = yes
}

ITA_gf_timeout = {
	# The King's unity cabinet, before any election has been held. There is one
	# candidate on the list and no second current to hand it to, and the ladder
	# below is all post-election machinery - snap elections, then Piano Solo,
	# then Borghese. Italy's first election is ITA_med.7004, fired when the
	# natunity tree ends, so none of that can be reached from here. Put the seven
	# days back on the clock and let the player keep trying.
	if = {
		limit = { has_country_flag = ITA_camera_natunity_mode }
		ITA_begin_gf = yes
	}

	#the largest current doesn't form a coalition, the second largest gets to try
	#there are no consequences yet
	else_if = {
		limit = {
			has_country_flag = ITA_gf_post_election
			NOT = {
				has_country_flag = ITA_gf_switched_currents
			}
		}

		# ITA_med.7000, not ita.7000 - same namespace-case bug as the two coups below.
		country_event = { id = ITA_med.7000 }

		set_variable = { ITA_winning_wing = ITA_second_wing }
		ITA_begin_gf = yes
		
		set_country_flag = ITA_gf_switched_currents
	}

	#You get to have 2 snap elections, if you try to have a third you get PS
	else_if = {
		limit = {
			NOT = {
				has_country_flag = ITA_piano_solo_already_happened
			}
		}
		#the funnies start happening

		if = {
			limit = { check_variable = { ITA_gf_failure_count = 0 } }

			country_event = { id = ITA_med.7001 }

			#1st snap election, radicals get a boost
			set_temp_variable = { ITA_snap_election_type = 1 }
			ITA_POP_Dem_election = yes
		}
		else_if = {
			limit = { check_variable = { ITA_gf_failure_count = 1 } }

			country_event = { id = ITA_med.7005 }

			#2nd snap election, the king warns the parties to stop bickering, the center gets a boost
			set_temp_variable = { ITA_snap_election_type = 2 }
			ITA_POP_Dem_election = yes
		}
		else_if = {
			ITA_camera_piano_solo = yes
		}


		add_to_variable = { ITA_gf_failure_count = 1 }
	}

	#You get to have 2 snap elections, if you try to have a third you get Borghese
	#(after PS has already happened)
	else = {

		if = {
			limit = { check_variable = { ITA_gf_failure_count = 0 } }

			country_event = { id = ITA_med.7006 }

			#1st snap election, radicals get a boost
			set_temp_variable = { ITA_snap_election_type = 1 }
			ITA_POP_Dem_election = yes
		}
		else_if = {
			limit = { check_variable = { ITA_gf_failure_count = 1 } }

			country_event = { id = ITA_med.7006 }

			#2nd snap election, radicals also get a boost since PS has already happened
			set_temp_variable = { ITA_snap_election_type = 1 }
			ITA_POP_Dem_election = yes
		}
		else_if = {
			# Player only, for now. Borghese has no content behind it - ITA_med.7003
			# plays a sound and stops, ITA_BORGHESE_COUP is never cleared, and
			# ITA_govform_active excludes that flag, so the Camera never runs
			# again. A human can choose to end there; AI Italy would simply
			# freeze for the rest of the game.
			limit = { is_ai = no }
			ITA_camera_borghese = yes
		}
		else = {
			# So AI Italy keeps going instead: another snap election, radicals
			# boosted, same as the two before it. Drop this branch once Borghese
			# has an ending of its own.
			country_event = { id = ITA_med.7006 }
			set_temp_variable = { ITA_snap_election_type = 1 }
			ITA_POP_Dem_election = yes
		}

		add_to_variable = { ITA_gf_failure_count = 1 }
	}
}

ITA_gf_update_negotiation_status = {
	clear_array = ITA_gf_current_negotiation

	set_variable = { ITA_coalition_overton = 0 }
	set_temp_variable = { coalition_members_num = 0}
	for_loop_effect = {
		start = 0
		end = 15
		value = i

		if = {
			limit = {
				check_variable = { ITA_camera_current_coalition^i = 2 }
			}
			add_to_variable = { ITA_coalition_overton = i }
			add_to_variable = { ITA_coalition_overton = 0.5 }
			add_to_temp_variable = { coalition_members_num = 1 }
		}
	}

	divide_variable = { ITA_coalition_overton = coalition_members_num }
	if = {
		limit = {
			has_country_flag = ITA_gf_post_election
		}
		set_variable = { ITA_policy_overton = ITA_camera_pm_overton^ITA_gf_selected_pm }
		add_to_variable = { ITA_policy_overton = 0.5 }
	}

	for_loop_effect = {
		start = 0
		end = 15
		value = i

		set_temp_variable = { difference = i }
		add_to_temp_variable = { difference = 0.5 }
		subtract_from_temp_variable = { difference = ITA_policy_overton }
		if = {
			limit = { check_variable = { difference < 0 }}
			multiply_temp_variable = { difference = -1 }
		}

		set_temp_variable = { pm_difference = i }
		subtract_from_temp_variable = { pm_difference = ITA_camera_pm_overton^ITA_gf_selected_pm }
		if = {
			limit = { check_variable = { pm_difference < 0 }}
			multiply_temp_variable = { pm_difference = -1 }
		}

		if = {
			limit = { AND = {
				check_variable = { pm_difference < 1.01 }
			}}
			add_to_array = { ITA_gf_current_negotiation = 5 }
		}
		else_if = {
			limit = {
				check_variable = { ITA_camera_current_coalition^i = 2 }
			}
			add_to_array = { ITA_gf_current_negotiation = 4 }
		}
		else_if = {
			limit = {
				check_variable = { var = difference value = ITA_gf_accept_window compare = less_than_or_equals }
			}
			add_to_array = { ITA_gf_current_negotiation = 3 }
		}
		else_if = {
			limit = {
				check_variable = { var = difference value = ITA_gf_negotiate_window compare = less_than_or_equals }
			}
			add_to_array = { ITA_gf_current_negotiation = 2 }
		}
		else = {
			add_to_array = { ITA_gf_current_negotiation = 1 }
		}
	}
}

ITA_recompute_ddl_available = {
	clear_array = ITA_ddl_available
	#Loop through every possible DDL
	for_loop_effect = {
		start = 1
		end = ITA_ddl_base_overton^num
		value = i

		if = {
			limit = {
				#If it is either:
				#Marked to be available to the current PM's curent
				#Is in Camera debug mode (debug mode shows all DDLs)
				OR = {
					has_country_flag = ITA_Camera_Debug_Tools_Enabled
					check_variable = { ITA_ddl_available_to^i = ITA_camera_pm_current }
				}
			}

			#Check if the DDLs has been passed
			set_temp_variable = { found = 0 }
			for_loop_effect = {
				start = 0
				end = ITA_ddl_history^num
				value = j
				break = brk

				if = {
					limit = {
						check_variable = { ITA_ddl_history^j = i }
						check_variable = { ITA_ddl_history_status^j = 1 } 
					}
					set_temp_variable = { found = 1 }
					set_temp_variable = { brk = 1 }
				}
			}

			if = {
				#if it wasn't found in history then add it
				limit = { check_variable = { found = 0 }}
				add_to_array = { ITA_ddl_available = i }
			}
		}
	}
}

# Call this wehenever you change ITA_ddl_active_ddl
# Sets up a DDL (or tears down a DDL when one is removed), including setting the amendment lists and calculating initial support
ITA_setup_new_ddl = {
	if = {
		limit = { ITA_ddl_active = no }
		clear_array = ITA_ddl_amendments
		resize_array = { array = ITA_ddl_amendments value = 0 size = 4 }

		clear_array = ITA_ddl_possible_amendments

		set_variable = { ITA_camera_ddl_view = 1 }
		set_variable = { ITA_camera_coalition_view = 2 }
		set_variable = { ITA_ddl_days_left_pct = 100 }

		if = {
			limit = { has_country_flag = ITA_gf_queued }

			ITA_gf_collapse = yes
		}
	}
	else = {
		# PASSING A NORMAL BILL
		if = {
			limit = { ITA_ddl_repealing = no }

			clear_array = ITA_ddl_amendments
			resize_array = { array = ITA_ddl_amendments value = 0 size = 4 }

			clear_array = ITA_ddl_possible_amendments

			set_temp_variable = { base = ITA_ddl_active_ddl }
			multiply_temp_variable = { base = 10 }
			for_loop_effect = {
				start = 0
				end = 10
				value = i

				set_temp_variable = { amendment_id = base }
				add_to_temp_variable = { amendment_id = i }

				if = {
					limit = {
						check_variable = { ITA_ddl_amend_available^amendment_id > 0 }
					}
					add_to_array = { ITA_ddl_possible_amendments = amendment_id }
				}
			}
		}
		# TRYING TO REPEAL A BILL
		else = {
			clear_array = ITA_ddl_amendments
			clear_array = ITA_ddl_possible_amendments

			set_temp_variable = { base = ITA_ddl_active_ddl }
			multiply_temp_variable = { base = 10 }
			for_loop_effect = {
				start = 0
				end = 10
				value = i

				set_temp_variable = { amendment_id = base }
				add_to_temp_variable = { amendment_id = i }

				if = {
					limit = {
						check_variable = { ITA_ddl_amend_available^amendment_id = 2 }
					}
					add_to_array = { ITA_ddl_amendments = amendment_id }
				}
			}

# Called when the timer on a DDL expires, triggering a successful or failed vote
ITA_ddl_hold_vote = {
	# NORMAL VOTE
	if = {
		limit = { ITA_ddl_repealing = no }
		# SUCCESS
		if = {
			limit = {
				check_variable = { ITA_ddl_total_support > 199.5 }
			}
			log = "[GetDateText]: [Root.GetName]: Current DDL passed with [?ITA_ddl_total_support] votes" 
			
			set_temp_variable = { base = ITA_ddl_active_ddl }
			multiply_temp_variable = { base = 10 }
			for_loop_effect = {
				start = 0
				end = 10
				value = i

				set_temp_variable = { amendment_id = base }
				add_to_temp_variable = { amendment_id = i }

				if = {
					limit = {
						is_in_array = { array = ITA_ddl_amendments value = amendment_id }
					}
					set_variable = { ITA_ddl_amend_available^amendment_id = 2 }
				}
				else = {
					set_variable = { ITA_ddl_amend_available^amendment_id = 1 }
				}
			}
			add_to_array = { array = ITA_ddl_history value = ITA_ddl_active_ddl index = 0 }
			add_to_array = { array = ITA_ddl_history_status value = 1 index = 0 }
			add_to_array = { ITA_ddl_passed_bills = ITA_ddl_active_ddl }

			set_variable = { ITA_ddl_final_overton^ITA_ddl_active_ddl = ITA_ddl_active_total_overton }
			set_variable = { ITA_ddl_final_focus^ITA_ddl_active_ddl = ITA_ddl_focus }

			set_temp_variable = { legge_id = ITA_ddl_active_ddl }
			set_temp_variable = { effectiveness = ITA_ddl_focus }
			
			log = "legge_id: [?legge_id]"

			ITA_ddl_success_effect = yes

			# Update Berlinguer Optimism
			if = {
				limit = { check_variable = { ITA_gf_selected_pm = 0 } }
				add_to_variable = { ITA_ddl_optimism = ITA_ddl_next_optimism_incr }
				ITA_gf_update_optimism = yes
			}

			set_temp_variable = { alert_id = 16 }
			set_temp_variable = { alert_data = ITA_ddl_active_ddl }
			add_TNO_alert = yes
		}
		# FAILURE
		else = {
			log = "[GetDateText]: [Root.GetName]: Current DDL failed with [?ITA_ddl_total_support] votes" 
			
			add_to_array = { array = ITA_ddl_history value = ITA_ddl_active_ddl index = 0 }
			add_to_array = { array = ITA_ddl_history_status value = 2 index = 0 }
			add_to_array = { ITA_ddl_failed_bills = ITA_ddl_active_ddl }

			# Update Berlinguer Optimism
			if = {
				limit = { check_variable = { ITA_gf_selected_pm = 0 } }
				set_variable = { ITA_ddl_optimism = 0 }
				ITA_gf_update_optimism = yes
			}

			set_temp_variable = { alert_id = 17 }
			set_temp_variable = { alert_data = ITA_ddl_active_ddl }
			add_TNO_alert = yes
		}
	}
	# REPEAL VOTES
	else = {
		# SUCCESS
		if = {
			limit = {
				check_variable = { ITA_ddl_total_support > 199.5 }
			}
			add_to_array = { array = ITA_ddl_history value = ITA_ddl_active_ddl index = 0 }
			add_to_array = { array = ITA_ddl_history_status value = 3 index = 0 }
			add_to_array = { ITA_ddl_repealed_bills = ITA_ddl_active_ddl }

			log = "[GetDateText]: [Root.GetName]: Current DDL repealed with [?ITA_ddl_total_support] votes" 
		}
		# FAILURE
		else = {
			add_to_array = { array = ITA_ddl_history value = ITA_ddl_active_ddl index = 0 }
			add_to_array = { array = ITA_ddl_history_status value = 4 index = 0 }
			add_to_array = { ITA_ddl_repeal_failed_bills = ITA_ddl_active_ddl }

			log = "[GetDateText]: [Root.GetName]: Current DDL repeal failed with [?ITA_ddl_total_support] votes" 
		}

		remove_from_array = {  ITA_ddl_can_repeal = ITA_ddl_active_ddl }
	}

	ITA_recompute_ddl_available = yes
	ITA_camera_update_policy_lean = yes

	set_variable = { ITA_ddl_active_ddl = 0 }
	set_variable = { ITA_ddl_is_repealing = 0 }
	ITA_setup_new_ddl = yes

	ITA_camera_update_coalition_status = yes

	set_temp_variable = { alert_id = 15 }
	remove_TNO_alert  = yes
}


# Computes the player's policy lean based on their DDL history
ITA_camera_update_policy_lean = {
	# Add in a weight of 3x your coalition overton for base policy overton
	set_variable = { ITA_policy_overton_num = 3 }
	set_variable = { ITA_policy_overton = ITA_policy_overton_base }
	add_to_variable = { ITA_policy_overton = 0.5 }
	multiply_variable = { ITA_policy_overton = 3 }

	set_temp_variable = { max_to_consider = ITA_ddl_history^num }
	subtract_from_temp_variable = { max_to_consider = ITA_ddl_history_limit}

	for_each_loop = {
		array = ITA_ddl_history
		value = legge_id
		index = j

		if = {
			limit = {
				check_variable = { j < max_to_consider }
				OR = {
					check_variable = { ITA_ddl_history_status^j = 1 }
					check_variable = { ITA_ddl_history_status^j = 3 }
				}
			}
			set_temp_variable = { overton = ITA_ddl_final_overton^legge_id }

			if = {
				limit = { check_variable = { ITA_ddl_history_status^j = 3 } }
				multiply_temp_variable = { overton = -1 }
				add_to_temp_variable = { overton = 15 }
			}

			add_to_variable = { ITA_policy_overton = overton }
			add_to_variable = { ITA_policy_overton_num = 1 }
		}

	}

	divide_variable = { ITA_policy_overton = ITA_policy_overton_num }
	add_to_variable = { ITA_policy_overton = ITA_policy_overton_modifier }
	clamp_variable = { var = ITA_policy_overton min = 0 max = 15 }

	set_temp_variable = { num_dislike = 0 }
	for_loop_effect = {
		start = 0
		end = 15
		value = i

		if = {
			limit = { check_variable = { ITA_policy_overton > i } }

			set_temp_variable = { difference = ITA_policy_overton }
			subtract_from_temp_variable = { difference = 0.5 }
			subtract_from_temp_variable = { difference = i }
		}
		else = {
			set_temp_variable = { difference = i }
			add_to_temp_variable = { difference = 0.5 }
			subtract_from_temp_variable = { difference = ITA_policy_overton }
		}

		if = {
			limit = { 
				check_variable = { difference > ITA_gf_accomidate_window } 
				check_variable = { ITA_camera_current_coalition^i = 2 }
			}
			add_to_temp_variable = { num_dislike = 1 }
			set_variable = { ITA_camera_current_coalition^i = 1 }
		}
	}

	if = {
		limit = { 
			check_variable = { num_dislike > 0 }
		}
		ITA_camera_update_coalition_status = yes
	}
}

ITA_camera_update_ddl_stato_focus = {
	clear_array = ITA_ddl_focus_relevant_stato_groups

	set_temp_variable = { total_stato_factor = 0 }
	set_temp_variable = { total_stato_support = 0 }

	set_temp_variable = { base = ITA_ddl_active_ddl }
	multiply_temp_variable = { base = 6 }
	for_loop_effect = {
		start = 0
		end = 6
		value = i

		set_temp_variable = { stato_factor_idx = base }
		add_to_temp_variable = { stato_factor_idx = i }

		set_temp_variable = { factor = ITA_ddl_stato_factor^stato_factor_idx }
		multiply_temp_variable = { factor = ITA_Stato_Influence^i }

		if = {
			limit = {
				check_variable = { factor > 0 }
			}
			add_to_array = { ITA_ddl_focus_relevant_stato_groups = i }
		}

		set_temp_variable = { support = factor }
		if = { 
			limit = { check_variable = { ITA_camera_pm_overton^ITA_gf_selected_pm < 5} }
			multiply_temp_variable = { support = ITA_Left_Stato_Support^i }
		}
		else_if = { 
			limit = { check_variable = { ITA_camera_pm_overton^ITA_gf_selected_pm < 10} }
			multiply_temp_variable = { support = ITA_Center_Stato_Support^i }
		}
		else = { 
			multiply_temp_variable = { support = ITA_Right_Stato_Support^i }
		}
		multiply_temp_variable = { factor = 100 }

		add_to_temp_variable = { total_stato_factor = factor }
		add_to_temp_variable = { total_stato_support = support }

		log = "[GetDateText]: [Root.GetName]: [?i]: [?support] / [?factor]" 
	}
	log = "[GetDateText]: [Root.GetName]: [?total_stato_support] / [?total_stato_factor]" 
	divide_temp_variable = { total_stato_support = total_stato_factor }

	if = {
		limit = { 
			check_variable = { ITA_gf_selected_pm = 1 }
			check_variable = { total_stato_support < 0.4 }
		}
		set_temp_variable = { total_stato_support = 0.4 }
	}
	multiply_temp_variable = { total_stato_support = 50 }
	set_variable = { ITA_ddl_focus_stato = total_stato_support }
	set_variable = { ITA_ddl_focus_stato_display = ITA_ddl_focus_stato }
	divide_variable = { ITA_ddl_focus_stato_display = 100 }
}


# Updates the current focus value and the indicator for how much the player's policy lean will change if a DDL passes
ITA_camera_update_ddl_policy_shift = {
	set_variable = { ITA_policy_overton_preview = ITA_policy_overton }
	multiply_variable = { ITA_policy_overton_preview = ITA_policy_overton_num }

	### Compute Focus
	# Compute amendment portion
	set_variable = { ITA_ddl_active_total_overton = ITA_ddl_base_overton^ITA_ddl_active_ddl }
	set_variable = { ITA_ddl_focus_amendments = 0 }
	for_each_loop = {
		array = ITA_ddl_amendments
		value = amendment_num
		
		if = {
			limit = { check_variable = { amendment_num > 0 }  }
			add_to_variable = { ITA_ddl_active_total_overton = ITA_ddl_amend_overton_mod^amendment_num }
			add_to_variable = { ITA_ddl_focus_amendments = ITA_ddl_amend_focus_mod^amendment_num }
		}
	}

	if = {
		limit = { 
			check_variable = { ITA_gf_selected_pm = 1 }
		}
		multiply_variable = { ITA_ddl_focus_amendments = 1.25 }
	}

	clamp_variable = { var = ITA_ddl_focus_amendments min = -50 max = 0 }

	set_variable = { ITA_ddl_focus = 50 }
	add_to_variable = { ITA_ddl_focus = ITA_ddl_focus_stato }
	add_to_variable = { ITA_ddl_focus = ITA_ddl_focus_amendments }
	# Umberto Bonus
	if = {
		limit = { check_variable = { ITA_gf_selected_pm = 15 } }
		add_to_variable = { ITA_ddl_focus = 5 }
		clamp_variable = { var = ITA_ddl_focus min = 0 max = 100 }
	}

	divide_variable = { ITA_ddl_focus_amendments = 100 }

	### Compute Overton
	if = { 
		limit = {
			ITA_ddl_repealing = no 
		}
		add_to_variable = { ITA_policy_overton_preview = ITA_ddl_active_total_overton }
	}
	else = {
		set_temp_variable = { inverse = 15 }
		subtract_from_temp_variable = { inverse = ITA_ddl_active_total_overton }
		add_to_variable = { ITA_policy_overton_preview = inverse }
	}

	set_temp_variable = { divisor = ITA_policy_overton_num }
	add_to_temp_variable = { divisor = 1 }
	divide_variable = { ITA_policy_overton_preview = divisor }
}

# Updates the ITA_camera_current_coalition array and the ITA_total_coalition_members
ITA_camera_update_coalition_status = {

	set_temp_variable = { offset = ITA_ddl_active_ddl }
	multiply_temp_variable = { offset = 15 }

	clear_array = ITA_ddl_support_current_deputies
	clear_array = ITA_ddl_support_current_pct
	for_loop_effect = {
		start = 0
		end = 15
		value = current_idx

		set_temp_variable = { base_support_idx = offset }
		add_to_temp_variable = { base_support_idx = current_idx }

		set_temp_variable = { support_amnt = ITA_ddl_base_support^base_support_idx}

		# Saragat Support Penalty
		if = {
			limit = { check_variable = { ITA_gf_selected_pm = 4 } }
			multiply_temp_variable = { support_amnt = 0.95 }
		}
		# Berlinguer Penalty / Buff
		else_if = {
			limit = { check_variable = { ITA_gf_selected_pm = 0 } }
			set_temp_variable = { multiplier = 0.95 }
			add_to_temp_variable = { multiplier = ITA_ddl_base_support_modifier_optimism }
			multiply_temp_variable = { support_amnt = multiplier }
		}
		# Fanfani Bonus
		if = {
			limit = { check_variable = { ITA_gf_selected_pm = 7 } }
			multiply_temp_variable = { support_amnt = 133.33 }
			divide_temp_variable = { support_amnt = 100 }
		}
		# Umberto Bonus
		if = {
			limit = { check_variable = { ITA_gf_selected_pm = 15 } }
			multiply_temp_variable = { support_amnt = 1.05 }
		}

		# Add support from each amendment
		for_each_loop = {
			array = ITA_ddl_amendments
			value = amendment_num

			set_temp_variable = { amend_offset = amendment_num }
			multiply_temp_variable = { amend_offset = 15 }
			add_to_temp_variable = { amend_offset = current_idx }

			if = {
				limit = { check_variable = { amendment_num > 0 } }
				add_to_temp_variable = { support_amnt = ITA_ddl_amend_support_mod^amend_offset }
			}
		}

		# Invert support if repealing a bill
		if = {
			limit = { ITA_ddl_repealing = yes }

			subtract_from_temp_variable = { support_amnt = 1 }
			multiply_temp_variable = { support_amnt = -1 }

			# 5% penalty for repeals, sorry!
			subtract_from_temp_variable = { support_amnt = 0.05 }
		}

		# Add bonus support from misc sources
		add_to_temp_variable = { support_amnt = ITA_ddl_support_bonus_support^current_idx }

		clamp_temp_variable = { var = support_amnt min = 0 max = 1}
		add_to_array = { ITA_ddl_support_current_deputies = ITA_camera_current_deputies^current_idx }
		multiply_variable = { ITA_ddl_support_current_deputies^current_idx = support_amnt }
		round_variable = ITA_ddl_support_current_deputies^current_idx

		# Add bonus deputies (ie from debug menu) 
		add_to_variable = { ITA_ddl_support_current_deputies^current_idx = ITA_ddl_support_bonus_deputies^current_idx }
		clamp_variable = { var = ITA_ddl_support_current_deputies^current_idx min = 0 max = ITA_camera_current_deputies^current_idx  } 

		add_to_array = { ITA_ddl_support_current_pct = ITA_ddl_support_current_deputies^current_idx }
		divide_variable = { ITA_ddl_support_current_pct^current_idx = ITA_camera_current_deputies^current_idx }
	}

	set_variable = { ITA_coalition_overton_base = -1 }
	set_variable = { ITA_coalition_overton_end = -1 }

	set_variable = { ITA_total_coalition_members = 0 }
	set_variable = { ITA_ddl_total_support = 0 }

	#clear_array = ITA_camera_current_coalition
	for_loop_effect = {
		start = 0
		end = 15
		value = i

		add_to_variable = { ITA_ddl_total_support = ITA_ddl_support_current_deputies^i }
		if = {
			limit = { AND = {
				check_variable = { ITA_camera_current_coalition^i = 2 }
			}}
			add_to_variable = { ITA_total_coalition_members = ITA_camera_current_deputies^i }

			if = { 
				limit = { check_variable = { ITA_coalition_overton_base = -1 }}
				set_variable = { ITA_coalition_overton_base = i }
			}
			set_variable = { ITA_coalition_overton_end = i }
		}
	}

	if = {
		limit = { 
			check_variable = { ITA_total_coalition_members < 201 }
			ITA_govform_active = no
		}

		if = {
			limit = {
				ITA_ddl_active = no
				check_variable = { ITA_camera_submenu = 1 }
			}

			ITA_gf_collapse = yes
		}
		else = {
			set_country_flag = ITA_gf_queued
		}
	}
}

# Redraw Camera semicircle display to show coalitions
ITA_camera_draw_coalition_arch = {
	clear_array = ITA_camera_seat_current
	clear_array = ITA_camera_seat_x_real
	clear_array = ITA_camera_seat_y_real

	set_temp_variable = { has_before = 0 }
	if = {
		limit = { check_variable = { ITA_camera_current_coalition^0 = 1 }}
		set_temp_variable = { has_before = 1 }
	}
	set_temp_variable = { has_after = 0 }
	if = {
		limit = { check_variable = { ITA_camera_current_coalition^14 = 1 }}
		set_temp_variable = { has_after = 1 }
	}

	set_temp_variable = { total = 0 }
	set_temp_variable = { before = 1 }
	for_loop_effect = {
		start = 0
		end = 15
		value = i

		set_temp_variable = { ir = i }
		# We render the PSI as a united block even though the radicals
		# are situated in the middle in terms of the overton window
		# HOWEVER we don't swap around the order if one party is in the coalition and the other isn't
		# This looks a lot grosser than it is, I am sorrt 
		if = {
			limit = { AND = {
				check_variable = { ir = 2 }
				check_variable = { ITA_camera_current_coalition^2 = ITA_camera_current_coalition^3 }
			}}
			set_temp_variable = { ir = 3 }
		}
		else_if = {
			limit = { AND = {
				check_variable = { ir = 3 }
				check_variable = { ITA_camera_current_coalition^2 = ITA_camera_current_coalition^3 }
			}}
			set_temp_variable = { ir = 2 }
		}

		set_temp_variable = { old_total = total }
		add_to_temp_variable = { total = ITA_camera_current_deputies^ir }
		resize_array = {
			array = ITA_camera_seat_current
			value = ir
			size = total
		}

		set_temp_variable = { x_offset = 15 }
		if = {
			limit = { check_variable = { ITA_coalition_overton < 5 }}
			set_temp_variable = { y_offset = -10 }
		}
		else = {
			set_temp_variable = { y_offset = 0 }
		}
		if = {
			limit = { check_variable = { ITA_camera_current_coalition^ir = 2 }}

			if = {
				limit = { check_variable = { has_before = 0}}
				set_temp_variable = { y_offset = 0 }
				set_temp_variable = { x_offset = -15 }
			}
			else_if = {
				limit = { check_variable = { has_after = 0}}
				set_temp_variable = { y_offset = 0 }
				set_temp_variable = { x_offset = 15 }
			}
			else = {
				set_temp_variable = { y_offset = -5 }
				set_temp_variable = { x_offset = 0 }
			}
			set_temp_variable = { before = 0 }
		}
		else_if = {
			limit = { check_variable = { before = 1 }}
			set_temp_variable = { x_offset = -15 }
			if = {
				limit = { check_variable = { ITA_coalition_overton > 10 }}
				set_temp_variable = { y_offset = -10 }
			}
			else = {
				set_temp_variable = { y_offset = 0 }
			}
		}


		for_loop_effect = {
			start = old_total
			end = total
			value = k
		
			add_to_array = { ITA_camera_seat_x_real = ITA_camera_seat_x^k }
			add_to_array = { ITA_camera_seat_y_real = ITA_camera_seat_y^k }
			add_to_variable = { ITA_camera_seat_x_real^k = x_offset } 
			add_to_variable = { ITA_camera_seat_y_real^k = y_offset } 
		}
	}
}

# Effects used by debug buttons to add or remove seats from a current
# You should set the `to_add` or `to_rem` temp variables, respectively
# And the `current_number` temp variable to specify a current to add/remove from
ITA_debug_add_seats = {
	set_temp_variable = { max_can_add = 400 }
	subtract_from_temp_variable = { max_can_add = ITA_camera_current_deputies^current_number }
	if = {
		limit = {
			check_variable = { max_can_add < to_add }
		}
		set_temp_variable = { to_add = max_can_add }
	}
	add_to_variable = { ITA_camera_current_deputies^current_number = to_add }

	set_temp_variable = { idx = current_number }
	add_to_variable = { idx = 1 }
	while_loop_effect = {
		limit = {
			check_variable = { to_add > 0 }
		}
		if = {
			limit = { AND = {
				check_variable = { ITA_camera_current_deputies^idx > 0 } 
				NOT = { check_variable = { idx = current_number } }
			}}
			subtract_from_variable = { ITA_camera_current_deputies^idx = 1 }
			clamp_variable = { var=ITA_ddl_support_current_deputies^idx min = 0 max = ITA_camera_current_deputies^idx  } 
			subtract_from_temp_variable = { to_add = 1 }
		}
		add_to_temp_variable = { idx = 1 }
		modulo_temp_variable = { idx = 15 }
	}
	ITA_camera_update_coalition_status = yes
	ITA_camera_redraw = yes
}
ITA_debug_rem_seats = {
	set_temp_variable = { max_can_rem = ITA_camera_current_deputies^current_number }

		modulo_temp_variable = { idx = 15 }
	}
	ITA_camera_update_coalition_status = yes
	ITA_camera_redraw = yes
}





# Debug function
# Called by the "refresh DDLs" button
# Recomputes all the values of ITA_ddl_final_overton and ITA_ddl_final_focus
ITA_debug_recompute_ddl_final_values = {
	for_each_loop = {
		array = ITA_ddl_history
		value = legge_id
		index = j

		if = {
			limit = { OR = {
				check_variable = { ITA_ddl_history_status^j = 1 }
			}}
			set_temp_variable = { overton = ITA_ddl_base_overton^legge_id }
			set_temp_variable = { focus = 100 }

            set_temp_variable = { base = legge_id }
            multiply_temp_variable = { base = 10 }
            for_loop_effect = {
                start = 0
                end = 10
                value = i

                set_temp_variable = { amendment_id = base }
                add_to_temp_variable = { amendment_id = i }

                if = {
                    limit = {
                        check_variable = { ITA_ddl_amend_available^amendment_id = 2 }
                    }
                    add_to_temp_variable = { overton = ITA_ddl_amend_overton_mod^amendment_id }
					add_to_temp_variable = { focus = ITA_ddl_amend_focus_mod^amendment_num }
                }
			}

			set_variable = { ITA_ddl_final_overton^legge_id = overton }
			set_variable = { ITA_ddl_final_focus^legge_id = focus }
		}
	}
}
