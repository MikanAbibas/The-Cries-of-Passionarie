IGW_play_card = {
	set_country_flag = IGW_picked_card
	set_variable = { IGW_selected_card = IGW_card^i }
	IGW_play_card_effect = yes
	add_to_array = { IGW_played = IGW_selected_card }
	remove_from_array = { IGW_card = IGW_selected_card }

	set_variable = {IGW_count = 0}
	if = {
		limit = { NOT = { has_country_flag = IGW_purge_cards }}
		for_each_loop = {
			array = IGW_card
			value = IGW_card_temp
			break = break

			add_to_variable = {IGW_count = 1}
			add_to_array = { IGW_discard = IGW_card^var:IGW_card_temp }
			remove_from_array = { IGW_card = IGW_card^var:IGW_card_temp }

			if = {
				limit = { check_variable = { IGW_count > 1 } }
				set_temp_variable = { break = 1 }
			}
		}
		else = {
			for_each_loop = {
				array = IGW_card
				value = IGW_card_temp
				break = break

				add_to_variable = {IGW_count = 1}
				if = {
					limit = {
						OR = {
							check_variable = { IGW_card_temp = 303 }
							check_variable = { IGW_card_temp = 306 }
							check_variable = { IGW_card_temp = 311 }
						}
					}
					RKI = {
						add_equipment_to_stockpile = {
							type = infantry_equipment
							amount = 5000
							producer = ITA
						}
					}
				}
				remove_from_array = { IGW_card = IGW_card^var:IGW_card_temp }

				if = {
					limit = { check_variable = { IGW_count > 1 } }
					set_temp_variable = { break = 1 }
				}
			}
			clr_country_flag = IGW_purge_cards
		}
	}
	IGW_count_cards = yes
	clamp_variable = { var = IGW_card_count min = 0 }
}
