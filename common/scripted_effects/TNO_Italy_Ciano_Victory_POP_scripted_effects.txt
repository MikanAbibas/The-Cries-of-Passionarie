	set_variable = { ITA_total_popolo_support = 0 }
	set_temp_variable = { total_possible_support = 0 }

	for_each_loop = {
		array = ITA_Popolo_Support
		index = i
		value = v

		set_temp_variable = { this_support = v }
		multiply_temp_variable = { this_support = ITA_CianoVic_Popolo_Support^i }
		divide_temp_variable = { this_support = 100 }
		add_to_variable = { ITA_total_popolo_support = this_support }
