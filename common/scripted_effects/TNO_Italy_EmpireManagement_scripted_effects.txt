## Used to calculate the variables tracking usage of envoys in regions
ITA_EmpireManagement_Calculate_EnvoyUsage = {
    # Reset variables
    set_variable = { ITA_EM_UsedGarrisons = 0 }
    set_variable = { ITA_EM_UsedDiplomats = 0 }

    # Calculate envoy usage
    for_each_loop = {
        array = ITA_EM_Regions_Envoys
        if = {
            limit = { check_variable = { ITA_EM_Regions_Status^i > 1 } } # If allied or greater - uses Diplomats
            add_to_variable = { ITA_EM_UsedDiplomats = v }
        }
        else = { # Else - uses garrisons
            add_to_variable = { ITA_EM_UsedGarrisons = v }
        }
    }

    # Calculate num of free envoys
    set_variable = { ITA_EM_FreeGarrisons = ITA_EM_TotalGarrisons }
    set_variable = { ITA_EM_FreeDiplomats = ITA_EM_TotalDiplomats }
    subtract_from_variable = { ITA_EM_FreeGarrisons = ITA_EM_UsedGarrisons }
    subtract_from_variable = { ITA_EM_FreeDiplomats = ITA_EM_UsedDiplomats }

    # Calculate progbar percentages
    set_variable = { ITA_EM_GarrisonUsagePerc = ITA_EM_FreeGarrisons }
    set_variable = { ITA_EM_DiplomatUsagePerc = ITA_EM_FreeDiplomats }
    divide_variable = { ITA_EM_GarrisonUsagePerc = ITA_EM_TotalGarrisons }
    divide_variable = { ITA_EM_DiplomatUsagePerc = ITA_EM_TotalDiplomats }
    multiply_variable = { ITA_EM_GarrisonUsagePerc = 100 }
    multiply_variable = { ITA_EM_DiplomatUsagePerc = 100 }

    add_to_variable = { TNO_EmpireManagement_dirty = 1 }
}

# Updates the bonuses of EM's dynamic modifier
ITA_EmpireManagement_Calculate_DynamicModifier = {
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_Calculate_DynamicModifier; Executing."

    # Reset array
    set_variable = { ITA_EM_ModifierEffects^0 = 0 } # Political Power Factor
    set_variable = { ITA_EM_ModifierEffects^1 = 0 } # Weekly Manpower
    set_variable = { ITA_EM_ModifierEffects^2 = 0 } # Stability Factor
    set_variable = { ITA_EM_ModifierEffects^3 = 0 } # Annual GDP Growth Factor
    set_variable = { ITA_EM_ModifierEffects^4 = 0 } # Consumer Goods Use
    set_variable = { ITA_EM_ModifierEffects^5 = 0 } # Oil
    
    for_each_loop = {
        array = ITA_EM_Regions_Status
        if = {
            limit = { check_variable = { v < 2 } } # Is Colony or Vassal
            add_to_variable = { ITA_EM_ModifierEffects^1 = ITA_EM_Regions_DiploValue^i } # Manpower (Control)
            add_to_variable = { ITA_EM_ModifierEffects^2 = ITA_EM_Regions_DiploValue^i } # Stability (Control)
            add_to_variable = { ITA_EM_ModifierEffects^4 = ITA_EM_Regions_EconValue^i  } # Consumer Goods Use (Exploitation)
        }
        else = { # Is anything else
            add_to_variable = { ITA_EM_ModifierEffects^0 = ITA_EM_Regions_DiploValue^i } # Political Power (Favour)
            add_to_variable = { ITA_EM_ModifierEffects^3 = ITA_EM_Regions_EconValue^i  } # Annual GDP Growth Factor (Dependency)
        }
    }
    divide_variable = { ITA_EM_ModifierEffects^0 = 6000 }
    multiply_variable = { ITA_EM_ModifierEffects^1 = 1.1 }
    divide_variable = { ITA_EM_ModifierEffects^2 = 10000 }
    divide_variable = { ITA_EM_ModifierEffects^3 = 5000 }
    divide_variable = { ITA_EM_ModifierEffects^4 = -10000 }
    round_variable = ITA_EM_ModifierEffects^1

    # Oil (Oil Exploitation)
    for_each_loop = {
        array = ITA_EM_HasOil
        var:ITA_EM_HasOil^i = { set_temp_variable = { ITA.RegionOil = resource_produced@oil } }
        multiply_temp_variable = { RegionOil = ITA_EM_OilExploitation^i }
        divide_temp_variable = { RegionOil = 5 }
        add_to_variable = { ITA_EM_ModifierEffects^5 = RegionOil }
    }
    round_variable = ITA_EM_ModifierEffects^5

    force_update_dynamic_modifier = yes
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_Calculate_DynamicModifier; ITA_EM_ModifierEffects^0: [?ITA_EM_ModifierEffects^0|%] ([?ITA_EM_ModifierEffects^0])"
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_Calculate_DynamicModifier; ITA_EM_ModifierEffects^1: [?ITA_EM_ModifierEffects^1]"
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_Calculate_DynamicModifier; ITA_EM_ModifierEffects^2: [?ITA_EM_ModifierEffects^2|%] ([?ITA_EM_ModifierEffects^2])"
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_Calculate_DynamicModifier; ITA_EM_ModifierEffects^3: [?ITA_EM_ModifierEffects^3|%] ([?ITA_EM_ModifierEffects^3])"
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_Calculate_DynamicModifier; ITA_EM_ModifierEffects^4: [?ITA_EM_ModifierEffects^4|%] ([?ITA_EM_ModifierEffects^4])"
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_Calculate_DynamicModifier; ITA_EM_ModifierEffects^5: [?ITA_EM_ModifierEffects^5]"
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_Calculate_DynamicModifier; Execution completed."
}

## Updates the display of alerts in the GUI
ITA_EmpireManagement_UpdateAlerts = {
    for_each_loop = {
        array = ITA_EM_Regions

        if = { # Ongoing Conflict
            limit = {
            	OR = {
            		AND = { #normal check
                		NOT = { check_variable = { ITA_EM_Regions^i = 0 } }
                		NOT = { check_variable = { ITA_EM_Regions^i = 2 } } #isn't sudan
                		check_variable = { ITA_EM_Regions_Status^i > 0 }
                		var:ITA_EM_Regions^i = {
                			#owner is used here because some scopes are states and some are countries
                		    owner = { has_war = yes  }
                		}
            		}
            		AND = { #sudan is marked as having an active conflict if egypt is at war
            			check_variable = { ITA_EM_Regions^i = 2 } #is sudan
            			EGY = {
            				has_war = yes
            			}
            		}
            		#IEA checks
            		AND = {
            			check_variable = { ITA_EM_Regions^i = 3 } # Eritrea 
            			1108 = { # Asmara
            				owner = { has_war = yes }
            			}
            		}
            		AND = {
            			check_variable = { ITA_EM_Regions^i = 4 } # Ethiopia
            			271 = { #Addis Abeba
            				owner = { has_war = yes }
            			}
            		}
            		AND = {
            			check_variable = { ITA_EM_Regions^i = 5 } # Somalia
            			1113 = { #Mogadisco
            				owner = { has_war = yes }
            			}
            		}
            		AND = {
            			check_variable = { ITA_EM_Regions^i = 6 } # Kenya
            			1092 = { #Nairobi
            				owner = { has_war = yes }
            			}
            		}
            		AND = {
            			check_variable = { ITA_EM_Regions^i = AGR }
            			OR = {
            				#either of the contenders in the Algerian civil war is still at war
            				FLA = { has_war = yes }
            				ALG = { has_war = yes }
            				SLG = { has_war = yes }
            				ILG = { has_war = yes }
            			}
            		}
            	}
            }
            set_variable = { ITA_EM_Alerts^i = 1 }
        }
        else_if = { # German Confrontation
            limit = {
                NOT = { check_variable = { ITA_EM_Regions^i = 0 } }
                check_variable = { ITA_EM_Regions_Status^i > 0 }
                always = no #FIXME: needs proper trigger
            }
            set_variable = { ITA_EM_Alerts^i = 2 }
        }
        else_if = { # Low Control
            limit = {
                NOT = { check_variable = { ITA_EM_Regions^i = 0 } }
                check_variable = { ITA_EM_Regions_Status^i < 2 }
                check_variable = { ITA_EM_Regions_DiploValue^i < 20 }
            }
            set_variable = { ITA_EM_Alerts^i = 3 }
        }
        else_if = { # ENI Action Available
            limit = { EmpireManagement_ENIActionIsAvailable = yes }
            set_variable = { ITA_EM_Alerts^i = 4 }
        }
        else = { set_variable = { ITA_EM_Alerts^i = 0 } }
    }

    add_to_variable = { TNO_EmpireManagement_dirty = 1 }
}

    set_variable = { ITA_EM_PolicySwitch_Cooldown^EM_SelectedEntry_i = 180 } # Setup policy switch cooldown (180 days)

    add_to_variable = { TNO_EmpireManagement_dirty = 1 }

## Applies Imperial Policy effects, automatically run every month
# To change the Imperial Policy, just switch the ITA_EM_ImperialPolicy variable to the desired policy ID then run this command
ITA_EmpireManagement_UpdateImperialPolicyEffects = {
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_UpdateImperialPolicyEffects; Executing."

        limit = { check_variable = { ITA_EM_ImperialPolicy = 3 } } # All Roads Lead to Rome

    }
    else_if = {
        limit = { check_variable = { ITA_EM_ImperialPolicy = 4 } } # The New Order
        ITA_EmpireManagement_ApplyImperialPolicyEffect_Giani = yes
    }
    else_if = {
        limit = { check_variable = { ITA_EM_ImperialPolicy = 5 } } # Una Nuova Italia

    }
    else_if = {
        limit = { check_variable = { ITA_EM_ImperialPolicy = 6 } } # Steady as She Goes

    }
    else_if = {
        limit = { check_variable = { ITA_EM_ImperialPolicy = 7 } } # Embers of a New Empire

}

## Adds/subtracts diplomats
# Arguments: DiplomatChange (Default Value: 1)
ITA_EmpireManagement_DiplomatChange = {
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DiplomatChange; Executing."
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DiplomatChange; Given DiplomatChange: [?DiplomatChange?1]"
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DiplomatChange; Operating. Old Total: [?ITA_EM_TotalDiplomats]"
    
    # Defining the tooltip
	if = {
		limit = { check_variable = { DiplomatChange?1 > 0 } }
		custom_effect_tooltip = ITA_EmpireManagement_Diplomats_add_tt
	}
	else = {
		custom_effect_tooltip = ITA_EmpireManagement_Diplomats_sub_tt
	}

	# The actual math
    add_to_variable = { ITA_EM_TotalDiplomats = DiplomatChange?1 }
    clamp_variable = { var = ITA_EM_TotalDiplomats min = 0 }
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DiplomatChange; Operating. New Total: [?ITA_EM_TotalDiplomats]"

    if = { # Edge-case scenario where all diplomats are currently in use and the command is run to remove one or more
        limit = { check_variable = { ITA_EM_TotalDiplomats < ITA_EM_UsedDiplomats } }
        log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DiplomatChange; ITA_EM_TotalDiplomats is inferior to ITA_EM_UsedDiplomats, running balancer script."
        
        # We'll basically just randomly remove diplomats from regions until the balance is achieved
        set_temp_variable = { NumOfDiplomatRemovals = ITA_EM_UsedDiplomats }
        subtract_from_temp_variable = { NumOfDiplomatRemovals = ITA_EM_TotalDiplomats }

        # Create a new array containing all possible target regions
        for_each_loop = {
            array = ITA_EM_Regions_Envoys
            if = {
                limit = {
                    check_variable = { ITA_EM_Regions_Status^i > 1 } # Has to be able to have diplomats in the first place
                    check_variable = { ITA_EM_Regions_Envoys^i > 0 } # Has to have more than one diplomat
                }
                add_to_array = { TargetRegions = i } # Add region index to target array
            }
        }

        # Now we'll run a loop over the amount of removals that we'll need to do
        for_loop_effect = {
            end = NumOfDiplomatRemovals
            
            # Set random variable
            set_temp_variable = { RandomIndex = 0 }
            set_temp_variable = { MaxIndex = TargetRegions^num }
            subtract_from_temp_variable = { MaxIndex = 1 }
            randomize_temp_variable = {
                var = RandomIndex
                distribution = uniform
                min = 0
                max = MaxIndex
            }
            round_temp_variable = RandomIndex # Just in case

            # *Finally* remove the diplomat envoy from a region
            set_temp_variable = { i = TargetRegions^RandomIndex } # Only doing this because array^array^index doesn't work
            subtract_from_variable = { ITA_EM_Regions_Envoys^i = 1 }
            if = { # Now remove the index from the array if the result of the subtraction is zero, so that when we run the loop once more we don't target it again
                limit = { check_variable = { ITA_EM_Regions_Envoys^i = 0 } }
                remove_from_array = { array = TargetRegions index = RandomIndex }
            }
        }
        clear_array = TargetRegions
        log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DiplomatChange; Balancer script completed execution."
    }
    # Yes, the edge case resolver script alone comprises over two thirds of this single command, which normally only adds to a variable then clamps
    # Very cool ain't it

    # Final envoy update
    ITA_EmpireManagement_Calculate_EnvoyUsage = yes

    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DiplomatChange; Completed execution."
}

    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_OilExploitationChange; Executing"

    if = {
        limit = { NOT = { has_variable = Region_i } }
        log = "(ERROR): [GetDateText]: [THIS.GetName]: ITA_EmpireManagement_OilExploitationChange; Handed unset ID. Breaking."
    }
    else = {
        log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_OilExploitationChange; Given Region_i: [?Region_i] OilExploitationChange_v: [?OilExploitationChange_v?0.1]"
        log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_OilExploitationChange; Old Value: [?ITA.ITA_EM_OilExploitation^Region_i]"
        custom_effect_tooltip = ITA_EmpireManagement_OilExploitationChange_tt

        add_to_variable = { ITA.ITA_EM_OilExploitation^Region_i = OilExploitationChange_v?0.1 }
        clamp_variable = { var = ITA.ITA_EM_OilExploitation^Region_i min = 0 max = 1 }

        # Update local resources factor in the target tag
        set_temp_variable = { global.OilExploitationPerc = ITA_EM_OilExploitation^ITA.Region_i } # Yes, this temp variable *has* to be on global scope. Array scopes are very finnicky to work with
        if = {
            limit = { check_variable = { Region_i > 5 } }
            var:ITA_EM_HasOil^Region_i = {
                set_variable = { EM_ENI_Resources_Factor = global.OilExploitationPerc }
                divide_variable = { EM_ENI_Resources_Factor = 4 }
                force_update_dynamic_modifier = yes
            }
        }

        add_to_variable = { ITA.TNO_EmpireManagement_dirty = 1 }
        log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_OilExploitationChange; New Value: [?ITA.ITA_EM_OilExploitation^Region_i]"
    }

    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_OilExploitationChange; Completed execution."
}

## Enables the usage of a specific policy in a specific region
# Arguments: Region_i, Policy_i, DiploValue, EconValue
ITA_EmpireManagement_EnablePolicy = {
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_EnablePolicy; Executing"

    if = {
        limit = { NOT = { has_variable = Region_i } }
        log = "(ERROR): [GetDateText]: [THIS.GetName]: ITA_EmpireManagement_EnablePolicy; Handed unset index. Breaking."
    }
    else_if = {
        limit = { NOT = { has_variable = Policy_i } }
        log = "(ERROR): [GetDateText]: [THIS.GetName]: ITA_EmpireManagement_EnablePolicy; Handed unset ID. Breaking."
    }
    else_if = {
        limit = { NOT = { has_variable = DiploValue } }
        log = "(ERROR): [GetDateText]: [THIS.GetName]: ITA_EmpireManagement_EnablePolicy; Handed unset diplo value. Breaking."
    }
    else_if = {
        limit = { NOT = { has_variable = EconValue } }
        log = "(ERROR): [GetDateText]: [THIS.GetName]: ITA_EmpireManagement_EnablePolicy; Handed unset econ value. Breaking."
    }
    else = {
        log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_EnablePolicy; Given Region_i: [?Region_i] Policy_i: [?Policy_i?1] DiploValue: [?DiploValue] EconValue: [?EconValue]"
        set_temp_variable = { Policy_ID = Policy_i } # Tooltip stuff
        custom_effect_tooltip = ITA_EmpireManagement_EnablePolicy_tt

        add_to_temp_variable = { Region_i = 1 } # Adding one here because the numbered scopes we use to store policies start at 1, not 0
        var:Region_i = {
            add_to_array = { ITA_EM_Policy_ID = ITA.Policy_i }
            add_to_array = { ITA_EM_Policy_DiploEffect = ITA.DiploValue }
            add_to_array = { ITA_EM_Policy_EconEffect = ITA.EconValue }
        }
    }

    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_EnablePolicy; Completed execution."
}

## Disables the usage of a specific policy in a specific region
# Arguments: Region_i, Policy_i
ITA_EmpireManagement_DisablePolicy = {
    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DisablePolicy; Executing"

    if = {
        limit = { NOT = { has_variable = Region_i } }
        log = "(ERROR): [GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DisablePolicy; Handed unset index. Breaking."
    }
    else_if = {
        limit = { NOT = { has_variable = Policy_i } }
        log = "(ERROR): [GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DisablePolicy; Handed unset ID. Breaking."
    }
    else = {
        log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DisablePolicy; Given Region_i: [?Region_i] Policy_i: [?Policy_i?1]"
        set_temp_variable = { Policy_ID = Policy_i } # Tooltip stuff
        custom_effect_tooltip = ITA_EmpireManagement_DisablePolicy_tt

        # In case the policy to be removed is currently being used, we reset the policy choice first
        if = {
            limit = { check_variable = { ITA_EM_Policies^Region_i = Policy_i } }
            log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DisablePolicy; Policy_i under active use, resetting."
            set_variable = { ITA_EM_Policies^Region_i = 0 }
        }

        add_to_temp_variable = { Region_i = 1 } # Adding one here because the numbered scopes we use to store policies start at 1, not 0
        var:Region_i = {
            # We run a check on the array to determine the index of the removed policy, so that we know where we clear the other arrays as well
            for_each_loop = {
                array = ITA_EM_Policy_ID
                if = {
                    limit = { check_variable = { v = ITA.Policy_i } }
                    set_temp_variable = { TempIndex = i }
                    set_temp_variable = { break = 1 } # This forcibly stops the loop once we get what we want
                }
            }
            remove_from_array = { ITA_EM_Policy_ID = ITA.Policy_i }
            remove_from_array = { array = ITA_EM_Policy_DiploEffect index = TempIndex }
            remove_from_array = { array = ITA_EM_Policy_EconEffect  index = TempIndex }
        }
    }

    log = "[GetDateText]: [THIS.GetName]: ITA_EmpireManagement_DisablePolicy; Completed execution."
}
