ITA_GCF_CalcQuadrumvirPow = {
	#There are 6 Gerarchi and 4 candidates, each candidate controlls a portion of each Gerarchi
	#A candidate gets the power of the geriarchi multiplied by the portion of the Gerarchi that the candidate controlls
	#So if Balbo controlls 10% of every geriarchi and every geriarchi has 70 power then he gets (0.1 * 70) * 6 power

	for_each_loop = {
		array = ITA_GCF_QuadrumvirPow
		index = i
		set_variable = { ITA_GCF_QuadrumvirPow^i = 0 }
		for_each_loop = {
			array = ITA_GCF_GerarchiPow
			index = j

			set_temp_variable = { loy_index = i }
			multiply_temp_variable = { loy_index = 10}
			add_to_temp_variable = { loy_index = j }

			set_temp_variable = { amount_to_add = ITA_GCF_GerarchiPow^j }
			multiply_temp_variable = { amount_to_add = ITA_GCF_GerarchiLoy^loy_index}
			multiply_temp_variable = { amount_to_add = 0.01} #loy is from 0-100, converti it to 0-1
			add_to_variable = { ITA_GCF_QuadrumvirPow^i = amount_to_add }
			round_variable = ITA_GCF_QuadrumvirPow^i
		}
	}

	add_to_variable = { ITA_GCF_dirty = 1 }
}

ITA_GCF_AddLoy = {
	#ITA_GCF_fella_to_add
	#ITA_GCF_gerarchi_to_add
	#ITA_GCF_amount_to_add
	#All of these must be set as temp variables.

	if = {
		limit = {
			OR = {
				NOT = {
					has_variable = ITA_GCF_fella_to_add
					has_variable = ITA_GCF_gerarchi_to_add
					has_variable = ITA_GCF_amount_to_add
				}
			}
		}
		set_temp_variable = { ITA_GCF_fella_to_add = 0 }
		set_temp_variable = { ITA_GCF_gerarchi_to_add = 0 }
		set_temp_variable = { ITA_GCF_amount_to_add = 0 }
		log = "[GetDateText]: [Root.GetName]: The proper variables were not set before ITA_GCF_AddLoy was called."
	}

	if = {
		limit = {
			check_variable = { ITA_GCF_amount_to_add > 0 }
		}
		custom_effect_tooltip = ITA_GCF_AddLoy_tt
	}
	else = {
		set_temp_variable = { decrease_in_tt = 1 }
		custom_effect_tooltip = ITA_GCF_AddLoy_tt
		set_temp_variable = { decrease_in_tt = 0 }
	}

	multiply_temp_variable = { ITA_GCF_amount_to_add = 100 }

	set_temp_variable = { fella_to_add_to = ITA_GCF_fella_to_add }
	multiply_temp_variable = { fella_to_add_to = 10 }
	add_to_temp_variable = { fella_to_add_to = ITA_GCF_gerarchi_to_add }

	set_temp_variable = { check_if_it_goes_above_100 = ITA_GCF_GerarchiLoy^fella_to_add_to }
	add_to_temp_variable = { check_if_it_goes_above_100 = ITA_GCF_amount_to_add }
	if = {
		limit = {
			check_variable = { check_if_it_goes_above_100 > 100 }
		}
		subtract_from_temp_variable = { check_if_it_goes_above_100 = 100 }
		subtract_from_temp_variable = { ITA_GCF_amount_to_add = check_if_it_goes_above_100 }
	}
	else_if = {
		limit = {
			check_variable = { check_if_it_goes_above_100 < 0 }
		}
		subtract_from_temp_variable = { ITA_GCF_amount_to_add = check_if_it_goes_above_100 }
	}

	set_temp_variable = {  everyone_else = 100 }
	subtract_from_temp_variable = { everyone_else = ITA_GCF_GerarchiLoy^fella_to_add_to }

	for_each_loop = {
		array = ITA_GCF_QuadrumvirPow
		index = i

		set_temp_variable = { current_fella = i }
		multiply_temp_variable = { current_fella = 10 }
		add_to_temp_variable = { current_fella = ITA_GCF_gerarchi_to_add }

		if = {
			limit = {
				check_variable = { fella_to_add_to = current_fella }
			}
			add_to_variable = { ITA_GCF_GerarchiLoy^current_fella = ITA_GCF_amount_to_add }
		}
		else = {
			set_temp_variable = { amount_to_subtract = ITA_GCF_GerarchiLoy^current_fella }
			divide_temp_variable = { amount_to_subtract = everyone_else }
			multiply_temp_variable = { amount_to_subtract = ITA_GCF_amount_to_add }
			subtract_from_variable = { ITA_GCF_GerarchiLoy^current_fella = amount_to_subtract }
		}

	}

	set_temp_variable = { adjusted_balbo = 40 }
	add_to_temp_variable = { adjusted_balbo = ITA_GCF_gerarchi_to_add }
	set_temp_variable = { unadjusted_balbo = 10 }
	add_to_temp_variable = { unadjusted_balbo = ITA_GCF_gerarchi_to_add }
	set_temp_variable = { pavolini = 0 }
	add_to_temp_variable = { pavolini = ITA_GCF_gerarchi_to_add }

	set_variable = { ITA_GCF_GerarchiLoy^adjusted_balbo = ITA_GCF_GerarchiLoy^unadjusted_balbo }
	add_to_variable = { ITA_GCF_GerarchiLoy^adjusted_balbo = ITA_GCF_GerarchiLoy^pavolini }

	clear_variable = ITA_GCF_fella_to_add
	clear_variable = ITA_GCF_gerarchi_to_add
	clear_variable = ITA_GCF_amount_to_add

	if = {
		limit = {
			has_active_mission = ITA_GCF_100_day_session
		}
		add_to_variable = { ITA_GCF_dirty = 1 }
	}
	else = {
		ITA_GCF_CalcQuadrumvirPow = yes
	}	
}
