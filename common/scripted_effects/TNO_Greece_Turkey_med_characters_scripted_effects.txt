# Leaders for the uprisings of Turkey's unrest mechanic (TNO_Turkey_DtS_scripted_effects.txt).
# The branch set the rising country's ruling party and left the leader to its characters;
# master's history files hold none for those parties, so each rising had no leader. Each
# effect sets a flag so a second rising does not add a second copy (a name check would
# not do: master's Georgia already has a fascist Shalva Loladze).
KUR_med_promote_Musa_Anter = {
	log = "[GetDateText]: [Root.GetName]: KUR_med_promote_Musa_Anter"
	if = {
		limit = { NOT = { has_country_flag = KUR_med_leader_Musa_Anter } }
		set_country_flag = KUR_med_leader_Musa_Anter
		create_country_leader = {
			name = "Musa Anter"
			picture = "gfx/leaders/KUR/Portrait_Musa_Anter.dds"
			expire = "1999.1.1"
			ideology = socialist_left_wing_nationalism_subtype
			traits = { }
		}
	}
}
