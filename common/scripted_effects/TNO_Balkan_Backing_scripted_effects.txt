# Balkan backing — resolving who sponsors whom.
#
# Each axis resolves once, into four global flags (BALKAN_<ITA|GER>_proxy_<TAG>). Everything
# downstream reads those, never the Great Game flags directly, so a later rule change only
# touches this file.

# ---- Transylvania: Hungary vs Romania ----------------------------------------------------

# ---- resolution ---------------------------------------------------------------------------
# Called when the crisis opens. Anything that is NOT a both-halves case settles itself here;
# the ambiguous cases are left for the decisions, with the mission as the deadline.
BALKAN_resolve_backing = {
	if = {
		limit = { NOT = { has_global_flag = BALKAN_transylvania_resolved } }
		if = {
			limit = {
				BALKAN_HUN_won_by_ITA = yes
				NOT = { BALKAN_ROM_won_by_ITA = yes }
			}
			set_global_flag = BALKAN_ITA_proxy_HUN
			set_global_flag = BALKAN_GER_proxy_ROM
			set_global_flag = BALKAN_transylvania_resolved
		}
		else_if = {
			limit = {
				BALKAN_HUN_won_by_GER = yes
				NOT = { BALKAN_ROM_won_by_GER = yes }
			}
			set_global_flag = BALKAN_GER_proxy_HUN
			set_global_flag = BALKAN_ITA_proxy_ROM
			set_global_flag = BALKAN_transylvania_resolved
		}
		else_if = {
			limit = {
				BALKAN_ROM_won_by_ITA = yes
				NOT = { BALKAN_HUN_won_by_ITA = yes }
			}
			set_global_flag = BALKAN_ITA_proxy_ROM
			set_global_flag = BALKAN_GER_proxy_HUN
			set_global_flag = BALKAN_transylvania_resolved
		}
		else_if = {
			limit = {
				BALKAN_ROM_won_by_GER = yes
				NOT = { BALKAN_HUN_won_by_GER = yes }
			}
			set_global_flag = BALKAN_GER_proxy_ROM
			set_global_flag = BALKAN_ITA_proxy_HUN
			set_global_flag = BALKAN_transylvania_resolved
		}
		else = {
			# Either one power holds both, or nobody won either. The first case waits on a
			# decision; the second has nothing to decide, so settle it now at random.
			if = {
				limit = {
					NOT = { BALKAN_ITA_holds_transylvania_pair = yes }
					NOT = { BALKAN_GER_holds_transylvania_pair = yes }
				}
				BALKAN_random_transylvania = yes
			}
			else = {
				ITA = { activate_mission = BALKAN_choose_transylvania_proxy }
				GER = { activate_mission = BALKAN_choose_transylvania_proxy }
			}
		}
	}
	if = {
		limit = { NOT = { has_global_flag = BALKAN_macedonia_resolved } }
		if = {
			limit = {
				BALKAN_BUL_won_by_ITA = yes
				NOT = { BALKAN_GRE_won_by_ITA = yes }
			}
			set_global_flag = BALKAN_ITA_proxy_BUL
			set_global_flag = BALKAN_GER_proxy_GRE
			set_global_flag = BALKAN_macedonia_resolved
		}
		else_if = {
			limit = {
				BALKAN_BUL_won_by_GER = yes
				NOT = { BALKAN_GRE_won_by_GER = yes }
			}
			set_global_flag = BALKAN_GER_proxy_BUL
			set_global_flag = BALKAN_ITA_proxy_GRE
			set_global_flag = BALKAN_macedonia_resolved
		}
		else = {
			if = {
				limit = {
					NOT = { BALKAN_ITA_holds_macedonia_pair = yes }
					NOT = { BALKAN_GER_holds_macedonia_pair = yes }
				}
				BALKAN_random_macedonia = yes
			}
			else = {
				ITA = { activate_mission = BALKAN_choose_macedonia_proxy }
				GER = { activate_mission = BALKAN_choose_macedonia_proxy }
			}
		}
	}
}

# ---- the deadline fired without a choice --------------------------------------------------
BALKAN_random_transylvania = {
	random_list = {
		50 = { BALKAN_favour_HUN_for_ITA = yes }
		50 = { BALKAN_favour_ROM_for_ITA = yes }
	}
}
BALKAN_random_macedonia = {
	random_list = {
		50 = { BALKAN_favour_BUL_for_ITA = yes }
		50 = { BALKAN_favour_GRE_for_ITA = yes }
	}
}

# If Germany is the one holding both, the random fallback has to land on Germany's side.
BALKAN_random_transylvania_for = {
	if = {
		limit = { BALKAN_GER_holds_transylvania_pair = yes }
		random_list = {
			50 = { BALKAN_favour_HUN_for_GER = yes }
			50 = { BALKAN_favour_ROM_for_GER = yes }
		}
	}
	else = {
		BALKAN_random_transylvania = yes
	}
}

	}

# ---- post-war settlement -------------------------------------------------------------------
# Each belligerent joins the faction of whichever power actually sponsored it, instead of the
# old hardcoding (Hungary winning used to drag BOTH sides into the Pakt). A nation that already
# holds a faction idea keeps it: the Balkan war does not re-badge someone who never changed camp.
BALKAN_place_with_sponsor = {
	if = {
		limit = {
			NOT = {
				OR = {
					has_idea = Pakt_Bundnispartner
					has_idea = Pakt_Mitstreiter
					has_idea = Pakt_Marionettenstaat
					has_idea = Pakt_Protektorat
					has_idea = TRI_Member
					has_idea = TRI_Member_Client
					has_idea = Rome_member
					has_idea = Rome_client
				}
			}
		}
		if = {
			limit = {
				OR = {
					AND = { tag = HUN  has_global_flag = BALKAN_GER_proxy_HUN }
					AND = { tag = ROM  has_global_flag = BALKAN_GER_proxy_ROM }
					AND = { tag = BUL  has_global_flag = BALKAN_GER_proxy_BUL }
					AND = { tag = GRE  has_global_flag = BALKAN_GER_proxy_GRE }
				}
			}
			add_ideas = Pakt_Bundnispartner
			GER = { add_to_faction = PREV }
		}
		else_if = {
			limit = {
				OR = {
					AND = { tag = HUN  has_global_flag = BALKAN_ITA_proxy_HUN }
					AND = { tag = ROM  has_global_flag = BALKAN_ITA_proxy_ROM }
					AND = { tag = BUL  has_global_flag = BALKAN_ITA_proxy_BUL }
					AND = { tag = GRE  has_global_flag = BALKAN_ITA_proxy_GRE }
				}
			}
			add_ideas = Rome_member
			ITA = { add_to_faction = PREV }
		}
	}
}

# A belligerent only breaks with its bloc when the enemy is a bloc partner. Fighting someone
# from the other camp costs it nothing — that is the whole point of a proxy war.
# Run in the belligerent's scope with the opponent saved as event_target:BALKAN_opponent.
BALKAN_sever_if_former_ally = {
	if = {
		limit = { is_in_faction_with = event_target:BALKAN_opponent }
		leave_faction = yes
	}
}
