SAW_record_outcome = {
	# New Scramble for Africa (plan 3.1). How the South African War ended. Each of the five endings sets
	# global.SAW_outcome first and then calls this, so exactly one SAW_outcome_* flag is set:
	#   1 OFN total (SAFWAR.13), 2 OFN leaning (.22), 3 status quo (.21), 4 Pakt leaning (.20), 5 Pakt total (.9)
	clr_global_flag = SAW_outcome_ofn_total
	clr_global_flag = SAW_outcome_ofn_leaning
	clr_global_flag = SAW_outcome_status_quo
	clr_global_flag = SAW_outcome_pakt_leaning
	clr_global_flag = SAW_outcome_pakt_total
	if = {
		limit = { check_variable = { global.SAW_outcome = 1 } }
		set_global_flag = SAW_outcome_ofn_total
	}
	else_if = {
		limit = { check_variable = { global.SAW_outcome = 2 } }
		set_global_flag = SAW_outcome_ofn_leaning
	}
	else_if = {
		limit = { check_variable = { global.SAW_outcome = 3 } }
		set_global_flag = SAW_outcome_status_quo
	}
	else_if = {
		limit = { check_variable = { global.SAW_outcome = 4 } }
		set_global_flag = SAW_outcome_pakt_leaning
	}
	else_if = {
		limit = { check_variable = { global.SAW_outcome = 5 } }
		set_global_flag = SAW_outcome_pakt_total
	}
	log = "[GetDateText]: South African War outcome recorded: [?global.SAW_outcome]"
}

SFA_cut_old_ties = {

SFA_arm_nations = {
	# user's test (2026-10-18): the collapse releases most nations with no army. Master's revolts only move states, and
	# the history OOBs those tags name (BFR_1962, CAF_1962) do not exist. Every scramble nation with fewer than three
	# divisions gets the techs for infantry, two templates, a militia division in each state it holds and three
	# infantry divisions at its capital. Runs once, from SFA_start; Italy's and Iberia's proxies bring their own OOBs.
	every_country = {
		limit = {
			has_country_flag = SFA_african_nation
			NOT = { has_country_flag = SFA_zanzibar }
			NOT = { has_army_size = { size > 2 } }
		}
		if = {
			limit = { NOT = { has_tech = infantry_weapons_1 } }
			set_technology = { infantry_weapons_1 = 1 }
		}
		if = {
			limit = { NOT = { has_tech = infantry_at } }
			set_technology = { infantry_at = 1 }
		}
		if = {
			limit = { NOT = { has_template = "Militia" } }
			division_template = {
				name = "Militia"
				regiments = {
					light_infantry = { x = 0 y = 0 }
					light_infantry = { x = 0 y = 1 }
					light_infantry = { x = 1 y = 0 }
					light_infantry = { x = 1 y = 1 }
				}
			}
		}
		if = {
			limit = { NOT = { has_template = "Infantry" } }
			division_template = {
				name = "Infantry"
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 1 y = 0 }
					infantry = { x = 1 y = 1 }
					infantry = { x = 2 y = 0 }
					infantry = { x = 2 y = 1 }
				}
			}
		}
		every_owned_state = {
			limit = { is_controlled_by = PREV }
			create_unit = {
				division = "division_template = \"Militia\" start_experience_factor = 0.2 start_equipment_factor = 1.0"
				owner = PREV
			}
		}
		capital_scope = {
			create_unit = {
				division = "division_template = \"Infantry\" start_experience_factor = 0.3 start_equipment_factor = 1.0"
				owner = PREV
				count = 3
			}
		}
		log = "[GetDateText]: New Scramble for Africa: [This.GetName] raises its army"
	}
}

SFA_spawn_italian_proxy = {
	# plan 4.1: SWA, the Autorità Militare dell'Africa Meridionale, replaces Luoland (user, 2026-10-16): it takes Kisumu
	# (1091) and everything else LUO holds, and LUO is gone. If LUO no longer holds Kisumu, SWA takes Kisumu from
	# whoever does, as long as they keep another state; failing that, Arusha (1079). Cores on all of it.
	# Runs once, from Italy's notice.
	if = {
		limit = { NOT = { country_exists = SWA } }
		if = {
			limit = {
				OR = {
					1091 = { is_owned_by = LUO }
					1091 = {
						OWNER = {
							any_owned_state = { NOT = { state = 1091 } }
						}
					}
				}
			}
			SWA = {
				set_state_owner = 1091
				set_state_controller = 1091
			}
		}
		else = {
			SWA = {
				set_state_owner = 1079
				set_state_controller = 1079
			}
		}
		if = {
			limit = { country_exists = LUO }
			LUO = {
				every_owned_state = {
					transfer_state_to = SWA
					set_state_controller_to = SWA
				}
			}
		}
		SWA = {
			every_owned_state = { add_core_of = SWA }
			add_to_array = { global.SFA_nations = THIS }
			set_country_flag = SFA_african_nation
			set_country_flag = SFA_patron_ITA
			SFA_set_nation_laws = yes
			inherit_technology = ITA
			# Italy's industry supplies it from home (user, 2026-10-18)
			add_offsite_building = { type = arms_factory level = 15 }
			add_manpower = 20000
			# about twice what its OOB fields (user, 2026-10-27): 24 infantry battalions (100 rifles, 15 AT each) and 8
			# light infantry (80 rifles) need 3040 rifles and 360 anti-tank
			add_equipment_to_stockpile = { type = infantry_equipment amount = 6000 producer = ITA }
			add_equipment_to_stockpile = { type = anti_tank_equipment amount = 800 producer = ITA }
			add_equipment_to_stockpile = { type = support_equipment amount = 800 producer = ITA }
			if = {
				limit = { owns_state = 1091 }
				set_capital = { state = 1091 }
				load_oob = "SWA_scramble_kisumu"
			}
			else = {
				set_capital = { state = 1079 }
				load_oob = "SWA_scramble_arusha"
			}
		}
		log = "[GetDateText]: New Scramble for Africa: SWA spawned for Italy"
		SFA_board_recompute = yes
		SFA_gui_refresh = yes
	}
}

	# (user, 2026-10-16: no absorption). Wars that ended other ways get their cooldowns first, so the region check and
	# the AI's war windows see the map as it stands.
	if = {
		# a save from before the wars existed
		limit = { NOT = { has_global_flag = SFA_nations_built } }
		SFA_build_nations = yes
	}
	if = {
		# a save from before the old ties were cut and the empty nations armed (2026-10-18)
		limit = { NOT = { has_global_flag = SFA_nations_armed } }
		set_global_flag = SFA_nations_armed
		SFA_cut_old_ties = yes
		SFA_arm_nations = yes
	}
	if = {
		# a save from before the scramble nations got their laws (2026-10-24)
		limit = { NOT = { has_global_flag = SFA_laws_set } }
		set_global_flag = SFA_laws_set
		SFA_set_all_nation_laws = yes
	}
	SFA_free_the_boers = yes
	SFA_adopt_late_allies = yes
	SFA_check_wars = yes
	SFA_check_iberian_wars = yes
	SFA_check_regions = yes
	SFA_update_home_regions = yes
	SFA_update_income = yes
	SFA_burgundy_african_supplies = yes
	SFA_update_land_strength = yes
	SFA_supply_repairs = yes

}

SFA_check_current_region = {
	# plan 4.3, for the states in global.SFA_cur (filled by SFA_check_region_<n>, which sets temp sfa_cur_region). Sets
	# temp sfa_result: 0 open, 1 US, 2 Germany, 3 Italy, 4 Japan, 5 Burgundy, 6 Iberia, 9 one independent holds it.
	# Only the nations based in the region count (user, 2026-10-19): it consolidates once every nation with its capital
	# there that holds land in it is on one side, or a single independent is left. Land that outsiders hold there (South
	# Africa's in Mosambik after an OFN-leaning war) neither blocks it nor merges. A region with no local holder
	# consolidates only when South Africa or the Boers hold all of it (user, 2026-10-19). Zanzibar never blocks a region
	# and joins the merged nation. Saves the survivor as event target SFA_survivor.
	set_temp_variable = { sfa_result = 0 }
	set_temp_variable = { sfa_have_cand = 0 }
	set_temp_variable = { sfa_whole_holder = 0 }
	random_scope_in_array = {
		array = global.SFA_cur
		limit = {
			NOT = { is_owned_by = ZZB }
			NOT = { SFA_minor_island = yes }
			SFA_held_locally = yes
		}
		OWNER = { save_event_target_as = SFA_cand }
		set_temp_variable = { sfa_have_cand = 1 }
	}
	if = {
		# no local holder: South Africa or the Boers holding the whole region
		limit = { check_variable = { sfa_have_cand = 0 } }
		random_scope_in_array = {
			array = global.SFA_cur
			limit = {
				NOT = { is_owned_by = ZZB }
				NOT = { SFA_minor_island = yes }
				OWNER = {
					OR = {
						tag = SAF
						tag = BOR
					}
				}
			}
			OWNER = { save_event_target_as = SFA_cand }
			set_temp_variable = { sfa_have_cand = 2 }
		}
		if = {
			limit = {
				check_variable = { sfa_have_cand = 2 }
				all_of_scopes = {
					array = global.SFA_cur
					OR = {
						is_owned_by = ZZB
						SFA_minor_island = yes
						is_owned_by = event_target:SFA_cand
					}
				}
			}
			event_target:SFA_cand = {
				SFA_get_patron_code = yes
				save_event_target_as = SFA_survivor
			}
			set_temp_variable = { sfa_result = sfa_code }
			set_temp_variable = { sfa_whole_holder = 1 }
		}
	}
	else = {
		if = {
			limit = {
				all_of_scopes = {
					array = global.SFA_cur
					OR = {
						is_owned_by = ZZB
						SFA_minor_island = yes
						NOT = { SFA_held_locally = yes }
						OWNER = { has_country_flag = SFA_patron_USA }
					}
				}
			}
			set_temp_variable = { sfa_result = 1 }
		}
		else_if = {
			limit = {
				all_of_scopes = {
					array = global.SFA_cur
					OR = {
						is_owned_by = ZZB
						SFA_minor_island = yes
						NOT = { SFA_held_locally = yes }
						OWNER = { has_country_flag = SFA_patron_GER }
					}
				}
			}
			set_temp_variable = { sfa_result = 2 }
		}
		else_if = {
			limit = {
				all_of_scopes = {
					array = global.SFA_cur
					OR = {
						is_owned_by = ZZB
						SFA_minor_island = yes
						NOT = { SFA_held_locally = yes }
						OWNER = { has_country_flag = SFA_patron_ITA }
					}
				}
			}
			set_temp_variable = { sfa_result = 3 }
		}
		else_if = {
			limit = {
				all_of_scopes = {
					array = global.SFA_cur
					OR = {
						is_owned_by = ZZB
						SFA_minor_island = yes
						NOT = { SFA_held_locally = yes }
						OWNER = { has_country_flag = SFA_patron_JAP }
					}
				}
			}
			set_temp_variable = { sfa_result = 4 }
		}
		else_if = {
			limit = {
				all_of_scopes = {
					array = global.SFA_cur
					OR = {
						is_owned_by = ZZB
						SFA_minor_island = yes
						NOT = { SFA_held_locally = yes }
						OWNER = { has_country_flag = SFA_patron_BRG }
					}
				}
			}
			set_temp_variable = { sfa_result = 5 }
		}
		else_if = {
			limit = {
				all_of_scopes = {
					array = global.SFA_cur
					OR = {
						is_owned_by = ZZB
						SFA_minor_island = yes
						NOT = { SFA_held_locally = yes }
						OWNER = { has_country_flag = SFA_patron_IBR }
					}
				}
			}
			set_temp_variable = { sfa_result = 6 }
		}
		else_if = {
			limit = {
				event_target:SFA_cand = {
					OR = {
						has_country_flag = SFA_african_nation
						has_country_flag = SFA_sovereign_ally
					}
				}
				all_of_scopes = {
					array = global.SFA_cur
					OR = {
						is_owned_by = ZZB
						SFA_minor_island = yes
						NOT = { SFA_held_locally = yes }
						is_owned_by = event_target:SFA_cand
					}
				}
			}
			set_temp_variable = { sfa_result = 9 }
		}
	}
	if = {
		limit = { check_variable = { sfa_result > 0 } }
		if = {
			limit = { check_variable = { sfa_whole_holder = 0 } }
			SFA_pick_survivor = yes
		}
		SFA_merge_into_survivor = yes
	}
}

SFA_pick_survivor = {
	# plan 4.3: among the nations based in the region (SFA_held_locally), the one with the most states there survives,
	# ties to the higher state value; a sovereign ally (the Madagascar winner taking the Comoros) always does. Zanzibar
	# never survives. Reads temp sfa_cur_region, set by SFA_check_region_<n>.
	for_each_scope_loop = {
		array = global.SFA_cur
		if = {
			limit = {
				SFA_held_locally = yes
				NOT = { SFA_minor_island = yes }
			}
			OWNER = {
				set_variable = { SFA_cur_states = 0 }
				set_variable = { SFA_cur_value = 0 }
			}
		}
	}
	for_each_scope_loop = {
		array = global.SFA_cur
		if = {
			limit = {
				SFA_held_locally = yes
				NOT = { SFA_minor_island = yes }
			}
			OWNER = {
				add_to_variable = { SFA_cur_states = 1 }
				add_to_variable = { SFA_cur_value = PREV.SFA_value }
			}
		}
	}
	set_temp_variable = { sfa_best = -1 }
	for_each_scope_loop = {
		array = global.SFA_cur
		if = {
			limit = {
				SFA_held_locally = yes
				NOT = { SFA_minor_island = yes }
			}
			OWNER = {
				if = {
					limit = { NOT = { tag = ZZB } }
					set_temp_variable = { sfa_score = SFA_cur_states }
					multiply_temp_variable = { var = sfa_score value = 1000 }
					add_to_temp_variable = { sfa_score = SFA_cur_value }
					if = {
						limit = { has_country_flag = SFA_sovereign_ally }
						add_to_temp_variable = { sfa_score = 2000000 }
					}
					if = {
						limit = { check_variable = { sfa_score > sfa_best } }
						set_temp_variable = { sfa_best = sfa_score }
						save_event_target_as = SFA_survivor
					}
				}
			}
		}
	}
}

SFA_merge_into_survivor = {
	# plan 4.3 (user): the survivor absorbs the other nations based in the region, and Zanzibar, whole; sovereign allies
	# and outsiders holding land there are never absorbed (user, 2026-10-19). A player on an absorbed nation moves to
	# the survivor first (change_tag_from, as TNO_Ostland2.txt does).
	event_target:SFA_survivor = { set_country_flag = SFA_merge_survivor }
	set_temp_variable = { sfa_order_code = sfa_result }
	for_each_scope_loop = {
		array = global.SFA_cur
		if = {
			limit = {
				OR = {
					is_owned_by = ZZB
					AND = {
						SFA_held_locally = yes
						NOT = { SFA_minor_island = yes }
					}
					AND = {
						SFA_minor_island = yes
						OWNER = { SFA_on_merge_side = yes }
					}
				}
			}
			OWNER = {
				if = {
					limit = {
						NOT = { has_country_flag = SFA_merge_survivor }
						NOT = { has_country_flag = SFA_sovereign_ally }
					}
					if = {
						limit = {
							is_ai = no
							event_target:SFA_survivor = { is_ai = yes }
						}
						event_target:SFA_survivor = { change_tag_from = PREV }
					}
					if = {
						# two human nations never merge
						limit = { is_ai = yes }
						log = "[GetDateText]: New Scramble for Africa: [SFA_survivor.GetName] absorbs [This.GetName] in the merger"
						event_target:SFA_survivor = { annex_country = { target = PREV transfer_troops = yes } }
					}
				}
			}
		}
	}
	event_target:SFA_survivor = {
		clr_country_flag = SFA_merge_survivor
		every_owned_state = {
			limit = {
				has_state_flag = SFA_state
				NOT = { is_core_of = PREV }
			}
			add_core_of = PREV
		}
	}
}
