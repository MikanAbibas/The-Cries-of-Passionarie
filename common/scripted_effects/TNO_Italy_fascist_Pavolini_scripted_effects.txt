ITA_RQSC_CALC = {
	ITA_RQSC_calc_duce_stability = yes

	for_each_loop = {
		array = ITA_RQSC_structure
		index = i
		value = val

		if = {
		limit = { check_variable = { i < 7 } }
		clamp_variable = {
			var = ITA_RQSC_structure^i
			min = 0
			max = 100
		}
	}
	}
	for_each_loop = {
		array = ITA_RQSC_control
		index = i
		value = val

		if = {
		limit = { check_variable = { i < 7 } }
		clamp_variable = {
			var = ITA_RQSC_control^i
			min = 0
			max = 100
		}
	}
	}
	clamp_variable = {
		var = ITA_RQSC_centralisation
		min = 0
		max = 100
	}
	#debuffs
	set_variable = { RQSC_church_debuff = ITA_RQSC_structure^1 }
	subtract_from_variable = { RQSC_church_debuff = 100 }
	divide_variable = { RQSC_church_debuff = 250 }
	set_variable = { RQSC_confidustria_debuff = ITA_RQSC_structure^2 }
	subtract_from_variable = { RQSC_confidustria_debuff = 100 }

ITA_RQSC_setup_interaction_array = {
	#array_num and interaction_target must be set as temp vars before this is called

	#set the available interactions in the array_num interaction slot to the interaction_target's possible interactions
	#interactions 0, 1, 2 are generic, available for everyone
	#after that every pillar has 3 interactions (3, 4, 5 is the church, 6, 7, 8 is confiduistria, etc.)

	if = {
		limit = {
			check_variable = { array_num = 1 }
		}

		clear_array = ITA_RQSC_GUI_Interactions_array_1

		add_to_temp_variable = { interaction_target = 1 }
		multiply_temp_variable = { interaction_target = 3 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_1 = interaction_target }
		add_to_temp_variable = { interaction_target = 1 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_1 = interaction_target }
		add_to_temp_variable = { interaction_target = 1 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_1 = interaction_target }

		add_to_array = { ITA_RQSC_GUI_Interactions_array_1 = 0 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_1 = 1 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_1 = 2 }

	}
	else_if = {
		limit = {
			check_variable = { array_num = 2 }
		}

		clear_array = ITA_RQSC_GUI_Interactions_array_2
		
		add_to_temp_variable = { interaction_target = 1 }
		multiply_temp_variable = { interaction_target = 3 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_2 = interaction_target }
		add_to_temp_variable = { interaction_target = 1 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_2 = interaction_target }
		add_to_temp_variable = { interaction_target = 1 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_2 = interaction_target }

		add_to_array = { ITA_RQSC_GUI_Interactions_array_2 = 0 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_2 = 1 }
		add_to_array = { ITA_RQSC_GUI_Interactions_array_2 = 2 }
	}
}

	add_to_temp_variable = { stab_average = ITA_RQSC_structure^2  }
	add_to_temp_variable = { stab_average = ITA_RQSC_structure^3  }
	add_to_temp_variable = { stab_average = ITA_RQSC_structure^4  }
	add_to_temp_variable = { stab_average = ITA_RQSC_structure^5  }
	add_to_temp_variable = { stab_average = ITA_RQSC_structure^6  }

	divide_temp_variable = { stab_average = 6 }
	divide_temp_variable = { stab_average = 2 }

	set_variable = { ITA_RQSC_structure^0 = 100 }
	subtract_from_variable = { ITA_RQSC_structure^0 = stab_average }
}

ITA_RQSC_change_control = {
	#temp_pillar, temp_amount must be set as temp variables before this is called

	if = {
		limit = {
			NOT = {
				OR = {
					has_variable = temp_pillar
					has_variable = temp_amount
				}
			}
		}
		#changes control by 0
		set_temp_variable = { temp_pillar = 0 }
		set_temp_variable = { temp_amount = 0 }
		set_temp_variable = { vars_not_set = 1 }
		log = "ERROR: ITA_RQSC_change_control was called without temp_pillar and temp_amount being set, nothing will happen."
	}

	add_to_variable = { ITA_RQSC_control^temp_pillar = temp_amount }
	ITA_RQSC_CALC = yes

	#if temp_amount is negative use it's absolute value
	#this is done to turn "decrease our control of X by -2" to "decrease our control of X by 2"
	#as the former is confusing
	if = {
		limit = {
			check_variable = { temp_amount < 0 }
		}
		set_temp_variable = { decrease_in_tt = 1 }
		set_temp_variable = { red_in_tt = 1 }
		multiply_temp_variable = { temp_amount = -1 }
	}
	else = {
		set_temp_variable = { decrease_in_tt = 0 }
		set_temp_variable = { red_in_tt = 0 }
	}

	#temp var for a scripted loc in the tooltip
	set_temp_variable = { group_temp = temp_pillar }

	#if some var was not set display an error tooltip
	if = {
		limit = {
			check_variable = { vars_not_set = 1 }
		}
		custom_effect_tooltip = ITA_RQSC_change_control_error_tt
	}
	else = {
		custom_effect_tooltip = ITA_RQSC_change_control_tt
	}

	clear_variable = temp_pillar
	clear_variable = temp_amount
	clear_variable = vars_not_set
}
