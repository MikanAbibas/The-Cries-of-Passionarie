# Bloc aid — the transfers.
#
# BLOC_AID_open_panel runs in the SELECTED country's scope (from the button). Everything else
# runs in the PLAYER's scope (from the panel), with the recipient as event_target:BLOC_AID_target.

# Stash the recipient, then raise the flag on the patron so the panel opens.
BLOC_AID_open_panel = {
	save_global_event_target_as = BLOC_AID_target
	every_country = {
		limit = { is_ai = no }
		set_country_flag = BLOC_AID_panel_open
	}
}

# Runs in the PLAYER's scope.
BLOC_AID_close_panel = {
	clr_country_flag = BLOC_AID_panel_open
}

BLOC_AID_send_arms = {
	log = "[GetDateText]: [This.GetName] bloc aid arms -> [BLOC_AID_target.GetName]"
	add_equipment_to_stockpile = { type = infantry_equipment amount = -4000 }
	add_equipment_to_stockpile = { type = support_equipment amount = -800 }
	# One transfer for every patron: 'producer' takes a scope keyword (vanilla uses
	# PREV 41 times), so the bloc leader no longer has to be one of six hard-coded tags.
	event_target:BLOC_AID_target = {
		add_equipment_to_stockpile = { type = infantry_equipment amount = 4000 producer = PREV }
		add_equipment_to_stockpile = { type = support_equipment amount = 800 producer = PREV }
	}
	BLOC_AID_stamp_cd_arms = yes
}

# Runs in the RECIPIENT scope. Only the six Balkan tags have an aid OOB written.
BLOC_AID_load_aid_oob = {
	if = { limit = { tag = ROM } load_oob = "ROM_bloc_aid" }
	if = { limit = { tag = HUN } load_oob = "HUN_bloc_aid" }
	if = { limit = { tag = BUL } load_oob = "BUL_bloc_aid" }

	if = { limit = { tag = CRO } load_oob = "CRO_bloc_aid" }
}

# Triumvirate, so Italy's pre- and post-Malta roles do not share a timer.
