#Corruption monthly gain scripted effect

ITA_balboimper_corruption_monthlygain_update = {
	
	if = {
		limit = {
			has_idea = tno_academic_base_golden_age
		}
		set_variable = { ITA_balboimper_corruption_gain = -0.02 }
	}
	else_if = {
		limit = {
			has_idea = tno_academic_base_tertiary_schooling
		}
		set_variable = { ITA_balboimper_corruption_gain = 0 }
	}
	else_if = {
		limit = {
			has_idea = tno_academic_base_secondary_schooling
		}
		set_variable = { ITA_balboimper_corruption_gain = 0.01 }
	}
	else_if = {
		limit = {
			has_idea = tno_academic_base_primary_schooling
		}
		set_variable = { ITA_balboimper_corruption_gain = 0.015 }
	}
	else_if = {
		limit = {
			has_idea = tno_academic_base_basic_literacy
		}
		set_variable = { ITA_balboimper_corruption_gain = 0.03 }
	}
	else_if = {
		limit = {
			has_idea = tno_academic_base_mass_illiteracy
		}
		set_variable = { ITA_balboimper_corruption_gain = 0.1 }
	}
}

#Calculate new monthly costs (After special "upgrade efficiency" decision)

ITA_balboimper_increase_monthlycosts = { #ALSO GIVES EFFICIENCY, DONT DOUBLE DIP IN DECISION EFFECTS
	
	##ADDING EFFICIENCY##
	
	add_to_variable = { ITA_balboimper_Efficiency = 0.04 }
	add_to_variable = { ITA_balboimper_GUIEfficiencyBar = 4 }
	ITA_balboimper_clamp_efficiency = yes
	
	##THE MONTHLY COST VARIABLE CHANGE##
	
	#Subtracts all previous project costs 
	
	set_temp_variable = { ITA_balboimper_S1S2_totalcosts_temp = ITA_balboimper_S1S2_costs }
	multiply_temp_variable = { ITA_balboimper_S1S2_totalcosts_temp = ITA_balboimper_ongoing_S1S2_projects }
	
	set_temp_variable = { ITA_balboimper_S3_totalcosts_temp = ITA_balboimper_S3_costs }
	multiply_temp_variable = { ITA_balboimper_S3_totalcosts_temp = ITA_balboimper_ongoing_S3_projects }
	
	subtract_from_variable = { ITA_balboimper_misccosts = ITA_balboimper_S1S2_totalcosts_temp }
	subtract_from_variable = { ITA_balboimper_misccosts = ITA_balboimper_S3_totalcosts_temp }
	
	#Calculates new project costs (IMPORTANT, WILL MAYBE BE CHANGED IN BALANCING)
	
	multiply_variable = { ITA_balboimper_S1S2_costs = 1.3 } #MODIFIER HERE
	multiply_variable = { ITA_balboimper_S3_costs = 1.3 } #MODIFIER HERE
	
	#Adds new project costs (same as in decision scripted effects i guess?)
	
	set_temp_variable = { ITA_balboimper_S1S2_totalcosts2_temp = ITA_balboimper_S1S2_costs }
	multiply_temp_variable = { ITA_balboimper_S1S2_totalcosts2_temp = ITA_balboimper_ongoing_S1S2_projects }
	
	set_temp_variable = { ITA_balboimper_S3_totalcosts2_temp = ITA_balboimper_S3_costs }
	multiply_temp_variable = { ITA_balboimper_S3_totalcosts2_temp = ITA_balboimper_ongoing_S3_projects }
	
	add_to_variable = { ITA_balboimper_misccosts = ITA_balboimper_S1S2_totalcosts2_temp }
	add_to_variable = { ITA_balboimper_misccosts = ITA_balboimper_S3_totalcosts2_temp }
	
	##AND THEN ALL OVER AGAIN FOR THE COSMETIC MONTHLY STUFF##
	
	#Subtracts cosmetic monthly costs
	
	set_temp_variable = { ITA_balboimper_S1S2_totalcosts_monthly_temp = ITA_balboimper_S1S2_costs_monthly }
	multiply_temp_variable = { ITA_balboimper_S1S2_totalcosts_monthly_temp = ITA_balboimper_ongoing_S1S2_projects }
	
	set_temp_variable = { ITA_balboimper_S3_totalcosts_monthly_temp = ITA_balboimper_S3_costs_monthly }
	multiply_temp_variable = { ITA_balboimper_S3_totalcosts_monthly_temp = ITA_balboimper_ongoing_S3_projects }
	
	subtract_from_variable = { ITA_balboimper_misccosts_monthly = ITA_balboimper_S1S2_totalcosts_monthly_temp }
	subtract_from_variable = { ITA_balboimper_misccosts_monthly = ITA_balboimper_S3_totalcosts_monthly_temp }
	
	#Calculates cosmetic monthly costs (IF CHANGES IN BALANCING, THIS ALSO HAS TO BE CHANGED)
	
	multiply_variable = { ITA_balboimper_S1S2_costs_monthly = 1.3 }
	multiply_variable = { ITA_balboimper_S3_costs_monthly = 1.3 }
	
	#Adds new cosmetic project costs
	
	set_temp_variable = { ITA_balboimper_S1S2_totalcosts2_monthly_temp = ITA_balboimper_S1S2_costs_monthly }
	multiply_temp_variable = { ITA_balboimper_S1S2_totalcosts2_monthly_temp = ITA_balboimper_ongoing_S1S2_projects }
	
	set_temp_variable = { ITA_balboimper_S3_totalcosts2_monthly_temp = ITA_balboimper_S3_costs_monthly }
	multiply_temp_variable = { ITA_balboimper_S3_totalcosts2_monthly_temp = ITA_balboimper_ongoing_S3_projects }
	
	add_to_variable = { ITA_balboimper_misccosts_monthly = ITA_balboimper_S1S2_totalcosts2_monthly_temp }
	add_to_variable = { ITA_balboimper_misccosts_monthly = ITA_balboimper_S3_totalcosts2_monthly_temp }
}

ITA_balboimper_monthly_gain_S1_1_NI = {
	
	set_variable = { ITA_balboimper_progress_change_S1_1_NI = 8 } #Progress change this tick
	
	#Efficiency malus from corruption: used in scripted loc for (X from local corruption)
	set_variable = { ITA_balboimper_progress_CORR_S1_1_NI = ITA_balboimper_corruption_northern_italy }
	divide_variable = { ITA_balboimper_progress_CORR_S1_1_NI = 2 }
	
	set_variable = { ITA_balboimper_local_efficiency_S1_1_NI = ITA_balboimper_Efficiency } #efficiency minus corruption malus
	subtract_from_variable = { ITA_balboimper_local_efficiency_S1_1_NI = ITA_balboimper_progress_CORR_S1_1_NI }
	
	multiply_variable = { ITA_balboimper_progress_change_S1_1_NI = ITA_balboimper_local_efficiency_S1_1_NI } #Final progress change

	add_to_variable = { ITA_balboimper_progress_S1_1_NI = ITA_balboimper_progress_change_S1_1_NI } #Total progress ticker
	
	set_temp_variable = { ITA_balboimper_realvalue_TEMP = ITA_balboimper_progress_S1_1_NI } ##Temp variable for percentage calculation
	
	clamp_variable = {
		var = ITA_balboimper_progress_S1_1_NI
		min = 0
		max = 100
	}
	
	add_to_variable = { ITA_balboimper_months_S1_1_NI = 1 } #counts how many months the project has taken so far
	
	#The percentage calculation
	
	if = { #If the percentage would be over 100 fix
		limit = { check_variable = { ITA_balboimper_progress_S1_1_NI = 100 } }
	
		subtract_from_temp_variable = { ITA_balboimper_realvalue_TEMP = 100 }
		
		subtract_from_variable = { ITA_balboimper_progress_change_S1_1_NI = ITA_balboimper_realvalue_TEMP }
	}
	
	add_to_variable = { ITA_balboimper_TOTALprogress_S1 = ITA_balboimper_progress_change_S1_1_NI } #Total progress S1
	set_temp_variable = { ITA_balboimper_TOTALprogress_S1_TEMP = ITA_balboimper_TOTALprogress_S1 }
	
	divide_temp_variable = { ITA_balboimper_TOTALprogress_S1_TEMP = 12 } 
	
	set_variable = { ITA_balboimper_PERCENTAGE_S1 = ITA_balboimper_TOTALprogress_S1_TEMP }
}
