ITA_Ciano_POP_calculate_total_popolo_support = {
	set_variable = { ITA_total_popolo_support = 0 }
	set_temp_variable = { total_possible_support = 0 }
	
	for_each_loop = {
		array = ITA_Ciano_POP_Ciano_wierd_workaround
		index = i
		value = v
	
		if = {
			limit = {
				check_variable = { v < 51 }
			}
			set_temp_variable = { this_support = ITA_Ciano_POP_Loyalties^v }
			divide_temp_variable = { this_support = 100 }
			multiply_temp_variable = { this_support = ITA_Popolo_Support^i }
			add_to_variable = { ITA_total_popolo_support = this_support }
			log = "this_support: [?this_support]"

			set_temp_variable = { this_total_possible_support = 100 }
			subtract_from_temp_variable = { this_total_possible_support = ITA_Ciano_POP_Loyalties^ITA_Ciano_POP_Apathetic_wierd_workaround^i }
			divide_temp_variable = { this_total_possible_support = 100 }
			multiply_temp_variable = { this_total_possible_support = ITA_Popolo_Support^i }
			add_to_temp_variable = { total_possible_support = this_total_possible_support }
			log = "this_total_possible_support: [?this_total_possible_support]"
		}
	}

	divide_variable = { ITA_total_popolo_support = total_possible_support }
	add_to_variable = { ITA_Ciano_POP_dirty = 1 }
}

ITA_Ciano_POP_calculate_total_stato_support = {
	set_variable = { ITA_total_stato_support = 0 }
	set_temp_variable = { total_possible_support = 0 }
	
	for_each_loop = {
		array = ITA_Ciano_POP_Ciano_wierd_workaround
		index = i
		value = v
	
		if = {
			limit = {
				check_variable = { v > 51 }
			}
			set_temp_variable = { this_support = ITA_Ciano_POP_Loyalties^v }
			divide_temp_variable = { this_support = 100 }

			set_temp_variable = { adjusted_i = i }
			subtract_from_temp_variable = { adjusted_i = 6 }
			multiply_temp_variable = { this_support = ITA_Stato_Support^adjusted_i }
			add_to_variable = { ITA_total_stato_support = this_support }

			set_temp_variable = { this_total_possible_support = 100 }
			subtract_from_temp_variable = { this_total_possible_support = ITA_Ciano_POP_Loyalties^ITA_Ciano_POP_Apathetic_wierd_workaround^i }
			divide_temp_variable = { this_total_possible_support = 100 }
			multiply_temp_variable = { this_total_possible_support = ITA_Stato_Support^adjusted_i }
			add_to_temp_variable = { total_possible_support = this_total_possible_support }
		}
	}

	divide_variable = { ITA_total_stato_support = total_possible_support }
	add_to_variable = { ITA_Ciano_POP_dirty = 1 }
}

	add_to_temp_variable = { check_if_it_goes_above_100 = ITA_Ciano_POP_amount_to_add }
	if = {
		limit = {
			check_variable = { check_if_it_goes_above_100 > 100 }
		}
		subtract_from_temp_variable = { check_if_it_goes_above_100 = 100 }
		subtract_from_temp_variable = { ITA_Ciano_POP_amount_to_add = check_if_it_goes_above_100 }
	}
	else_if = {
		limit = {
			check_variable = { check_if_it_goes_above_100 < 0 }
		}
		subtract_from_temp_variable = { ITA_Ciano_POP_amount_to_add = check_if_it_goes_above_100 }
	}

	set_temp_variable = { everyone_else = 100 }
	subtract_from_temp_variable = { everyone_else = ITA_Ciano_POP_Loyalties^fella_to_add_to }

	resize_array = { array_with_4_elements = 4 }
	for_each_loop = {
		array = array_with_4_elements
		index = i

		set_temp_variable = { current_fella = ITA_Ciano_POP_pillar_to_add }
		multiply_temp_variable = { current_fella = 10 }
		add_to_temp_variable = { current_fella = i }

		if = {
			limit = {
				check_variable = { fella_to_add_to = current_fella }
			}
			add_to_variable = { ITA_Ciano_POP_Loyalties^current_fella = ITA_Ciano_POP_amount_to_add }
		}
		else = {
			set_temp_variable = { amount_to_subtract = ITA_Ciano_POP_Loyalties^current_fella }
			divide_temp_variable = { amount_to_subtract = everyone_else }
			multiply_temp_variable = { amount_to_subtract = ITA_Ciano_POP_amount_to_add }
			subtract_from_variable = { ITA_Ciano_POP_Loyalties^current_fella = amount_to_subtract }
		}

	}

	set_temp_variable = { adjusted_fascist = ITA_Ciano_POP_pillar_to_add }
	multiply_temp_variable = { adjusted_fascist = 10 }
	add_to_temp_variable = { adjusted_fascist = 4 }

	multiply_temp_variable = { unadjusted_fascist = 10 }
	add_to_temp_variable = { unadjusted_fascist = 1 }

	set_temp_variable = { ciano_for_adjustment = ITA_Ciano_POP_pillar_to_add }
	multiply_temp_variable = { ciano_for_adjustment = 10 }

	set_variable = { ITA_Ciano_POP_Loyalties^adjusted_fascist = 0 }
	add_to_variable = { ITA_Ciano_POP_Loyalties^adjusted_fascist = ITA_Ciano_POP_Loyalties^unadjusted_fascist }
	add_to_variable = { ITA_Ciano_POP_Loyalties^adjusted_fascist = ITA_Ciano_POP_Loyalties^ciano_for_adjustment }

		limit = {
			check_variable = { ITA_Ciano_POP_pillar_to_add < 6 }

		ITA_Ciano_POP_calculate_total_popolo_support = yes
	}
	else = {
		ITA_Ciano_POP_calculate_total_stato_support = yes
	}

	clear_variable = ITA_Ciano_POP_pillar_to_add

ITA_Ciano_POP_convert_to_verona = {

	clr_country_flag = Ciano_PoP_Active

	set_temp_variable = { ciano_pop = 0 }
	set_temp_variable = { fascist_pop = 0 }
	set_temp_variable = { popular_pop = 0 }

	for_each_loop = {
		#loop repeats 12 times, once for every pillar
		array = ITA_Ciano_POP_Radial_Progress_Bars_Array

		set_temp_variable = { this_pillar = i }
		multiply_temp_variable = { this_pillar = 10 }

		if = {
			limit = {
				check_variable = { i > 5 }
			}
			set_temp_variable = { new_i = i }
			subtract_from_temp_variable = { new_i = 6 }
			set_temp_variable = { starting_influence = ITA_Stato_Support^new_i }
		}
		else = {
			set_temp_variable = { starting_influence = ITA_Popolo_Support^i }
		}
		#starting_influence = the power of a pillar (the number written in the middle)

		set_temp_variable = { influence = starting_influence }
		multiply_temp_variable = { influence = ITA_Ciano_POP_Loyalties^this_pillar }
		add_to_temp_variable = { ciano_pop = influence }
		#ciano pop = power of every pillar multiplied by it's support for ciano 

		add_to_temp_variable = { this_pillar = 1 }
		set_temp_variable = { influence = starting_influence }
		multiply_temp_variable = { influence = ITA_Ciano_POP_Loyalties^this_pillar }
		add_to_temp_variable = { fascist_pop = influence }
		#ciano pop = power of every pillar multiplied by it's support for the fascists

		add_to_temp_variable = { this_pillar = 2 }
		set_temp_variable = { influence = starting_influence }
		multiply_temp_variable = { influence = ITA_Ciano_POP_Loyalties^this_pillar }
		add_to_temp_variable = { popular_pop = influence }
		#popular pop = power of every pillar multiplied by it's support for the popular opposition 
	}

	#decrease Ciano's support by radicalization ( multiply ciano_pop by (100-radicalization) )
	set_temp_variable = { adjusted_radicalization = 100 }
	subtract_from_temp_variable = { adjusted_radicalization = ITA_radicalization }
	divide_temp_variable = { adjusted_radicalization = 100 }
	multiply_temp_variable = { ciano_pop = adjusted_radicalization }

	#same as with radicalization except with the opposition supports
	set_temp_variable = { adjusted_opposition = ITA_opposition }
	add_to_temp_variable = { adjusted_opposition = 100 }
	divide_temp_variable = { adjusted_opposition = 100 }
	multiply_temp_variable = { fascist_pop = adjusted_opposition }
	multiply_temp_variable = { popular_pop = adjusted_opposition }

	#turn the ciano and fascist pops into percent
	set_temp_variable = { combined_pops = ciano_pop }
	add_to_temp_variable = { combined_pops = fascist_pop }
	divide_temp_variable = { combined_pops = 100 }

	#set the Verona vars
	set_variable = { ITA_Verona_Ciano_Percent = ciano_pop }
	divide_variable = { ITA_Verona_Ciano_Percent = combined_pops }
	set_variable = { ITA_Verona_Fascist_Opposition_Percent = fascist_pop }
	divide_variable = { ITA_Verona_Fascist_Opposition_Percent = combined_pops }
	
	round_variable = ITA_Verona_Ciano_Percent
	round_variable = ITA_Verona_Fascist_Opposition_Percent

	#make sure they add up to 100
	set_temp_variable = { 100_check = 100 }
	subtract_from_temp_variable = { 100_check = ITA_Verona_Ciano_Percent }
	subtract_from_temp_variable = { 100_check = ITA_Verona_Fascist_Opposition_Percent }

	add_to_variable = { ITA_Verona_Ciano_Percent = 100_check }

	#the outside pressure is set to popular support divided by the sum of ciano and fascist support 
	#outside pressure = popular / ( ciano + fascist )
	set_temp_variable = { ciano_fascist = ciano_pop }
	add_to_temp_variable = { ciano_fascist = fascist_pop }
	divide_temp_variable = { popular_pop = ciano_fascist }
	multiply_temp_variable = { popular_pop = 100 }
	round_temp_variable = popular_pop
	set_variable = { ITA_Verona_outside_pressure = popular_pop }

	divide_temp_variable = { popular_pop = 10 }
	set_variable = { ITA_Verona_win_margin = 50 }
	add_to_variable = { ITA_Verona_win_margin = popular_pop }
}

ITA_Ciano_POP_change_power = {
	#ITA_Ciano_POP_pillar_to_add
	#ITA_Ciano_POP_amount_to_add
	#All of these must be set as temp variables.
	if = {
		limit = {
			NOT = {
				has_variable = ITA_Ciano_POP_pillar_to_add
				has_variable = ITA_Ciano_POP_amount_to_add
			}
		}
		set_temp_variable = { ITA_Ciano_POP_pillar_to_add = 0 }
		set_temp_variable = { ITA_Ciano_POP_amount_to_add = 0 }
	}
	else = {
		set_temp_variable = { abs_change = ITA_Ciano_POP_amount_to_add }
		if = {
			limit = {
				check_variable = { abs_change < 0 }
			}
			multiply_temp_variable = { abs_change = -1 }
		}
		custom_effect_tooltip = ITA_Ciano_POP_change_power_tt
	}

	if = {
		limit = {
			check_variable = { ITA_Ciano_POP_pillar_to_add < 6 }
		}
		add_to_variable = { ITA_Popolo_Support^ITA_Ciano_POP_pillar_to_add = ITA_Ciano_POP_amount_to_add }
		clamp_variable = {
			var = ITA_Popolo_Support^ITA_Ciano_POP_pillar_to_add
			min = 3
		}
	}
	else = {
		subtract_from_temp_variable = { ITA_Ciano_POP_pillar_to_add = 6 }
		add_to_variable = { ITA_Stato_Support^ITA_Ciano_POP_pillar_to_add = ITA_Ciano_POP_amount_to_add }
		clamp_variable = {
			var = ITA_Stato_Support^ITA_Ciano_POP_pillar_to_add
			min = 3
		}
	}

	ITA_Ciano_POP_calculate_total_popolo_support = yes
	ITA_Ciano_POP_calculate_total_stato_support = yes

	add_to_variable = { ITA_Ciano_POP_dirty = 1 }
}
