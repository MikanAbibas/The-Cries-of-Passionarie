	set_variable = { ITA_giani_stability_pp = stability }

	multiply_variable = { ITA_giani_stability_pp = 0.5 }

	set_variable = { ITA_giani_stability_pop_growth = stability }
	add_to_variable = { ITA_giani_stability_pop_growth = -0.5 }
	multiply_variable = { ITA_giani_stability_pop_growth = 0.1 }

	set_variable = { ITA_giani_stability_factory_output = stability }
	add_to_variable = { ITA_giani_stability_factory_output = -0.5 }
	multiply_variable = { ITA_giani_stability_factory_output = 0.2 }

	set_variable = { ITA_giani_stability_construction_speed = stability }
	add_to_variable = { ITA_giani_stability_construction_speed = -0.5 }
	multiply_variable = { ITA_giani_stability_construction_speed = 0.2 }

	set_variable = { ITA_giani_stability_gdp_growth = stability }
	add_to_variable = { ITA_giani_stability_gdp_growth = -0.5 }
	multiply_variable = { ITA_giani_stability_gdp_growth = 100 } #gdp_growth_modifier is in literal procentages (1 = 1%)
	multiply_variable = { ITA_giani_stability_gdp_growth = 0.06 }
}
ITA_GUI_giani_calculate_stability_from_conversion_rate = {
	set_temp_variable = { result = ITA_GUI_giani_total_conversion }
	divide_temp_variable = { result = 100 }

	if = { #when conversion rate is really low, there's not much chaos
		limit = {
			check_variable = { result < 0.5 }
		}
		multiply_temp_variable = { result = -0.5 }
		add_to_temp_variable = { result = 0.1}
	}
	else = { #stability is the worst when conversion rate reaches 50%
		multiply_temp_variable = { result = 0.6 }
		add_to_temp_variable = { result = -0.45 }
	}

	set_variable = { ITA_giani_stability_from_conversion_rate = result }
}
ITA_GUI_giani_idea_check = {
	if = {
		limit = {
			OR = {
				check_variable = {ITA_GUI_giani_fervour < 30}
				check_variable = {ITA_GUI_giani_fervour > 80}
			}
			NOT = {
				has_idea = ITA_Giani_Angry_fanatics
			}
		}
		add_timed_idea = {
    		idea = ITA_Giani_Angry_fanatics
    		days = 60
		}
	}
}
ITA_Giani_change_fervour = {
	#fervour_change must be set as a temp_var when this is called

	add_to_variable = { ITA_GUI_giani_fervour = fervour_change }
	
	#ITA_Ciano_POP_amount_to_add is used here since I'm reusing old scripted loc
	if = {
		limit = {
			check_variable = { fervour_change < 0 }
		}
		set_temp_variable = { ITA_Ciano_POP_amount_to_add = -1 }
		multiply_temp_variable = { fervour_change = -1 }
	}
	else = {
		set_temp_variable = { ITA_Ciano_POP_amount_to_add = 1 }
	}

	custom_effect_tooltip = ITA_giani_gui_change_fervor_tt

}
