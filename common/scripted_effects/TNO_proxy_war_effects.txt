# Removes one of the old advisor_level ideas. Nothing grants those any more; the war-end events that
# call this only clean up saves from before the Military Advisors window. Content that withdraws
# advisors calls TNO_military_advisors_lower.
proxy_war_decrease_advisors = {
	if = {
		limit = {			
			has_idea = advisor_level_1
		}
			remove_ideas = advisor_level_1
		}					 
	else_if = {
		limit = {			
			has_idea = advisor_level_2
		}
		swap_ideas = {
			remove_idea = advisor_level_2
			add_idea = advisor_level_1
		}
	}
	else_if = {
		limit = {			
			has_idea = advisor_level_3
		}
		swap_ideas = {
			remove_idea = advisor_level_3
			add_idea = advisor_level_2
		}
	}
	else_if = {
		limit = {			
			has_idea = advisor_level_4
		}
		swap_ideas = {
			remove_idea = advisor_level_4
			add_idea = advisor_level_3
		}
	}
	else_if = {
		limit = {			
			has_idea = advisor_level_5
		}
		swap_ideas = {
			remove_idea = advisor_level_5
			add_idea = advisor_level_4
		}
	}
}
