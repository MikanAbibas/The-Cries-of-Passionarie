GER_reichstag_law_start_voting = {
	set_country_flag = GER_reichstag_vote_in_progress
	# set_variable = {
	# 	GER_majority_seats = GER_reichstag_seats
	# }
	# divide_variable = {
	# 	GER_majority_seats = 2
	# }
	# round_variable = GER_majority_seats
	set_variable = {
		GER_current_votes = 0
	}
	if = {
		limit = {
			OR = {
				GER_nsdap_government = yes
				has_country_flag = GER_reichstag_nsdap_in_coalition
			}
		}
		add_to_variable = {
			GER_current_votes = GER_nsdap_seats
		}
	}
	if = {
		limit = {
			OR = {
				GER_vup_government = yes
				has_country_flag = GER_reichstag_vup_in_coalition
			}
		}
		add_to_variable = {
			GER_current_votes = GER_vup_seats
		}
	}
	if = {
		limit = {
			OR = {
				GER_rsp_government = yes
				has_country_flag = GER_reichstag_rsp_in_coalition
			}
		}
		add_to_variable = {
			GER_current_votes = GER_rsp_seats
		}
	}
	if = {
		limit = {
			OR = {
				GER_vaaf_government = yes
				has_country_flag = GER_reichstag_vaaf_in_coalition
			}
		}
		add_to_variable = {
			GER_current_votes = GER_vaaf_seats
		}
	}
	if = {
		limit = {
			OR = {
				GER_dg_government = yes
				has_country_flag = GER_reichstag_dg_in_coalition
			}
		}
		add_to_variable = {
			GER_current_votes = GER_dg_seats
		}
	}
	if = {
		limit = {
			OR = {
				has_country_flag = GER_reichstag_dap_in_coalition
			}
		}
		add_to_variable = {
			GER_current_votes = GER_dap_seats
		}
	}
	if = {
		limit = {
			OR = {
				has_country_flag = GER_reichstag_ler_in_coalition
			}
		}
		add_to_variable = {
			GER_current_votes = GER_ler_seats
		}
	}
	# Auto-support: the party whose law tab is selected automatically votes in favor
	if = {
		limit = {
			check_variable = { GER_gui_party_laws_visible = 0 }
			NOT = { has_government = national_socialism }
			NOT = { has_government = fascism }
		}
		add_to_variable = {
			GER_current_votes = GER_nsdap_seats
		}
		set_country_flag = GER_reichstag_nsdap_already_negotiated
	}
	if = {
		limit = {
			check_variable = { GER_gui_party_laws_visible = 1 }
			NOT = { has_government = social_conservative }
			NOT = { has_government = social_democrat }
			NOT = { has_country_flag = GER_reichstag_vup_in_coalition }
		}
		add_to_variable = {
			GER_current_votes = GER_vup_seats
		}
		set_country_flag = GER_reichstag_vup_already_negotiated
	}
	if = {
		limit = {
			check_variable = { GER_gui_party_laws_visible = 2 }
			NOT = { has_government = authoritarian_democrat }
			NOT = { has_government = market_liberal }
		}
		add_to_variable = {
			GER_current_votes = GER_rsp_seats
		}
		set_country_flag = GER_reichstag_rsp_already_negotiated
	}
	if = {
		limit = {
			check_variable = { GER_gui_party_laws_visible = 3 }
			NOT = { has_government = socialist }
			NOT = { has_government = communist }
			NOT = { has_country_flag = GER_reichstag_vaaf_in_coalition }
		}
		add_to_variable = {
			GER_current_votes = GER_vaaf_seats
		}
		set_country_flag = GER_reichstag_vaaf_already_negotiated
	}
	if = {
		limit = {
			check_variable = { GER_gui_party_laws_visible = 4 }
			NOT = {
				has_government = social_liberal
			}
			NOT = {
				has_country_flag = GER_reichstag_dg_in_coalition
			}
		}
		add_to_variable = {
			GER_current_votes = GER_dg_seats
		}
		set_country_flag = GER_reichstag_dg_already_negotiated
	}
	# Voting window now opens via GER_reichstag_vote_in_progress flag
	# Player negotiates with remaining opposition parties before finalizing
}

	set_country_flag = GER_reichstag_nsdap_already_negotiated
	add_to_variable = { GER_current_votes = GER_nsdap_seats }
	set_temp_variable = { GER_temp_compromise_stab = -0.1 }
	multiply_temp_variable = { GER_temp_compromise_stab = GER_reichstag_compromise_cost_modifier }
	set_temp_variable = { GER_temp_compromise_pp = -0.1 }
	multiply_temp_variable = { GER_temp_compromise_pp = GER_reichstag_compromise_cost_modifier }
	# Store applied amounts for recovery event

	add_to_variable = { GER_reichstag_law_modifier_army_morale_factor = GER_temp_compromise_pp }

# Finalize vote - checks majority, passes or fails, cleans up
GER_reichstag_finalize_vote = {
	if = {
		limit = {
			check_variable = {
				var = GER_current_votes
				value = GER_reichstag_seats_majority
				compare = greater_than_or_equals
			}
		}
		# Law passes
		GER_pass_law_effects = yes
		custom_effect_tooltip = GER_reichstag_law_passed_tt
	}
	# else = {
	# 	# Law fails
	# 	custom_effect_tooltip = GER_reichstag_law_failed_tt
	# 	GER_reichstag_law_clear_flags = yes
	# }
	# Clean up
	set_variable = { GER_current_votes = 0 }
	clr_country_flag = GER_reichstag_vote_in_progress
	GER_reichstag_law_clear_vote_flags = yes
	# 60-day cooldown before next vote
	set_country_flag = { flag = GER_reichstag_vote_cooldown days = 60 value = 1 }
	# notify the player when the cooldown ends
	country_event = { id = GER_reichstag_voting.8 days = 61 }
}
