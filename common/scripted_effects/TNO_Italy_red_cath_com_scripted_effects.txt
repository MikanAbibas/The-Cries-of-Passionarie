ITA_RED_rodano_increase_vatican_opinion = {

}

ITA_RED_rodano_increase_middle_opinion = {
	set_temp_variable = { ITA_RED_rodano_opinions_before = ITA_RED_rodano_opinions }
	subtract_from_temp_variable = { ITA_RED_rodano_opinions_before = 50 }
	hidden_effect = {
		if = {
			limit = { check_variable = { ITA_RED_rodano_opinions < 50 } }
			add_to_temp_variable = { ITA_RED_rodano_opinions_before = ITA_RED_rodano_opinions_temp }
			if = {
				limit = { check_variable = { ITA_RED_rodano_opinions_before > 0 } }
				subtract_from_temp_variable = { ITA_RED_rodano_opinions_temp = ITA_RED_rodano_opinions_before }
			}
			ITA_RED_rodano_increase_pcdi_opinion = yes
		}
		else = {
			subtract_from_temp_variable = { ITA_RED_rodano_opinions_before = ITA_RED_rodano_opinions_temp }
			if = {
				limit = { check_variable = { ITA_RED_rodano_opinions_before < 0 } }
				subtract_from_temp_variable = { ITA_RED_rodano_opinions_temp = ITA_RED_rodano_opinions_before }
			}
			ITA_RED_rodano_increase_vatican_opinion = yes
		}
	}
	custom_effect_tooltip = ITA_RED_rodano_increase_middle_opinion_tt 
}

ITA_RED_rodano_math = {
	clamp_variable = { var = ITA_RED_rodano_opinions min = 0 max = 100 }
	set_variable = { ITA_RED_rodano_opinions_dynmod = ITA_RED_rodano_opinions }
	subtract_from_variable = { ITA_RED_rodano_opinions_dynmod = 50 }
	set_variable = { ITA_RED_rodano_opinions_absolute_value = ITA_RED_rodano_opinions_dynmod }
	if = {
		limit = { check_variable = { ITA_RED_rodano_opinions_absolute_value < 0 } }
		multiply_variable = { ITA_RED_rodano_opinions_absolute_value = -1 }
	}
	
	# Centrist Math

	set_variable = { ITA_RED_rodano_opinions_dynmod_center = 25 }
	if = {
		limit = { check_variable = { ITA_RED_rodano_opinions_dynmod > 0 } } #pcdi leaning
		subtract_from_variable = { ITA_RED_rodano_opinions_dynmod_center = ITA_RED_rodano_opinions_dynmod }
		else = { # vatican leaning
			add_to_variable = { ITA_RED_rodano_opinions_dynmod_center = ITA_RED_rodano_opinions_dynmod }
		}	
	}
	
	set_variable = { ITA_RED_rodano_opinions_dynmod_center_culrev_negative = ITA_RED_rodano_opinions_dynmod_center }
	multiply_variable = { ITA_RED_rodano_opinions_dynmod_center_culrev_negative = -0.1 }
	
	divide_variable = { ITA_RED_rodano_opinions_dynmod_center = 100 }
	
	# CultRev shifts

	divide_variable = { ITA_RED_rodano_opinions_dynmod = 10 }
	set_variable = { ITA_RED_rodano_opinions_dynmod_percent = ITA_RED_rodano_opinions_dynmod }
	
	# For stability and percentage effects
	
	divide_variable = { ITA_RED_rodano_opinions_dynmod_percent = -12.5 }
	set_variable = { ITA_RED_rodano_opinions_dynmod_percent_negative = ITA_RED_rodano_opinions_dynmod_percent }
	multiply_variable = { ITA_RED_rodano_opinions_dynmod_percent_negative = -1 }
	set_variable = { ITA_RED_rodano_opinions_dynmod_percent_gdp = ITA_RED_rodano_opinions_dynmod_percent }
	multiply_variable = { ITA_RED_rodano_opinions_dynmod_percent_gdp = 20 } 
}
