#The way that influence changes work is that when the influence of one faction rises, the influence of all the other factions decrease so they all add up to a hundred and same with decrease.
#This is done by increasing the popularyty of the party and then scaling down all the popularities of a wing so that they add up to 1 again.
#At the end there is an extra step where a random party is given the amount of popularyty required for the sum to be 1.
#Note: the pupularites are stored as numbers form 0-1 not 0-100

ITA_change_party_pop = {
	divide_temp_variable = { constant = 100 }

	#increase/ decrease the popularity
	add_to_variable = { ITA_Party_pop^changer = constant }
	clamp_variable = {
		var = ITA_Party_pop^changer
		min = 0
		max = 1
	}

	#loop throuhg the 3 wings and make sure the sum of pops for each of them is 1
	for_loop_effect = {
		start = 0
		end = 3
		value = wing_i

		#set party_index to the first party in the current wing
		set_temp_variable = { party_index = wing_i }
		multiply_temp_variable = { party_index = 5 }
		set_temp_variable = { pops_sum = 0 } #sum of the popularites of this wing

		for_loop_effect = { #calculate pops_sum
			start = 0
			end = 5

			add_to_temp_variable = { pops_sum = ITA_Party_pop^party_index }
			add_to_temp_variable = { party_index = 1 }
		}

		#calculate the value that the popularities need to be multiplied with so that they add up to 1
		#this would in theory be perfectly accurate but when you're working with 3 decimal places of precision it isn't
		set_temp_variable = { adjust_var_multiplication = 1 }
		divide_temp_variable = { adjust_var_multiplication = pops_sum }
		
		#multiply every party in this wing's pop with adjust_var_multiplication
		set_temp_variable = { party_index = wing_i }
		multiply_temp_variable = { party_index = 5 }
		for_loop_effect = {
			start = 0
			end = 5

			multiply_variable = { ITA_Party_pop^party_index = adjust_var_multiplication }
			clamp_variable = {
				var = ITA_Party_pop^party_index
				min = 0
				max = 1
			}
			add_to_temp_variable = { party_index = 1 }
		}

		#reset pops_sum and party_index for another calculation
		set_temp_variable = { party_index = wing_i }
		multiply_temp_variable = { party_index = 5 }
		set_temp_variable = { pops_sum = 0 }

		for_loop_effect = { #calculate pops_sum again
			start = 0
			end = 5

			add_to_temp_variable = { pops_sum = ITA_Party_pop^party_index }
			add_to_temp_variable = { party_index = 1 }
		}

		#calulate the number that need to be added subtracted to a party so that all parties together add up to 1
		#thsi is here so I don't get 10 gitlab issues all saying that the popularities actually add up to 98.7% and not 100%
		set_temp_variable = { adjust_var_addition = 1 }
		subtract_from_temp_variable = { adjust_var_addition = pops_sum }
		randomize_temp_variable = {
			var = bonus_pop
			distribution = uniform
			min = 0
			max = 4
		}
		round_temp_variable = bonus_pop
		#give a random party the popularity
		#a potential issue here is adjust_var_addition being negative and the party not having less popularity that that,
		#but everything functions normaly even if the pops don't add up to exactly 100% and the whole thins is gonna get rounded next time ITA_change_party_pop is called anyway
		set_temp_variable = { bonus_pop_with_wing = wing_i }

ITA_dem_update_party_pop_wheel = {
	#Updates the party pop wheel to follow the popularities in the Camera mech

	#an array with the index being the ideology that should be added
	resize_temp_array = { pops = 8 }

	#socialism
	set_temp_variable = { pops^0 = ITA_camera_current_deputies^0 } #Berlinguer
	add_to_temp_variable = { pops^0 = ITA_camera_current_deputies^1 } #Nenni

	#social_democrat
	set_temp_variable = { pops^1 = ITA_camera_current_deputies^2 } #Panella
	add_to_temp_variable = { pops^1 = ITA_camera_current_deputies^3 } #Lombardi
	add_to_temp_variable = { pops^1 = ITA_camera_current_deputies^4 } #Saragat

	#social_liberal
	set_temp_variable = { pops^2 = ITA_camera_current_deputies^5 } #La Malfa
	add_to_temp_variable = { pops^2 = ITA_camera_current_deputies^6 } #Donat-Cattin
	add_to_temp_variable = { pops^2 = ITA_camera_current_deputies^7 } #Fanfani

	#lib con
	set_temp_variable = { pops^3 = ITA_camera_current_deputies^8 } #Moro

	#social_conservative
	add_to_temp_variable = { pops^4 = ITA_camera_current_deputies^9 } #Andreotti
	set_temp_variable = { pops^4 = ITA_camera_current_deputies^10 } #Malagodi

	#authoritarian_democrat
	add_to_temp_variable = { pops^5 = ITA_camera_current_deputies^11 } #Lauro
	add_to_temp_variable = { pops^5 = ITA_camera_current_deputies^12 } #De Marzio

	#fascism
	set_temp_variable = { pops^6 = ITA_camera_current_deputies^13 } #Almirante

	#eso naz
	set_temp_variable = { pops^7 = ITA_camera_current_deputies^14 } #Rauti


	#we now have to convert the seat numbers taken from camera to procentages
	#they HAVE to add up to 100 or the set_popularities effect won't take them

	#calculate their total sum and multiply each of them with 100/sum
	set_temp_variable = { sum_pops = 0 }
	for_each_loop = {
		array = pops

		add_to_temp_variable = { sum_pops = v }
	}
	set_temp_variable = { adjustment_var = 100 }
	divide_temp_variable = { adjustment_var = sum_pops }

	for_loop_effect = {
		start = 0
		end = 8

		multiply_temp_variable = { pops^v = adjustment_var }
	}

	round_temp_variable = pops^0
	round_temp_variable = pops^1
	round_temp_variable = pops^2
	round_temp_variable = pops^3
	round_temp_variable = pops^4
	round_temp_variable = pops^5
	round_temp_variable = pops^6
	round_temp_variable = pops^7

	#Check if they add up to 100
	set_temp_variable = { hundered_check = 100 }
	for_each_loop = {
		array = pops

		subtract_from_temp_variable = { hundered_check = v }
	}
	if = {
		#they add up to less than 100
		limit = {
			check_variable = { hundered_check > 0 }
		}
		#give away procentages until they add up to 100
		for_loop_effect = {
			start = 1
			end = 100
			break = break

			randomize_temp_variable = {
				var = pop_to_add
				distribution = uniform
				min = 0
				max = 7
			}
			round_temp_variable = pop_to_add
			add_to_temp_variable = { pops^pop_to_add = 1 }

			if = {
				limit = {
					check_variable = { v = hundered_check }
				}
				set_temp_variable = { break = 1 }
			}
		}
	}
	else_if = {
		#they add up to more than 100
		limit = {
			check_variable = { hundered_check < 0 }
		}
		#take away procentages untill they add up to 100
		for_loop_effect = {
			start = 1
			end = 100
			break = break

			randomize_temp_variable = {
				var = pop_to_subtract
				distribution = uniform
				min = 0
				max = 6
			}
			round_temp_variable = pop_to_subtract
			if = {
				limit = {
					check_variable = { pops^pop_to_subtract > 1 }
				}
				subtract_from_temp_variable = { pops^pop_to_subtract = 1 }
				add_to_temp_variable = { hundered_check = 1 }
			}

			if = {
				limit = {
					OR = {
						check_variable = { hundered_check = 0 }
						check_variable = { hundered_check > 0 }
					}
				}
				set_temp_variable = { break = 1 }
			}
		}
	}

	#finnaly set the popularities
	hidden_effect = {
		set_popularities = {
			communist = 0 #
			socialist = var:pops^0 #PSI-B, PSI-Max
			social_democrat = var:pops^1 #PR, PSI-Min, PSDI
			social_liberal = var:pops^2 #PRI, DC-RD, DC-NC
			market_liberal = var:pops^3 #DC-D
			social_conservative = var:pops^4 #DC-P, PLI
			authoritarian_democrat = var:pops^5 #PDI, MSI-DN
			fascism = var:pops^6 #MSI-A
			national_socialism = 0
			burgundian_system = var:pops^7 #MSI-ON
		}
	}
}
