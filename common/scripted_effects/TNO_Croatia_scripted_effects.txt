# Croatia — Government Control mechanic (Phase 1)
# CRO_gov_control: 0-100. High = the Italian-backed government holds. Low = collapse.
# Boosted by Italian influence + player focuses/decisions; decays via a repeating mission.
# Below the coup threshold (25) the government falls and the civil war begins (cro.coup, Phase 3).

CRO_gov_control_recalc = {
	clamp_variable = { var = CRO_gov_control min = 0 max = 100 }
	# Mirror to a GLOBAL display var so both CRO and ITA can read the same bar (KN-bar pattern).
	set_variable = { global.CRO_gov_control_display = CRO_gov_control }
	CRO_sync_italian_power = yes

	# Coup trigger: government control collapses.
	if = {
		limit = {
			check_variable = { CRO_gov_control < 25 }
			NOT = { has_country_flag = CRO_couped }
			NOT = { has_country_flag = CRO_path_resolved }
		}
		country_event = { id = cro.100 days = 0 }
	}
}

# OOSG-style: caller sets the temp var, this applies it and shows the exact change in the tooltip.
# Usage:  set_temp_variable = { CRO_gov_control_change = 10 }  then  CRO_apply_gov_control_change = yes
CRO_apply_gov_control_change = {
	custom_effect_tooltip = CRO_gov_control_change_tt
	hidden_effect = {
		add_to_variable = { CRO_gov_control = CRO_gov_control_change }
		CRO_gov_control_recalc = yes
	}
}

# Italian Power is a 0-1 projection of ITA's own influence score for Croatia,
# ITA.CRO_ITA_Influence (0-500 on the Italian Influence GUI; 1962 start = 300 -> 0.6).
# The GUI score is the single source of truth: this only reads it.
CRO_sync_italian_power = {
	if = {
		limit = { ITA = { exists = yes } }
		set_variable = { CRO_italian_power = ITA.CRO_ITA_Influence }
		divide_variable = { CRO_italian_power = 500 }
		clamp_variable = { var = CRO_italian_power min = 0 max = 1 }
	}
	else = {
		set_variable = { CRO_italian_power = 0 }
	}
	set_variable = { global.CRO_italian_power_display = CRO_italian_power }
}

# Usage:  set_temp_variable = { CRO_italian_power_change = 0.1 }  then  CRO_apply_italian_power_change = yes
# Writes back to ITA's influence score (the source), then re-derives the 0-1 value.
CRO_apply_italian_power_change = {
	custom_effect_tooltip = CRO_italian_power_change_tt
	hidden_effect = {
		set_temp_variable = { CRO_ita_influence_delta = CRO_italian_power_change }
		multiply_temp_variable = { CRO_ita_influence_delta = 500 }
		ITA = {
			add_to_variable = { CRO_ITA_Influence = CRO_ita_influence_delta }
			clamp_variable = { var = CRO_ITA_Influence min = 0 max = 500 }
		}
		CRO_sync_italian_power = yes
	}
}

			CRO_Edvard_Kardelj_for
		}

		if = {
			limit = { CRO = { has_country_flag = CRO_partisans_ready } }
			add_equipment_to_stockpile = { type = infantry_equipment amount = 5000 producer = BOS }
			add_manpower = 25000
			add_war_support = 0.15

		country_event = { id = cro.105 }

	set_politics = { ruling_party = national_socialism }

		national_socialism = 45

	create_country_leader = {
		name = "Jure Francetić"
		desc = "POLITICS_Jure_Francetic_DESC"
		picture = "Portrait_Croatia_Jure_Francetic_DDS.dds"

	# so end_puppet must run before leave_faction / declare_war_on below.
	ITA = { end_puppet = CRO }
	leave_faction = yes
	set_country_flag = ustasevictory
	# Ustase groundwork (final intermission focus) makes the seizure far stronger.
	if = {
		limit = { has_country_flag = CRO_ustase_ready }
		add_equipment_to_stockpile = { type = infantry_equipment amount = 5000 producer = CRO }
		add_manpower = 25000
		add_war_support = 0.15
		army_experience = 50
	}

	# The civil war: Ustase Croatia vs Tito's Bosnia.
	declare_war_on = { target = BOS type = annex_everything }
	# Italy intervenes against both breakaways after a delay (Phase 5 handles the survival window).
	ITA = { country_event = { id = cro.103 days = 260 } }
}

# ===================== Phase 6: Titotangle (clone of WRRFTangle) =====================
# Three communist factions compete for leadership of the new Yugoslavia.
# Corners:  bottom-left = Tito (A), bottom-right = Hebrang (B), top = Bakaric (C).
CROTitoTangle_calcXY = {
	clamp_variable = { var = CRO_tito_balance min = 0 max = 95 }
	clamp_variable = { var = CRO_hebrang_balance min = 0 max = 95 }
	clamp_variable = { var = CRO_bakaric_balance min = 0 max = 95 }

	set_temp_variable = { A = CRO_tito_balance }
	set_temp_variable = { B = CRO_hebrang_balance }
	set_temp_variable = { C = CRO_bakaric_balance }
	set_temp_variable = { ABC = A }
	add_to_temp_variable = { ABC = B }
	add_to_temp_variable = { ABC = C }

	# X = 1/2 * ((2B + C) / (A+B+C))
	set_variable = { CROTitoTangle_X = B }
	multiply_variable = { CROTitoTangle_X = 2 }
	add_to_variable = { CROTitoTangle_X = C }
	divide_variable = { CROTitoTangle_X = ABC }
	multiply_variable = { CROTitoTangle_X = 0.5 }

	# Y = sqrt(3)/2 * (C / (A+B+C))
	set_variable = { CROTitoTangle_Y = C }
	divide_variable = { CROTitoTangle_Y = ABC }
	multiply_variable = { CROTitoTangle_Y = 0.866 }

	# Scale to the GUI triangle (matches the WRRF outline sprite extents).
	multiply_variable = { CROTitoTangle_X = 496 }
	multiply_variable = { CROTitoTangle_Y = 392 }
	multiply_variable = { CROTitoTangle_Y = -1 }
	add_to_variable = { CROTitoTangle_Y = 392 }
}

# Start the leadership struggle (called from cro.110 once Yugoslavia outlasts Italy).
CRO_start_titotangle = {
	set_country_flag = CRO_revolution_active
	# Tito starts ahead as the victor of the civil war; the others contend.
	set_variable = { CRO_tito_balance = 40 }
	set_variable = { CRO_hebrang_balance = 30 }
	set_variable = { CRO_bakaric_balance = 30 }
	CROTitoTangle_calcXY = yes
}

# Resolve the struggle to whichever faction holds the highest balance.
CRO_resolve_titotangle = {
	clr_country_flag = CRO_revolution_active
	if = {
		limit = {
			check_variable = { var = CRO_tito_balance value = CRO_hebrang_balance compare = greater_than_or_equals }
			check_variable = { var = CRO_tito_balance value = CRO_bakaric_balance compare = greater_than_or_equals }
		}
		set_country_flag = CRO_tito_leader
	}
	else_if = {
		limit = { check_variable = { var = CRO_hebrang_balance value = CRO_bakaric_balance compare = greater_than_or_equals } }
		set_country_flag = CRO_hebrang_leader
	}
	else = {
		set_country_flag = CRO_bakaric_leader
	}
}

# Display math + dominant flags + popularity sync (GAY_candidate_math clone).

	hidden_effect = {
		# Stacked-bar display vars (cumulative; window draws widest-first).
		set_variable = { CRO_francetic_power_display = CRO_francetic_power }
		set_variable = { CRO_lorkovic_power_display = CRO_francetic_power_display }
		add_to_variable = { CRO_lorkovic_power_display = CRO_lorkovic_power }
		set_variable = { CRO_luburic_power_display = CRO_lorkovic_power_display }
		add_to_variable = { CRO_luburic_power_display = CRO_luburic_power }
		clamp_variable = { var = CRO_francetic_power min = 0 max = 100 }
		clamp_variable = { var = CRO_lorkovic_power min = 0 max = 100 }
		clamp_variable = { var = CRO_luburic_power min = 0 max = 100 }
		clamp_variable = { var = CRO_francetic_power_display min = 0 max = 100 }
		clamp_variable = { var = CRO_lorkovic_power_display min = 0 max = 100 }
		clamp_variable = { var = CRO_luburic_power_display min = 0 max = 100 }
		clr_country_flag = CRO_francetic_dominant
		clr_country_flag = CRO_lorkovic_dominant
		clr_country_flag = CRO_luburic_dominant
		if = {
			limit = { check_variable = { CRO_francetic_power > 50 } }
			set_country_flag = CRO_francetic_dominant
		}
		else_if = {
			limit = { check_variable = { CRO_lorkovic_power > 50 } }
			set_country_flag = CRO_lorkovic_dominant
		}
		else_if = {
			limit = { check_variable = { CRO_luburic_power > 50 } }
			set_country_flag = CRO_luburic_dominant
		}
		CRO_ideologies_var = yes
	}
}

# Popularity mapping (GAY_ideologies_var clone, adapted): the ruling Ustase spectrum
# is a 60% pool split by candidate power; HSS (authdem) 25 + communists 15 stay fixed.
# Normalized by the power sum so _flat adds (which inflate the sum) can't break the total.

			set_variable = { CRO_ns_var = CRO_francetic_power }
			divide_variable = { CRO_ns_var = CRO_power_sum }
			multiply_variable = { CRO_ns_var = 60 }
			set_variable = { CRO_fa_var = CRO_lorkovic_power }
			divide_variable = { CRO_fa_var = CRO_power_sum }
			multiply_variable = { CRO_fa_var = 60 }
			set_variable = { CRO_bs_var = CRO_luburic_power }
			divide_variable = { CRO_bs_var = CRO_power_sum }
			multiply_variable = { CRO_bs_var = 60 }

				national_socialism = CRO_ns_var
				fascism = CRO_fa_var
				burgundian_system = CRO_bs_var

				communist = 15

	}
}

# Resolve the struggle (GAY_sort_candidate clone). Highest power wins; tie -> incumbent.
CRO_sort_leader = {
	if = {
		limit = {
			check_variable = { CRO_lorkovic_power > CRO_luburic_power }
			check_variable = { CRO_lorkovic_power > CRO_francetic_power }
		}
		CRO_lorkovic_wins = yes
	}
	else_if = {
		limit = {
			check_variable = { CRO_francetic_power > CRO_luburic_power }
			check_variable = { CRO_francetic_power > CRO_lorkovic_power }
		}
		CRO_francetic_wins = yes
	}
	else_if = {
		limit = {
			check_variable = { CRO_luburic_power > CRO_francetic_power }
