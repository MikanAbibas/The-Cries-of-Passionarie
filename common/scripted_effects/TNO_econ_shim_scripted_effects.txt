### Economy-library bridges.
###
### Upstream TNO ships an economy library TNORA never forked, and the imported
### content calls into it constantly. These definitions stop every such call
### throwing "Unknown effect-type" and silently doing nothing.
###
### They are no longer accumulators. Where TNORA has the variable the upstream
### effect meant - money_reserves, GDP, state_value - the call now moves the real
### one. Where it does not, the call is redirected to the nearest thing TNORA
### does model, which is a societal-development or law ladder, chosen by the sign
### of the magnitude the author wrote. Nothing banks into a variable no one reads.
###
### NOT defined here, because master already defines the real effect:
###   econ_gdp_growth_change, econ_flat_national_debt_change, econ_misc_costs_change,
###   econ_misc_income_change, econ_interest_rates_change
### and these are scripted TRIGGERS in TNO_useful_scripted_triggers.txt, not effects:
###   econ_can_use_economy_system, econ_can_add_modifier_military/_civilian

### --- treasury: money_reserves is real ---------------------------------------
econ_spend_money_once_effect_raw_money = {
	subtract_from_variable = { money_reserves = temp_econ_spending_amount }
	custom_effect_tooltip = TNO_econ_shim_spend_money_tt
}

econ_money_reserves_change_raw_money = {
	add_to_variable = { money_reserves = money_reserves_temp }
	custom_effect_tooltip = TNO_econ_shim_money_reserves_tt
}

econ_money_reserves_change_percent_gdp = {
	set_temp_variable = { TNO_econ_bridge_temp = GDP }
	multiply_temp_variable = { TNO_econ_bridge_temp = temp_gdp_gained }
	add_to_variable = { money_reserves = TNO_econ_bridge_temp }
	custom_effect_tooltip = TNO_econ_shim_money_reserves_tt
}

### --- one-off GDP moves: GDP is real ----------------------------------------
econ_GDP_boost_once_flat = {
	add_to_variable = { GDP = temp_econ_GDP_boost_flat }
	custom_effect_tooltip = TNO_econ_shim_gdp_boost_tt
}

econ_GDP_boost_once_percentage = {
	set_temp_variable = { TNO_econ_bridge_temp = temp_econ_GDP_boost_percentage }
	add_to_temp_variable = { TNO_econ_bridge_temp = 1 }
	multiply_variable = { GDP = TNO_econ_bridge_temp }
	custom_effect_tooltip = TNO_econ_shim_gdp_boost_tt
}

### --- centralisation: the nearest TNORA idea is administrative capacity -----
econ_centralization_change = {
	if = {
		limit = { check_variable = { econ_centralization_temp > 0 } }
		TNO_improve_academic_base_low = yes
	}
	else_if = {
		limit = { check_variable = { econ_centralization_temp < 0 } }
		TNO_worsen_academic_base_low = yes
	}
}
