# total seats of held states -> RAJ_LS_house_size, majority -> RAJ_LS_majority
RAJ_LS_recount_house = {
	set_variable = {
		RAJ_LS_house_size = 0
	}
	for_loop_effect = {
		value = i
		start = 0
		end = RAJ_LS_state_count
		if = {
			limit = {
				check_variable = {
					RAJ_LS_held^i > 0
				}
			}
			add_to_variable = {
				RAJ_LS_house_size = RAJ_LS_seats^i
			}
		}
	}
	set_variable = {
		RAJ_LS_majority = RAJ_LS_house_size
	}
	divide_variable = {
		RAJ_LS_majority = 2
	}
	round_variable = RAJ_LS_majority
	add_to_variable = {
		RAJ_LS_majority = 1
	}
}

# ---------- support changes (OOSG display pattern) ----------
# caller sets:
#   RAJ_LS_t_party (temp, slot 0-5)
#   RAJ_LS_t_state (temp, index; -1 = all held states)
#   RAJ_LS_t_amount (temp, +/- support points)
# then: RAJ_LS_add_support = yes
# BJS (1) and CPI (2) gains are halved and capped at 40 per state:
# their road to power is not the ballot box (content pending).
RAJ_LS_add_support = {
	if = {
		limit = {
			check_variable = {
				RAJ_LS_t_amount > 0
			}
			OR = {
				check_variable = {
					RAJ_LS_t_party = 1
				}
				check_variable = {
					RAJ_LS_t_party = 2
				}
			}
		}
		divide_temp_variable = {
			RAJ_LS_t_amount = 2
		}
	}
	if = {
		limit = {
			check_variable = {
				RAJ_LS_t_state = -1
			}
		}
		for_loop_effect = {
			value = i
			start = 0
			end = RAJ_LS_state_count
			set_temp_variable = {
				RAJ_LS_t_i = i
			}
			RAJ_LS_apply_support_one = yes
		}
	}
	else = {
		set_temp_variable = {
			RAJ_LS_t_i = RAJ_LS_t_state
		}
		RAJ_LS_apply_support_one = yes
	}
	custom_effect_tooltip = RAJ_LS_add_support_tt
}

# ---------- INC collapse (call from demindia.110) ----------
# splits the umbrella: INC keeps ~55%, LibSoc 25%, Conservative 15%, Liberal 5%
RAJ_LS_inc_split = {
	for_loop_effect = {
		value = i
		start = 0
		end = RAJ_LS_state_count
		set_temp_variable = {
			ls_base = RAJ_LS_sup_0^i
		}
		# PSP 10%
		set_temp_variable = {
			ls_part = ls_base
		}
		multiply_temp_variable = {
			ls_part = 0.1
		}
		round_temp_variable = ls_part
		add_to_variable = {
			RAJ_LS_sup_5^i = ls_part
		}
		subtract_from_variable = {
			RAJ_LS_sup_0^i = ls_part
		}
		# Janata 24%
		set_temp_variable = {
			ls_part = ls_base
		}
		multiply_temp_variable = {
			ls_part = 0.24
		}
		round_temp_variable = ls_part
		add_to_variable = {
			RAJ_LS_sup_4^i = ls_part
		}
		subtract_from_variable = {
			RAJ_LS_sup_0^i = ls_part
		}
		# Swatantra 16%
		set_temp_variable = {
			ls_part = ls_base
		}
		multiply_temp_variable = {
			ls_part = 0.16
		}
		round_temp_variable = ls_part
		add_to_variable = {
			RAJ_LS_sup_3^i = ls_part
		}
		subtract_from_variable = {
			RAJ_LS_sup_0^i = ls_part
		}
	}
	set_variable = {
		RAJ_LS_party_active^3 = 1
	}
	set_variable = {
		RAJ_LS_party_active^4 = 1
	}
	set_variable = {
		RAJ_LS_party_active^5 = 1
	}
	# sitting seats redistribute the same way (MPs walk out)
	set_temp_variable = {
		ls_base = RAJ_LS_party_seats^0
	}
	set_temp_variable = {
		ls_part = ls_base
	}
	multiply_temp_variable = {
		ls_part = 0.1
	}
	round_temp_variable = ls_part
	add_to_variable = {
		RAJ_LS_party_seats^5 = ls_part
	}
	subtract_from_variable = {
		RAJ_LS_party_seats^0 = ls_part
	}
	set_temp_variable = {
		ls_part = ls_base
	}
	multiply_temp_variable = {
		ls_part = 0.24
	}
	round_temp_variable = ls_part
	add_to_variable = {
		RAJ_LS_party_seats^4 = ls_part
	}
	subtract_from_variable = {
		RAJ_LS_party_seats^0 = ls_part
	}
	set_temp_variable = {
		ls_part = ls_base
	}
	multiply_temp_variable = {
		ls_part = 0.16
	}
	round_temp_variable = ls_part
	add_to_variable = {
		RAJ_LS_party_seats^3 = ls_part
	}
	subtract_from_variable = {
		RAJ_LS_party_seats^0 = ls_part
	}
	custom_effect_tooltip = RAJ_LS_inc_split_tt
}

# ---------- elections ----------
# projections: seats per state shared by support^2 (FPTP sweep emerges
# naturally: 45% vs 10% support = 20:1 weight)
RAJ_LS_compute_projections = {
	for_loop_effect = {
		value = i
		start = 0
		end = RAJ_LS_party_count
		set_variable = {
			RAJ_LS_proj^i = 0
		}
	}
	for_loop_effect = {
		value = i
		start = 0
		end = RAJ_LS_state_count
		if = {
			limit = {
				check_variable = {
					RAJ_LS_held^i > 0
				}
			}
			set_temp_variable = {
				ls_i = i
			}
			RAJ_LS_project_state = yes
		}
	}
}

# call at election-period end: final tally, award seats, pick winner
RAJ_LS_finish_election = {
	RAJ_LS_compute_projections = yes
	for_loop_effect = {
		value = i
		start = 0
		end = RAJ_LS_party_count
		set_variable = {
			RAJ_LS_party_seats^i = RAJ_LS_proj^i
		}
	}
	# winner = largest party
	set_variable = {
		RAJ_LS_winner = 0
	}
	set_variable = {
		ls_best = RAJ_LS_party_seats^0
	}
	for_loop_effect = {
		value = i
		start = 1
		end = RAJ_LS_party_count
		if = {
			limit = {
				check_variable = {
					RAJ_LS_party_seats^i > ls_best
				}
			}
			set_variable = {
				ls_best = RAJ_LS_party_seats^i
			}
			set_variable = {
				RAJ_LS_winner = i
			}
		}
	}
	set_variable = {
		RAJ_LS_ruling = RAJ_LS_winner
	}
	# fresh mandate: coalition resets
	for_loop_effect = {
		value = i
		start = 0
		end = RAJ_LS_party_count
		set_variable = {
			RAJ_LS_coalition^i = 0
		}
	}
	clr_country_flag = RAJ_LS_election_active
	custom_effect_tooltip = RAJ_LS_finish_election_tt
}

# ---------- bills ----------
# Start from a focus/decision:
#   1. set_country_flag = RAJ_LS_bill_<yourbill>   (your payoff dispatch key)
#   2. RAJ_LS_start_bill = yes
#   3. activate_mission = RAJ_LS_bill_vote_mission
# Pass: mission completes -> your on-pass handling keyed off your flag.
# Fail: mission times out -> RAJ_LS_bill_failed fires + support malus.
# AI never runs the mission: give AI paths an instant stability roll instead.
RAJ_LS_start_bill = {
	set_country_flag = RAJ_LS_bill_active
	RAJ_LS_recount_house = yes
	# pledged = ruling+coalition seats scaled by stability (65-95%) + jitter
	set_variable = {
		RAJ_LS_bill_base = 0
	}
	for_loop_effect = {
		value = i
		start = 0
		end = RAJ_LS_party_count
		if = {
			limit = {
				OR = {
					check_variable = {
						RAJ_LS_ruling = i
					}
					check_variable = {
						RAJ_LS_coalition^i > 0
					}
				}
			}
			add_to_variable = {
				RAJ_LS_bill_base = RAJ_LS_party_seats^i
			}
		}
	}
	set_temp_variable = {
		ls_rate = stability
	}
	multiply_temp_variable = {
		ls_rate = 0.3
	}
	add_to_temp_variable = {
		ls_rate = 0.65
	}
	random_list = {
		25 = {
			add_to_temp_variable = {
				ls_rate = -0.04
			}
		}
		50 = {
		}
		25 = {
			add_to_temp_variable = {
				ls_rate = 0.04
			}
		}
	}
	clamp_temp_variable = {
		var = ls_rate
		min = 0.5
		max = 0.98
	}
	set_variable = {
		RAJ_LS_bill_pledged = RAJ_LS_bill_base
	}
	multiply_variable = {
		RAJ_LS_bill_pledged = ls_rate
	}
	round_variable = RAJ_LS_bill_pledged
	# track per-party wooing exhaustion
	for_loop_effect = {
		value = i
		start = 0
		end = RAJ_LS_party_count
		set_variable = {
			RAJ_LS_wooed^i = 0
		}
	}
	custom_effect_tooltip = RAJ_LS_start_bill_tt
}

# woo a party: caller sets RAJ_LS_t_party (temp).
# grants 30% of the party's not-yet-pledged seats (own/coalition parties
# start mostly pledged already, so wooing them yields the remainder cheap)
RAJ_LS_woo_party = {
	set_temp_variable = {
		ls_pool = RAJ_LS_party_seats^RAJ_LS_t_party
	}
	subtract_from_temp_variable = {
		ls_pool = RAJ_LS_wooed^RAJ_LS_t_party
	}
	# outside parties only offer up to half their bench
	if = {
		limit = {
			NOT = {
				check_variable = {
					RAJ_LS_ruling = RAJ_LS_t_party
				}
			}
			check_variable = {
				RAJ_LS_coalition^RAJ_LS_t_party < 1
			}
		}
		set_temp_variable = {
			ls_cap = RAJ_LS_party_seats^RAJ_LS_t_party
		}
		divide_temp_variable = {
			ls_cap = 2
		}
		set_temp_variable = {
			ls_pool2 = ls_cap
		}
		subtract_from_temp_variable = {
			ls_pool2 = RAJ_LS_wooed^RAJ_LS_t_party
		}
		if = {
			limit = {
				check_variable = {
					ls_pool2 < ls_pool
				}
			}
			set_temp_variable = {
				ls_pool = ls_pool2
			}
		}
	}
	if = {
		limit = {
			check_variable = {
				ls_pool < 0
			}
		}
		set_temp_variable = {
			ls_pool = 0
		}
	}
	set_temp_variable = {
		ls_gain = ls_pool
	}
	multiply_temp_variable = {
		ls_gain = 0.3
	}
	round_temp_variable = ls_gain
	add_to_variable = {
		RAJ_LS_bill_pledged = ls_gain
	}
	add_to_variable = {
		RAJ_LS_wooed^RAJ_LS_t_party = ls_gain
	}
	set_temp_variable = {
		RAJ_LS_woo_display = ls_gain
	}
	custom_effect_tooltip = RAJ_LS_woo_party_tt
}
