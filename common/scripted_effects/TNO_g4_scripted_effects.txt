		}
		set_variable = {
			GER_kn_detente_consumer_goods_factor = -0.08

	set_variable = { GER_g4_wehrmacht_coup_aftermath_war_support_factor = -0.7 }

#victory: branch by ruling party
GER_g4_endgame_victory = {
	#radical VAAF, territory intact - load radical tree
	if = {
		limit = {
			OR = {
				has_country_flag = GER_g4_radicals_takeover_vaaf
				has_country_flag = ger_vaaf_radical_coup
			}
		}
		load_focus_tree = GER_gfour_vaaf_radical_tree
		mark_focus_tree_layout_dirty = yes
		set_country_flag = GER_g4_endgame_resolved
		hidden_effect = {
			country_event = { id = GER_gfour_vaaf_rad.3 days = 2 }
		}
	}
	#Die Grunen victory
	else_if = {
		limit = { has_country_flag = GER_g4_dg_takeover }
		load_focus_tree = GER_gfour_dg_tree
		mark_focus_tree_layout_dirty = yes
		set_country_flag = GER_g4_endgame_resolved
		hidden_effect = {
			country_event = { id = GER_gfour_dg_true.4 days = 2 }
		}
	}
	#NSDAP with a two-term mandate that already passed law_national_unity_front - its tree loads now
	else_if = {
		limit = {
			GER_nsdap_government = yes
			check_variable = { GER_nsdap_terms_won = 2 }
			has_country_flag = GER_reichstag_law_national_unity_front_passed
		}
		load_focus_tree = GER_gfour_nsdap_unity_tree
		mark_focus_tree_layout_dirty = yes
		set_country_flag = GER_g4_endgame_resolved
		hidden_effect = {
			country_event = { id = GER_gfour_nsdap.1 days = 2 }
		}
	}
	#NSDAP with a two-term mandate that already passed majority_law - its tree loads now
	else_if = {
		limit = {
			GER_nsdap_government = yes
			check_variable = { GER_nsdap_terms_won = 2 }
			has_country_flag = GER_reichstag_majority_law_passed
		}
		load_focus_tree = GER_gfour_nsdap_majority_tree
		mark_focus_tree_layout_dirty = yes
		set_country_flag = GER_g4_endgame_resolved
		hidden_effect = {
			country_event = { id = GER_gfour_nsdap.1 days = 2 }
		}
	}
	#NSDAP with a two-term mandate - wait for majority_law or law_national_unity_front (tree loads when law passes)
	else_if = {
		limit = {
			GER_nsdap_government = yes
			check_variable = { GER_nsdap_terms_won = 2 }
		}
		set_country_flag = GER_g4_endgame_awaiting_nsdap_law
		custom_effect_tooltip = GER_g4_endgame_nsdap_law_reminder_tt
	}
	#NSDAP without two terms - can't pass the consolidation law, fall back to the unity tree
	else_if = {
		limit = { GER_nsdap_government = yes }
		load_focus_tree = GER_gfour_nsdap_unity_tree
		mark_focus_tree_layout_dirty = yes
		set_country_flag = GER_g4_endgame_resolved
		hidden_effect = {
			country_event = { id = GER_gfour_nsdap.1 days = 2 }
		}
	}
	#VUP that already passed imperial_restoration_law before the endgame fired - its tree loads now
	else_if = {
		limit = {
			GER_vup_government = yes
			has_country_flag = GER_hapsburg_active
			has_country_flag = GER_reichstag_imperial_restoration_law_passed
			check_variable = { GER_vup_terms_won = 2 }
		}
		load_focus_tree = GER_gfour_vup_imperial_tree
		mark_focus_tree_layout_dirty = yes
		set_country_flag = GER_g4_endgame_resolved
		hidden_effect = {
			#coronation of Otto von Habsburg
			country_event = { id = GER_gfour_vup.2 }
		}
	}
	#VUP with Habsburg active, a two-term mandate, and restoration prereqs met - wait for imperial_restoration_law
	else_if = {
		limit = {
			GER_vup_government = yes
			has_country_flag = GER_hapsburg_active
			has_country_flag = GER_reichstag_general_rights_act_passed
			check_variable = { GER_vup_terms_won = 2 }
		}
		set_country_flag = GER_g4_endgame_awaiting_vup_law
		custom_effect_tooltip = GER_g4_endgame_vup_law_reminder_tt
	}
	#VUP without Habsburg, without restoration prereqs, or without two terms - load republican tree immediately
	else_if = {
		limit = { GER_vup_government = yes }
		load_focus_tree = GER_gfour_vup_republican_tree
		mark_focus_tree_layout_dirty = yes
		set_country_flag = GER_g4_endgame_resolved
		hidden_effect = {
			country_event = { id = GER_gfour_vup.3 days = 2 }
		}
	}
	#RSP victory
	else_if = {
		limit = { has_government = authoritarian_democrat }
		load_focus_tree = GER_gfour_rsp_tree
		mark_focus_tree_layout_dirty = yes
		set_country_flag = GER_g4_endgame_resolved
		hidden_effect = {
			country_event = { id = GER_gfour_rsp.1 days = 2 }
		}
	}
	#VAAF moderate
	else_if = {
		limit = { has_government = socialist }
		load_focus_tree = GER_gfour_vaaf_tree
		mark_focus_tree_layout_dirty = yes
		set_country_flag = GER_g4_endgame_resolved
		hidden_effect = {
			country_event = { id = GER_gfour_vaaf.1 days = 2 }
		}
	}
}
