ITA_clamp_centralization_var = {
	round_variable = ITA_centralization
	clamp_variable = { var = ITA_centralization min = 0 max = 100 }

	set_variable = { ITA_decentralization = 100 } #only for display
	subtract_from_variable = { ITA_decentralization = ITA_centralization }

	set_variable = { ITA_centralization_x_axis = 4.39 } 
	multiply_variable = { ITA_centralization_x_axis = ITA_centralization }
	round_variable = ITA_centralization_x_axis
	add_to_variable = { ITA_centralization_x_axis = 37 }
}
