	DOW_casualties_impact = {
		#TUR_DOW_casualties_per_morale_decrease defined in TNO_Turkey_on_actions.txt
		
		#This decision is a (crude) way to decrease war support roughly in line with casualties
		#Once your casualties pass the threshhold (8000 dead men more than the previous time),
		#you lose 10 support in DOW, then it re-enables with a new threshhold. 8000 men away
		allowed = {
			tag = TUR
		}
		available = {
			OR = {
				custom_trigger_tooltip = {
					tooltip = DOW_casualties_impact_trigger_tt
					casualties > TUR_DOW_next_casualty_level
				}
				has_war = no
			}
		}
		activation = {
			has_global_flag = ITA_MED_WIP_PATHS
			OR = { 
				has_war_with = IME 
				has_war_with = GRE
				has_war_with = IID
			}
			
		}
		is_good = yes
		#fire_only_once = no
		days_mission_timeout = days_mission_timeout@War_Timer #to the end of the war
		# timeout_effect = {
			# #None
		# }
		
		complete_effect = {
			if = {
				limit = { has_war = yes }
				
				custom_effect_tooltip = TUR_dow_tt_down_large
				add_to_variable = { TUR_DOW = -10 }
				TUR_DoW_change_stage = yes
				
				set_variable = { TUR_DOW_next_casualty_level = casualties }
				add_to_variable = { TUR_DOW_next_casualty_level = TUR_DOW_casualties_per_morale_decrease }
				
				#Once the new threshhold is set, re-activate this mission
				custom_effect_tooltip = DOW_casualties_impact_effect_tt
				hidden_effect = { activate_mission = DOW_casualties_impact }
			}
		}
	}
