	# ---------- bill voting ----------
	# Bill flow (see scripted effects header): a focus/decision sets its own
	# dispatch flag, calls RAJ_LS_start_bill, activates RAJ_LS_bill_vote_mission.
	RAJ_LS_bill_vote_mission = {
		icon = GFX_decision_generic_political_discourse
		allowed = { tag = RAJ }
		activation = {
			has_country_flag = RAJ_LS_bill_active
		}
		available = {
			check_variable = { RAJ_LS_bill_pledged > RAJ_LS_majority }
		}
		is_good = yes
		selectable_mission = yes
		days_mission_timeout = 35
		fire_only_once = no
		cancel_trigger = {
			NOT = { has_country_flag = RAJ_LS_bill_active }
		}
		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision RAJ_LS_bill_vote_mission PASSED"
			custom_effect_tooltip = RAJ_LS_bill_passed_tt
			set_country_flag = RAJ_LS_bill_passed
			RAJ_LS_bill_resolve = yes
			hidden_effect = {
				RAJ_LS_end_bill = yes
			}
		}
		timeout_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision RAJ_LS_bill_vote_mission FAILED"
			custom_effect_tooltip = RAJ_LS_bill_failed_tt
			set_country_flag = RAJ_LS_bill_failed
			hidden_effect = {
				RAJ_LS_fail_bill = yes
			}
		}
	}

			for_loop_effect = {
				start = 0
				end = RAJ_LS_party_count
				value = i
				if = {
					limit = {
						OR = {
							check_variable = { RAJ_LS_ruling = i }
							check_variable = { RAJ_LS_coalition^i > 0 }
						}
					}
					set_temp_variable = { RAJ_LS_t_party = i }
					RAJ_LS_woo_party = yes
				}
			}
		}
	}
