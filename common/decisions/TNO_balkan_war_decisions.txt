	# ---------------------------------------------------------------------------------
	# Pressing a claim. One decision per contested state, so each carries exactly the
	# claimants who have business with it. A nation can run one border war at a time, must
	# rest 180 days between them, and stops entirely once it has fought the open war.
	# A state's 1962 holder is a claimant too, so it can take the state back once lost; Greece,
	# which did not hold Thrace in 1962, only once a border war has taken it from Greece.
	# ---------------------------------------------------------------------------------
	BALKAN_press_76 = {
		allowed = {
			OR = {
				tag = HUN
				tag = ROM
			}
		}
		visible = {
			has_global_flag = BALKAN_BORDER_WARS
			NOT = { has_country_flag = BALKAN_open_war_resolved }
			NOT = { 76 = { is_owned_by = ROOT } }
		}
		available = {
			BALKAN_presser_ready = yes
			76 = { BALKAN_state_contestable = yes }
		}
		highlight_states = {
			highlight_state_targets = { state = 76 }
		}
		on_map_mode = map_and_decisions_view
		cost = 35
		days_remove = 30
		fire_only_once = no
		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision BALKAN_press_76 (Bihor)"
		}
		remove_effect = {
			log = "[GetDateText]: [Root.GetName]: BALKAN_press_76 resolves into a border war"
			76 = { save_global_event_target_as = BALKAN_defender_state }
			BALKAN_begin_border_war = yes
		}
		ai_will_do = {
			factor = 0
			modifier = {
				add = 25
				has_war = no
			}
			modifier = {
				factor = 0
				NOT = { has_political_power > 60 }
