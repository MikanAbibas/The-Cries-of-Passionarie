# Stopgap content for after the Oil Crisis civil war, until proper Egypt content
# is written. Gated on OIL_Egypt_Critical (set when the civil war begins) so the
# decisions only appear after the war, not at game start when EGB/EGQ also do not
# exist yet. The AI is made to always take both decisions.
EGY_reconquest_decisions = {

	# Bring the warring Sudanese statelets to heel.
	EGY_invade_sudan = {

		icon = generic_operation

		available = {
			has_global_flag = OIL_Egypt_Critical
			NOT = { country_exists = EGB }
			NOT = { country_exists = EGQ }
			has_war = no
			OR = {
				country_exists = RPS
				country_exists = SDS
				country_exists = SDF
				country_exists = AZA
				country_exists = AYA
			}
		}

		visible = {
			OR = {
				country_exists = RPS
				country_exists = SDS
				country_exists = SDF
				country_exists = AZA
				country_exists = AYA
			}
		}

		cost = 0

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision EGY_invade_sudan"
			if = { limit = { country_exists = RPS } declare_war_on = { target = RPS type = annex_everything } }
			if = { limit = { country_exists = SDS } declare_war_on = { target = SDS type = annex_everything } }
			if = { limit = { country_exists = SDF } declare_war_on = { target = SDF type = annex_everything } }
			if = { limit = { country_exists = AZA } declare_war_on = { target = AZA type = annex_everything } }
			if = { limit = { country_exists = AYA } declare_war_on = { target = AYA type = annex_everything } }
		}

		ai_will_do = {
			factor = 100
		}
	}

	# Once the Sudan is pacified, return to Italy's alliance as an independent member.
	EGY_rejoin_rome_pakt = {

		icon = generic_form_nation

		available = {
			has_global_flag = OIL_Egypt_Critical
			NOT = { country_exists = EGB }
			NOT = { country_exists = EGQ }
			NOT = { country_exists = RPS }
			NOT = { country_exists = SDS }
			NOT = { country_exists = SDF }
			NOT = { country_exists = AZA }
			NOT = { country_exists = AYA }
			has_war = no
			ITA = {
				exists = yes
				is_faction_leader = yes
			}
		}

		visible = {
			NOT = { country_exists = RPS }
			NOT = { country_exists = SDS }
			NOT = { country_exists = SDF }
			NOT = { country_exists = AZA }
			NOT = { country_exists = AYA }
		}

		cost = 0

		fire_only_once = yes

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision EGY_rejoin_rome_pakt"
			if = { limit = { has_idea = TRI_Member_Client } remove_ideas = TRI_Member_Client }
			if = { limit = { has_idea = TRI_Member } remove_ideas = TRI_Member }
			add_ideas = Rome_member
			ITA = { add_to_faction = EGY }
		}

		ai_will_do = {
			factor = 100
		}
	}
}
