#Defense in depth: a tenth of the army waits around Germania and joins the fight
#only once the front reaches Germania or Poland (defend_germania_zone in ai_areas)
GER_vengeance_war_defense_in_depth = {
	allowed = {
		original_tag = GER
	}
	enable = {
		tag = GER
		has_global_flag = rus_vengeance_war
		any_enemy_country = { is_russian_unifier = yes }
	}

	abort_when_not_enabled = yes

	ai_strategy = {
		type = put_unit_buffers
		ratio = 0.1
		order_id = 1
		states = {
			64 #Brandenburg
			1026 #Eastern Brandenburg
			65 #Sachsen
			68 #Wartheland
		}
		area = defend_germania_zone
		subtract_invasions_from_need = no
		subtract_fronts_from_need = yes
	}
}
