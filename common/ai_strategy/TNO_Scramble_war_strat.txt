# New Scramble for Africa (plan 4.4; user, 2026-10-16: warlords expand aggressively and try to unify).
# In a scramble war the AI attacks instead of holding the line. Scopes (vanilla _documentation.md): in country_trigger
# THIS is the enemy and FROM is us. Priority 80 is the war-specific rush rung (CLAUDE.md): it overrides the generic
# attack order (50) and yields to end_pocket (90) and going_slow (400).

SFA_scramble_war_rush = {
	enable = {
		has_country_flag = SFA_in_scramble_war
	}

	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		country_trigger = {
			has_country_flag = SFA_in_scramble_war
			check_variable = { SFA_war_id = FROM.SFA_war_id }
		}
		priority = 80
		ordertype = front
		execution_type = rush
		manual_attack = yes
		execute_order = yes
	}
}
