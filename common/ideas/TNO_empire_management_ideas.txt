# Generic Empire-Management client states, added to any client by ITA's investment system.
ideas = {
	country = {

		# Granted after ~5 years of sustained high Italian investment. Blocks the civil war.
		empire_stabilized = {
			on_add = {log = "[GetDateText]: [Root.GetName]: add idea empire_stabilized"}
			picture = GFX_idea_ITA_GenericEconomy
			allowed = { always = no }
			removal_cost = -1
			modifier = {
				stability_factor = 0.15
				political_power_gain = 0.10
				war_support_factor = 0.10
			}
		}

		# Upgrade from stabilized after sustained max investment - a powerful Italian ally.
		empire_keystone_ally = {
			on_add = {log = "[GetDateText]: [Root.GetName]: add idea empire_keystone_ally"}
			picture = GFX_idea_ITA_GenericMilitary
			allowed = { always = no }
			removal_cost = -1
			modifier = {
				stability_factor = 0.20
				war_support_factor = 0.15
				army_attack_factor = 0.10
				army_defence_factor = 0.10
				production_speed_buildings_factor = 0.10
			}
		}
	}
}
