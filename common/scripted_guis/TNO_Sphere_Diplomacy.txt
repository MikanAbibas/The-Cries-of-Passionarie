	TNO_JapanDiploChinaProper_ROC_diplobox_open = {
		context_type = player_context
		parent_window_token = "TNO_JapanDiploChinaProper_countryview"
		window_name = "TNO_JapanDiploChinaProper_ROC_diplobox"

		visible = {
			has_country_flag = JapanDiploChinaInteractionsVisible
		}

		triggers = {
			ChinaProper_ROC_interaction_1_button_click_enabled = {
				check_variable = {
					var = political_power
					value = 150
					compare = greater_than_or_equals
				}
				country_exists = CHI
				NOT = { has_war_with = CHI }
				CHI = {
					OR = {
						is_in_faction_with = JAP
						is_in_faction = no
					}
				}
				NOT = { has_country_flag = JAP_sphere_cd_ChinaProper_ROC_interaction_1_button }
			}
			ChinaProper_ROC_interaction_2_button_click_enabled = {
				check_variable = {
					var = political_power
					value = 150
					compare = greater_than_or_equals
				}
				country_exists = CHI
				NOT = { has_war_with = CHI }
				CHI = {
					OR = {
						is_in_faction_with = JAP
						is_in_faction = no
					}
				}
				NOT = { has_country_flag = JAP_sphere_cd_ChinaProper_ROC_interaction_2_button }
			}
			ChinaProper_ROC_interaction_3_button_click_enabled = {
				check_variable = {
					var = political_power
					value = 200
					compare = greater_than_or_equals
				}
				country_exists = CHI
				NOT = { has_war_with = CHI }
				CHI = {
					OR = {
						is_in_faction_with = JAP
						is_in_faction = no
					}
				}
				NOT = { has_country_flag = JAP_sphere_cd_ChinaProper_ROC_interaction_3_button }
				check_variable = { JAP_sphere_offsite_CHI < 3 }
			}
			ChinaProper_ROC_interaction_4_button_click_enabled = {
				check_variable = {
					var = political_power
					value = 300
					compare = greater_than_or_equals
				}
				country_exists = CHI
				NOT = { has_war_with = CHI }
				CHI = {
					OR = {
						is_in_faction_with = JAP
						is_in_faction = no
					}
				}
				NOT = { has_country_flag = JAP_sphere_cd_ChinaProper_ROC_interaction_4_button }
			}
		}

		effects = {
			ChinaProper_ROC_close_button_click = {
				clr_country_flag = JapanDiploChinaInteractionsVisible
			}

			# Increase Output: Japan takes the output, China bears the cost
			ChinaProper_ROC_interaction_1_button_click = {
				if = {
					limit = {
						check_variable = {
							var = political_power
							value = 150
							compare = greater_than_or_equals
						}
					}
					add_political_power = -150
					set_country_flag = { flag = JAP_sphere_cd_ChinaProper_ROC_interaction_1_button days = 180 value = 1 }
					JAP = {
						add_timed_idea = { idea = JAP_sphere_output_CHI days = 180 }
					}
					CHI = {
						if = {
							limit = { JAP_sphere_is_subordinate = yes }
							add_timed_idea = { idea = JAP_sphere_output_taken_t1 days = 180 }
						}
						else = {
							add_timed_idea = { idea = JAP_sphere_output_shared_t1 days = 180 }
						}
					}
				}
			}
			# Population Exchange: Chinese labour for Japan
			ChinaProper_ROC_interaction_2_button_click = {
				if = {
					limit = {
						check_variable = {
							var = political_power
							value = 150
							compare = greater_than_or_equals
						}
					}
					add_political_power = -150
					set_country_flag = { flag = JAP_sphere_cd_ChinaProper_ROC_interaction_2_button days = 365 value = 1 }
					JAP = {
						add_manpower = 5000
					}
					CHI = {
						if = {
							limit = { JAP_sphere_is_subordinate = yes }
							add_manpower = -5000
						}
						else = {
							add_manpower = 5000
						}
					}
				}
			}
			# Industrial Cooperation (as Guangdong; capped at 3 like the others)
			ChinaProper_ROC_interaction_3_button_click = {
				if = {
					limit = {
						check_variable = {
							var = political_power
							value = 200
							compare = greater_than_or_equals
						}
					}
					add_political_power = -200
					set_country_flag = { flag = JAP_sphere_cd_ChinaProper_ROC_interaction_3_button days = 365 value = 1 }
					add_to_variable = { JAP_sphere_offsite_CHI = 1 }
					JAP = {
						add_offsite_building = { type = industrial_complex level = 1 }
					}
					CHI = {
						add_offsite_building = { type = industrial_complex level = 1 }
					}
				}
			}
			# Address the Army (as the other clients' Address the Army)
			ChinaProper_ROC_interaction_4_button_click = {
				if = {
					limit = {
						check_variable = {
							var = political_power
							value = 300
							compare = greater_than_or_equals
						}
					}
					add_political_power = -300
					set_country_flag = { flag = JAP_sphere_cd_ChinaProper_ROC_interaction_4_button days = 365 value = 1 }
					CHI = {
						army_experience = -25
					}
					JAP = {
						add_tech_bonus = {
							bonus = 0.15
							uses = 1
							category = land_doctrine
						}
					}
				}
			}
		}
	}
