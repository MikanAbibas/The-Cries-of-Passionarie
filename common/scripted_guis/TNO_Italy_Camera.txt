		effects = {
			amendments_list_close_button_click = {
				clr_country_flag = TNO_ITA_Show_DDL_Amendments
			}
			add_amend_click = {
				# Find first open slot (if one exists)
				set_temp_variable = { slot = 0 }
				while_loop_effect = {
					limit = {
						check_variable = { ITA_ddl_amendments^slot > 0 }
					}
					add_to_temp_variable = { slot = 1 }
				}
				if = {
					limit = {
						check_variable = { slot < 4 }
					}
					# Add amendment, remove from possible list, hide selector UI
					set_variable = { ITA_ddl_amendments^slot = amendment_num }
					remove_from_array = { array = ITA_ddl_possible_amendments value = amendment_num }
					clr_country_flag = TNO_ITA_Show_DDL_Amendments

					ITA_camera_update_ddl_policy_shift = yes
					ITA_camera_update_coalition_status = yes
					ITA_camera_redraw = yes
				}
			}
		}
