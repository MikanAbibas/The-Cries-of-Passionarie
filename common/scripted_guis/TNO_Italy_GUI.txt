	ITA_GUI_Promises_GUI = {
		context_type = player_context
		window_name = "ITA_GUI_Promises_Container_GUI"
		
		visible = {
			has_country_flag = ITA_promises_show_adding_window
		}
		
		triggers = {
			ITA_GUI_Promises_Entry_Button_click_enabled = {
				has_political_power > ITA_avalaible_promises_cost_array^ITA_avalaible_promises_array^ITA_avalaible_promises_i
			}
		}

		dynamic_lists = {
			ITA_POP_GUI_avalaible_promises_grid = {
				array = ITA_avalaible_promises_array
				change_scope = no
				entry_container = ITA_GUI_Promises_Entry
				index = ITA_avalaible_promises_i
			}
		}
		
		effects = {
			ITA_GUI_Promises_Entry_Button_click = {
				set_temp_variable = { pp_to_subtract = ITA_avalaible_promises_cost_array^ITA_avalaible_promises_array^ITA_avalaible_promises_i }
				if = {
					limit = {
						check_variable = { pp_to_subtract > 0 }
					}
					multiply_temp_variable = { pp_to_subtract = -1 }
					add_political_power = pp_to_subtract
				}
				
				set_temp_variable = { promise_num = ITA_avalaible_promises_array^ITA_avalaible_promises_i }

				if = {
					limit = { check_variable = { promise_num = 1 } }
					#activate_mission = ITA_dem_pop_keep_promise_1
					set_variable = { changer = 0 } #farers
					set_temp_variable = { constant = 5 } #5%
					ITA_increase_left_Popolo = yes

					set_variable = { changer = 1 } #urban workers
					set_temp_variable = { constant = 15 } #15%
					ITA_increase_left_Popolo = yes

					set_variable = { changer = 2 } #middle classes
					set_temp_variable = { constant = 10 } #10%
					ITA_increase_left_Popolo = yes

					set_variable = { changer = 4 } #students
					set_temp_variable = { constant = 25 }
					ITA_increase_left_Popolo = yes

					set_variable = { changer = 3 } #military brass
					set_temp_variable = { constant = 10 }
					ITA_decrease_left_Stato = yes

					set_variable = { changer = 5 } #cleregy
					set_temp_variable = { constant = 10 }
					ITA_decrease_left_Stato = yes
				}
				else_if = {
					limit = { check_variable = { promise_num = 2 } }
					#activate_mission = ITA_dem_pop_keep_promise_2
					set_variable = { changer = 0 } #farers
					set_temp_variable = { constant = 30 }
					ITA_increase_left_Popolo = yes

					set_variable = { changer = 0 } #civil servants
					set_temp_variable = { constant = 5 }
					ITA_decrease_left_Stato = yes

					set_variable = { changer = 5 } #cleregy
					set_temp_variable = { constant = 10 }
					ITA_decrease_left_Stato = yes
				}
				else_if = {
					limit = { check_variable = { promise_num = 3 } }
					#activate_mission = ITA_dem_pop_keep_promise_3
					set_variable = { changer = 1 } #urban workers
					set_temp_variable = { constant = 30 }
					ITA_increase_left_Popolo = yes

					set_variable = { changer = 0 } #civil servants
					set_temp_variable = { constant = 5 }
					ITA_decrease_left_Stato = yes
