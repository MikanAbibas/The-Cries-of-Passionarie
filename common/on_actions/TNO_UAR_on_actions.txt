# A Ba'athist nation that capitulates is out of the Oil Crisis struggle. When the last
# one falls, the struggle ends: with the UAR standing, clean-up and rewards; without it,
# clean-up only. See UAR_check_baathist_struggle in TNO_UAR_scripted_effects.txt.
on_actions = {
	on_capitulation = {
		# ROOT is the capitulated country, FROM the winner.
		effect = {
			if = {
				limit = { UAR_is_baathist_nation = yes }
				set_country_flag = UAR_baathist_defeated
				UAR_check_baathist_struggle = yes
			}
		}
	}
}
