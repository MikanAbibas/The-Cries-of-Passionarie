on_actions = {
	# New Scramble for Africa (plan 7.3; user, 2026-10-16): a patron's war mark updates the moment any war involving
	# an African nation starts, scramble war or not, and the players' boxes redraw. ROOT attacks, FROM defends. Elsewhere
	# in the world it costs one flag check.
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					has_global_flag = SFA_active
					OR = {
						ROOT = { has_country_flag = SFA_african_nation }
						FROM = { has_country_flag = SFA_african_nation }
					}
				}
				ROOT = { SFA_alert_patron = yes }
				FROM = { SFA_alert_patron = yes }
				SFA_count_proxies_at_war = yes
				SFA_gui_refresh = yes
			}
		}
	}
}
