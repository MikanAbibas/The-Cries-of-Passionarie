	id = ITA_erase_sansepolcro
	icon = GFX_goal_unknown
	cost = 1.00
	available_if_capitulated = no
	ai_will_do = { factor = 1 }
	prerequisite = {
		focus = ITA_take_the_stage
	}
	
	x = 0
	y = 1
	relative_position_id = ITA_take_the_stage
	completion_reward = {
		log = "[GetDateText]: [Root.GetName]: Focus ITA_erase_sansepolcro"
		set_temp_variable = {econ_centralization_temp = 5}
		econ_centralization_change = yes 
		# Pavolini's inauguration ends when the tree is spent. Princeps Civitatis needs both
		# soft options (Iron Pen, Empty the Offices); either hard option (Fire and Lead,
		# Evict Rome's Traitors) rules it out, and then the tree ends once both branches
		# and the Sansepolcro line are done. This check sits on every focus that can finish
		# last, and treats its own focus as done.
		if = {
			limit = {
				OR = {
					has_completed_focus = ITA_princeps_civitatis
					AND = {
						OR = { has_completed_focus = ITA_evict_romes_traitors has_completed_focus = ITA_with_fire_and_lead }
						OR = { has_completed_focus = ITA_evict_romes_traitors has_completed_focus = ITA_crush_the_old_guard }
						OR = { has_completed_focus = ITA_with_fire_and_lead has_completed_focus = ITA_gag_the_fanatics }
						always = yes
					}
				}
			}
			hidden_effect = {
				load_focus_tree = ITA_pavolini_trees_main
				mark_focus_tree_layout_dirty = yes
			}
		}
	}
}
