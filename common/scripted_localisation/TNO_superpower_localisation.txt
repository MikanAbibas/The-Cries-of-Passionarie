	name = GetRankName
	text = {
		trigger = {
			tag = THIS
			has_idea = ZZZ_superpower_rank_1
		}
		localization_key = TNO_world_hegemon_tt
	}
	text = {
		trigger = {
			tag = THIS
			has_idea = ZZZ_superpower_rank_2
		}
		localization_key = TNO_global_contender_tt
	}
	text = {
		trigger = {
			tag = THIS
			has_idea = ZZZ_superpower_rank_3
		}
		localization_key = TNO_rising_force_tt
	}
	text = {
		trigger = {
			tag = THIS
			NOT = {
				OR = {
					has_idea = ZZZ_superpower_rank_1

					has_idea = ZZZ_superpower_rank_3
				}
			}
		}
		localization_key = TNO_normal_superpower_tt
	}
}
