# Uses a recursive scripted loc (ITA_Get_DDL_Relevant_Stato_Groups_inner) which modifies temp variables 
# to print out a list of all stato groups relevant for this DDL
defined_text = {
	name = ITA_Get_DDL_Relevant_Stato_Groups
	text = {
		trigger = {
			check_variable = { ITA_ddl_focus_relevant_stato_groups^num = 0 }
		}
		localization_key = "ITA_ddl_focus_stato_support"
	}
	text = {
		trigger = {
			set_temp_variable = { sgi = -1 }
			subtract_from_temp_variable = { cn = 1 }
			set_temp_variable = { max_sgi = ITA_ddl_focus_relevant_stato_groups^num }
			subtract_from_temp_variable = { max_sgi = 2 }
		}
		localization_key = "[ITA_Get_DDL_Relevant_Stato_Groups_inner]"
	}
}

defined_text = {
	name = Get_Current_Convince_Tooltip
	text = {
		trigger = {
			set_temp_variable = { supporting_deputies = ITA_ddl_support_current_deputies^current_number }

			set_temp_variable = { new_pct = ITA_ddl_support_current_pct^current_number }
			# Saragat Negotiation Bonus
			if = {
				limit = { check_variable = { ITA_gf_selected_pm = 4 } }
				multiply_temp_variable = { new_pct = 1.15 }
			}
			# Berlinguer Bonus/Penalty
			else_if = {
				limit = { check_variable = { ITA_gf_selected_pm = 0 } }
				set_temp_variable = { bonus = ITA_ddl_negotiation_effectiveness_modifier_optimism }
				multiply_temp_variable = { bonus = 0.1 }
				add_to_temp_variable = { bonus = 1.00 }
				multiply_temp_variable = { new_pct = bonus }
			}
			# Fanfani Penalty
			else_if = {
				limit = {
					check_variable = { ITA_gf_selected_pm = 7 }
					check_variable = { ITA_camera_current_coalition^current_number < 2 }
				}
				multiply_temp_variable = { new_pct = 1.00 }
			}
			# Moro Bonus
			else_if = {
				limit = { check_variable = { ITA_gf_selected_pm = 8 } }
				set_temp_variable = { bonus = ITA_ddl_negotiation_effectiveness_modifier_cooperation }
				multiply_temp_variable = { bonus = 0.1 }
				add_to_temp_variable = { bonus = 1.05 }
				multiply_temp_variable = { new_pct = bonus }
			}
			else = {
				multiply_temp_variable = { new_pct = 1.05 }
			}
			clamp_temp_variable = { var = new_pct min = 0 max = 1}
			set_temp_variable = { min_new_support = ITA_camera_current_deputies^current_number }
			multiply_temp_variable = { min_new_support = new_pct }
			round_temp_variable = min_new_support
			subtract_from_temp_variable = { min_new_support = supporting_deputies }
			
			set_temp_variable = { new_pct = ITA_ddl_support_current_pct^current_number }
			# Saragat Negotiation Bonus
			if = {
				limit = { check_variable = { ITA_gf_selected_pm = 4 } }
				multiply_temp_variable = { new_pct = 1.25 }
			}
			# Berlinuger Bonus/Penalty
			else_if = {
				limit = { check_variable = { ITA_gf_selected_pm = 0 } }
				set_temp_variable = { bonus = ITA_ddl_negotiation_effectiveness_modifier_optimism }
				multiply_temp_variable = { bonus = 0.1 }
				add_to_temp_variable = { bonus = 1.10 }
				multiply_temp_variable = { new_pct = bonus }
			}
			# Fanfani Penalty
			else_if = {
				limit = {
					check_variable = { ITA_gf_selected_pm = 7 }
					check_variable = { ITA_camera_current_coalition^current_number < 2 }
				}
				multiply_temp_variable = { new_pct = 1.10 }
			}
			# Moro Bonus
			else_if = {
				limit = { check_variable = { ITA_gf_selected_pm = 8 } }
				set_temp_variable = { bonus = ITA_ddl_negotiation_effectiveness_modifier_cooperation }
				multiply_temp_variable = { bonus = 0.1 }
				add_to_temp_variable = { bonus = 1.15 }
				multiply_temp_variable = { new_pct = bonus }
			}
			else = {
				multiply_temp_variable = { new_pct = 1.15 }
			}
			clamp_temp_variable = { var = new_pct min = 0 max = 1}
			set_temp_variable = { max_new_support = ITA_camera_current_deputies^current_number }
			multiply_temp_variable = { max_new_support = new_pct }
			round_temp_variable = max_new_support
			subtract_from_temp_variable = { max_new_support = supporting_deputies }

			set_temp_variable = { pp_cost = 20 }
			if = {
				limit = { check_variable = { ITA_camera_current_coalition^current_number = 2 } }
				set_temp_variable = { pp_cost = 10 }
			}

			ITA_ddl_active = no
		}
		localization_key = ITA_camera_current_convince_tt_no_legge
	}
	text = {
		trigger = {
			check_variable = { ITA_ddl_has_convinced^current_number = 1 }
		}
		localization_key = ITA_camera_current_convince_tt_already_done
	}
	text = {
		trigger = {
			check_variable = { ITA_camera_current_coalition^current_number = 2 }
			check_variable = { max_new_support > 0 }
		}
		localization_key = ITA_camera_current_convince_tt_ok_coal
	}
	text = {
		trigger = {
			check_variable = { max_new_support > 0 }
		}
		localization_key = ITA_camera_current_convince_tt_ok
	}
	text = {
		localization_key = ITA_camera_current_convince_tt_not_enough_support
	}
}

# to print out a list of all coalition partners
defined_text = {
	name = Get_Coalition_Current_Names
	text = {
		trigger = {
			set_temp_variable = { cn = ITA_coalition_overton_base }
			subtract_from_temp_variable = { cn = 1 }
			set_temp_variable = { max_current_number = ITA_coalition_overton_end }
			subtract_from_temp_variable = { max_current_number = 1 }
			set_temp_variable = { total_size = max_current_number }
			subtract_from_temp_variable = { total_size = cn }
		}
		localization_key = "[Get_Coalition_Current_Names_inner]"
	}
}

defined_text = {
	name = Get_Coalition_Current_Names_inner
	text = {
		trigger = {
			check_variable = { total_size = 0 }
			check_variable = { cn = max_current_number }
			add_to_temp_variable = { cn = 1 }
		}
		localization_key = "[ITA_Get_Current_Icon_dynamic_alt] [Get_Current_Name_Short_dynamic_alt]"
	}
	text = {
		trigger = {
			check_variable = { cn = max_current_number }
			add_to_temp_variable = { cn = 1 }
		}
		localization_key = "and [ITA_Get_Current_Icon_dynamic_alt] [Get_Current_Name_Short_dynamic_alt]"
	}
	text = {
		trigger = {
			check_variable = { total_size = 1 }
			add_to_temp_variable = { cn = 1 }
		}
		localization_key = "[ITA_Get_Current_Icon_dynamic_alt] [Get_Current_Name_Short_dynamic_alt] [Get_Coalition_Current_Names_inner]"
	}
	text = {
		trigger = {
			add_to_temp_variable = { cn = 1 }
		}
		localization_key = "[ITA_Get_Current_Icon_dynamic_alt] [Get_Current_Name_Short_dynamic_alt], [Get_Coalition_Current_Names_inner]"
	}
}

			subtract_from_temp_variable = { max_current_number = 1 }
			set_temp_variable = { total_size = max_current_number }
			subtract_from_temp_variable = { total_size = cn }
		}
		localization_key = "[Get_Coalition_Current_Names_inner]"
	}
}
