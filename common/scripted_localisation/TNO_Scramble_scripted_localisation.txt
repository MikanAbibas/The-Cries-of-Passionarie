defined_text = {
	# Order an attack
	name = SFA_GetPopupOrderLine
	text = {
		trigger = {
			has_country_flag = SFA_order_cooldown
		}
		localization_key = SFA_popup_order_cooldown
	}
	text = {
		trigger = {
			set_temp_variable = { sfa_order_code = SFA_patron_code }
			var:SFA_selected_state = { OWNER = { SFA_has_patron_code = yes } }
		}
		localization_key = SFA_popup_own_side
	}
	text = {
		trigger = {
			var:SFA_selected_state = { OWNER = { SFA_keeps_out_of_wars = yes } }
		}
		localization_key = SFA_popup_saf
	}
	text = {
		trigger = {
			var:SFA_selected_state = { OWNER = { has_country_flag = SFA_zanzibar } }
		}
		localization_key = SFA_popup_zanzibar
	}
	text = {
		trigger = {
			var:SFA_selected_state = { OWNER = { NOT = { OR = { has_country_flag = SFA_african_nation has_country_flag = SFA_sovereign_ally } } } }
		}
		localization_key = SFA_popup_not_scramble
	}
	text = {
		trigger = {
			var:SFA_selected_state = { OWNER = { OR = { is_subject = yes is_in_faction = yes } } }
		}
		localization_key = SFA_popup_protected
	}
	text = {
		trigger = {
			var:SFA_selected_state = { has_state_flag = SFA_order_target }
		}
		localization_key = SFA_popup_targeted
	}
	text = {
		trigger = {
			set_temp_variable = { sfa_order_code = SFA_patron_code }
			var:SFA_selected_state = { NOT = { any_neighbor_state = { OWNER = { SFA_is_order_proxy = yes } } } }
		}
		localization_key = SFA_popup_no_proxy
	}
	text = {
		# warlords take land only in their own home region (user, 2026-10-18)
		trigger = {
			set_temp_variable = { sfa_order_code = SFA_patron_code }
			var:SFA_selected_state = {
				OWNER = { NOT = { has_country_flag = SFA_regional_state } }
				set_temp_variable = { sfa_tr = SFA_region }
				any_neighbor_state = {
					OWNER = {
						SFA_is_order_proxy = yes
						NOT = { has_country_flag = SFA_regional_state }
					}
				}
				NOT = {
					any_neighbor_state = { SFA_neighbor_can_take = yes }
				}
			}
		}
		localization_key = SFA_popup_region
	}
	text = {
		trigger = {
			set_temp_variable = { sfa_order_code = SFA_patron_code }
			var:SFA_selected_state = { NOT = { SFA_can_order_state = yes } }
		}
		localization_key = SFA_popup_tier
	}
	text = {
		trigger = {
			NOT = { SFA_has_order_pp = yes }
		}
		localization_key = SFA_popup_order_pp
	}
	text = {
		localization_key = SFA_popup_order_ok
	}
}
