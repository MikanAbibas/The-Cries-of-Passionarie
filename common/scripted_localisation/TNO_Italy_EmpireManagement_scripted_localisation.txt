# Checks to see if the country has oil available or not

	name = EmpireManagement_GetOilAvailability
	text = {
		trigger = {
			OR = {
				check_variable = { ITA_EM_HasOil^EM_SelectedEntry_i = 0 } # Doesn't have oil
				check_variable = { ITA_EM_Regions_Status^EM_SelectedEntry_i > 2 } # Isn't in sphere
			}
		}
		localization_key = "ITA_EmpireManagement_OilExploitation_Quantity_Unnavailable"
	}
	text = {
		localization_key = "ITA_EmpireManagement_OilExploitation_Quantity_Available"
	}
}

# Returns the description w/values for Oil Exploitation effects
defined_text = {
	name = EmpireManagement_GetOilAvailabilityDesc
	text = {
		trigger = {
			NOT = {
				check_variable = { ITA_EM_HasOil^EM_SelectedEntry_i = 0 } # Doesn't have oil
				check_variable = { ITA_EM_Regions_Status^EM_SelectedEntry_i > 2 } # Isn't in sphere
			}
			check_variable = { EM_SelectedEntry_i > 5 }
			var:ITA_EM_HasOil^EM_SelectedEntry_i = { set_temp_variable = { ITA.RegionOilExploitation_OilGain = resource_produced@oil } }
			multiply_temp_variable = { RegionOilExploitation_OilGain = ITA_EM_OilExploitation^i }
			divide_temp_variable = { RegionOilExploitation_OilGain = 5 }

			set_temp_variable = { RegionOilExploitation_ResourceGain = ITA_EM_OilExploitation^EM_SelectedEntry_i }
			divide_temp_variable = { RegionOilExploitation_ResourceGain = 4 }
		}
		localization_key = "ITA_EmpireManagement_OilExploitation_Quantity_tt"
	}
}

# Here we make some quick calcs to get the PP and money costs without needing to put in a bunch of different scripted loc checks
defined_text = {
	name = EmpireManagement_GetOilExploitationButton_Cost
	text = {
		trigger = {
			set_temp_variable = { PPCost = 50 }
			set_temp_variable = { MoneyCost = 0.05 }
			if = {
				limit = {
					OR = {
						check_variable = { EM_SelectedEntry_i = 1  } # Sudan
						check_variable = { EM_SelectedEntry_i = 23 } # Oman
						check_variable = { EM_SelectedEntry_i = 26 } # Egypt
					}
				}
				multiply_temp_variable = { PPCost = 2 }
				multiply_temp_variable = { MoneyCost = 2 }
			}
			else_if = {
				limit = {
					OR = {
						check_variable = { EM_SelectedEntry_i = 0  } # Libya
						check_variable = { EM_SelectedEntry_i = 19 } # Iraq
						check_variable = { EM_SelectedEntry_i = 20 } # Saudi Arabia
						check_variable = { EM_SelectedEntry_i = 21 } # Gulf
						check_variable = { EM_SelectedEntry_i = 24 } # Algeria
					}
				}
				multiply_temp_variable = { PPCost = 3 }
				multiply_temp_variable = { MoneyCost = 3 }
			}
		}
		localization_key = "ITA_EmpireManagement_OilExploitation_Button_Cost_tt"
	}
}

	name = EmpireManagement_GetDiplomatDeploymentsEntry

			check_variable = { ITA_EM_Regions_Status^Recursion_i > 1 } # Must be able to have Diplomats
			set_temp_variable = { Region_i = Recursion_i }
		}
		localization_key = "ITA_EmpireManagement_Diplomats_Progbar_Entry_tt"
	}
	text = {
		trigger = { check_variable = { ITA_EM_Regions_Envoys^num > Recursion_i } } # If the above conditions return false for the region but the loop still hasn't been terminated, it will simply just skip the region and continue
		localization_key = "[EmpireManagement_GetDiplomatDeploymentsEntry]"
	}
}
