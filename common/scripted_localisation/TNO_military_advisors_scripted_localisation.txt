# Military Advisors window (interface/TNO_military_advisors.gui). Scope: the player's country.
# A variable that was never set reads 0, so a country that has not touched the window shows level 0.

defined_text = {
	name = TNO_GetMilitaryAdvisorsBonus
	text = {
		trigger = { check_variable = { TNO_military_advisors_level = 0 } }
		localization_key = TNO_MILITARY_ADVISORS_BONUS_NONE
	}
	text = {
		trigger = { check_variable = { TNO_military_advisors_level = 1 } }
		localization_key = TNO_MILITARY_ADVISORS_BONUS_ONE
	}
	text = {
		localization_key = TNO_MILITARY_ADVISORS_BONUS_MANY
	}
}

# the price line under the button
defined_text = {
	name = TNO_GetMilitaryAdvisorsCost
	text = {
		trigger = {
			OR = {
				check_variable = { TNO_military_advisors_level > 4 }
				NOT = { econ_can_use_economy_system = yes }
			}
		}
		localization_key = TNO_MILITARY_ADVISORS_EMPTY
	}
	text = {
		localization_key = TNO_MILITARY_ADVISORS_COST_SHOWN
	}
}

# the next level's price, in billions from 1000 M up
defined_text = {
	name = TNO_GetMilitaryAdvisorsPrice
	text = {
		trigger = { check_variable = { TNO_military_advisors_price > 999.999 } }
		localization_key = TNO_MILITARY_ADVISORS_PRICE_B

	text = {
		trigger = { check_variable = { TNO_military_advisors_price > 9.999 } }

	}

		localization_key = TNO_MILITARY_ADVISORS_PRICE_M_SMALL
	}
}
